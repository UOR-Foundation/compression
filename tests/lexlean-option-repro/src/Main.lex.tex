  name: toEvent,
  parameters: [chosen : (Option UInt8)],

    | Option.none => (ctor Event.missing)
    | Option.some byteValue => (ctor Event.present byteValue)),

  name: observe,
  parameters: [eventValue : Event],
  result: (Option UInt8),

    | Event.missing => (ctor Option.none [UInt8])
    | Event.present byteValue => (ctor Option.some [UInt8] byteValue)),
  axioms: []
}

  name: observation_round_trip,
  parameters: [chosen : (Option UInt8)],
  statement: ((observe (toEvent chosen)) = chosen),
  proof: term_cases{definitions: [observe, toEvent], scrutinee: chosen},
  axioms: [propext]
