definition{
  name: flatten,
  parameters: [rope : Uorc.Core.Bytes::Rope],
  result: Bytes,
  body: (match rope
    | Uorc.Core.Bytes::Rope.empty => 0x
    | Uorc.Core.Bytes::Rope.leaf chunk => chunk
    | Uorc.Core.Bytes::Rope.branch size depth leftTree rightTree => (prim append (flatten leftTree) (flatten rightTree) : Bytes)
    | Uorc.Core.Bytes::Rope.view backing offset count => (match (prim slice backing offset count : (Option Bytes))
      | Option.none => 0x
      | Option.some chunk => chunk)),
  axioms: [],
  recursive_argument: rope
}

definition{
  name: drive,
  parameters: [work : Nat, state : Uorc.Search.Engine::Step, best : (Option Bytes)],
  result: (Option Bytes),
  body: (match work
    | Nat.zero => best
    | Nat.succ remaining => (match state
      | Uorc.Search.Engine::Step.advanced cursor => (drive remaining (Uorc.Search.Engine::step cursor) best)
      | Uorc.Search.Engine::Step.improved eligible cursor => (drive remaining (Uorc.Search.Engine::step cursor) (ctor Option.some [Bytes] (archiveBytes eligible)))
      | Uorc.Search.Engine::Step.completed outcome => (ctor Option.some [Bytes] (archiveBytes (project outcome best)))
      | Uorc.Search.Engine::Step.blocked failure cursor => best)),

  recursive_argument: work
}

definition{

  result: (Option Bytes),
  body: (let codec : Nat := (if (Uorc.Codec.Archive::x86Dense input) then 2 else 1); (let linked : (Option Bytes) := (Uorc.Codec.Archive::windowCandidate codec window input); (let quick : (Option Bytes) := (match linked
    | Option.none => (Uorc.Codec.Archive::rawArchive input)
    | Option.some kernel => (if ((prim length kernel : Nat) <? (rawLength (prim length input : Nat))) then (ctor Option.some [Bytes] kernel) else (Uorc.Codec.Archive::rawArchive input))); (if (steps =? 0) then quick else (smaller quick (search input steps)))))),

definition{
  name: decodedOutput,
  parameters: [work : Nat, state : Uorc.Search.ArchiveTrial::Step],
  result: (Option Bytes),
  body: (match work
    | Nat.zero => (ctor Option.none [Bytes])
    | Nat.succ remaining => (match state
      | Uorc.Search.ArchiveTrial::Step.pending cursor => (match (project cursor phase)
        | Uorc.Search.ArchiveTrial::Phase.splitting event reader ended output => (ctor Option.some [Bytes] (flatten output))
        | Uorc.Search.ArchiveTrial::Phase.checking verifier => (ctor Option.none [Bytes])
        | Uorc.Search.ArchiveTrial::Phase.normalizing inner => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.building inner => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.parsing event reader ended => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.hashing inner => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.boundaries inner => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.validating inner => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.admitting graph basis => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.basisIdentity graph table identity => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.executing graph table credit => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.rawDebt raw debt => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.boundaryDebt raw objects => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor))
        | Uorc.Search.ArchiveTrial::Phase.basisNormalizing inner raw => (decodedOutput remaining (Uorc.Search.ArchiveTrial::step cursor)))
      | Uorc.Search.ArchiveTrial::Step.complete eligible => (ctor Option.none [Bytes])
      | Uorc.Search.ArchiveTrial::Step.rejected failure usage ledger events => (ctor Option.none [Bytes]))),
  axioms: [Quot.sound, propext],
  recursive_argument: work

    | Option.some output => (ctor Option.some [Bytes] output)
    | Option.none => (match window
      | List.nil => (referenceDecode (match (Uorc.Codec.Archive::materialize archive)
        | Option.none => archive

      | List.cons oldest rest => (match (Uorc.Codec.Archive::windowMaterialize archive window)
        | Option.none => (ctor Option.none [Bytes])
        | Option.some whole => (match (Uorc.Codec.Archive::windowBasis archive window)
          | Option.none => (ctor Option.none [Bytes])
          | Option.some named => (referenceDecode whole (ctor Uorc.Wire.Frame::Scope.conditional (first named)) (cons (Uorc.Core.Bytes::fromBytes (second named)) (windowRopes window)) work))))),
  axioms: [Quot.sound, propext]
