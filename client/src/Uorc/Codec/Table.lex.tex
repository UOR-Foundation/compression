definition{
  name: readByte,
  parameters: [t : Bytes, at : Nat],
  result: Nat,
  body: (match (prim index t at : (Option UInt8))
    | Option.none => 0
    | Option.some b => (prim to_nat b : Nat)),
  executable: true,
  axioms: [Quot.sound, propext]
}

definition{
  name: writeByte,
  parameters: [t : Bytes, at : Nat, v : Nat],
  result: Bytes,
  body: (match (prim update t at (byteOf v) : (Option Bytes))
    | Option.none => 0x
    | Option.some u => u),
  executable: true,
  axioms: [Quot.sound, propext]
}

definition{
  name: getLoop,
  parameters: [fuel : Nat, t : Bytes, at : Nat, acc : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => acc
    | Nat.succ k => (let b : Nat := (readByte t at); (getLoop k t (at + 1) ((prim multiply 256 acc : Nat) + b)))),
  executable: true,
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: get,
  parameters: [width : Nat, t : Bytes, key : Nat],
  result: Nat,
  body: (getLoop width t (prim multiply width key : Nat) 0),
  executable: true,
  axioms: [Quot.sound, propext]
}

definition{
  name: setLoop,
  parameters: [fuel : Nat, t : Bytes, at : Nat, v : Nat],

  body: (match fuel

  axioms: [Quot.sound, propext],
  recursive_argument: fuel

definition{
  name: set,
  parameters: [width : Nat, t : Bytes, key : Nat, v : Nat],
  result: Bytes,
  body: (setLoop width t (prim subtract ((prim multiply width key : Nat) + width) 1 : Nat) v),
  executable: true,
  axioms: [Quot.sound, propext]
}

definition{
  name: double,
  parameters: [fuel : Nat, b : Bytes],
  result: Bytes,
  body: (match fuel
    | Nat.zero => b
    | Nat.succ k => (double k (prim append b b : Bytes))),
  executable: true,
  axioms: [],

}

definition{
  name: zeros,
  parameters: [width : Nat],
  result: Bytes,
  body: (match (prim slice 0x00000000 0 width : (Option Bytes))
    | Option.none => 0x
    | Option.some z => z),
  executable: true,
  axioms: []
}

definition{

  axioms: [Quot.sound, propext]
