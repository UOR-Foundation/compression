\begin{lexlean}{Uorc.Diagnostic.RecursiveScrutinee}
\title{Parameters}
\begin{semanticmodule}
inductive{
  name: Tree,
  type_parameters: [],
  parameters: [],
  constructors: [{name: leaf, fields: []}, {name: branch, fields: [Tree]}]
}

definition{
  name: observe,
  parameters: [tree : Tree],
  result: Bool,
  body: true,
  axioms: [],
  executable: true
}

definition{
  name: walk,
  parameters: [tree : Tree],
  result: Bool,
  body: (match tree
    | Tree.leaf => true
    | Tree.branch child => (if (observe tree) then (walk child) else false)),
  axioms: [],
  executable: true,
  recursive_argument: tree
}
\end{semanticmodule}
\end{lexlean}
