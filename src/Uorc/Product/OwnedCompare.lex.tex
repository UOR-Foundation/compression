definition{
  name: load,
  parameters: [arena : Uorc.Core.Heap::Arena, side : Side, reader : Reader],
  result: Load,
  body: (match (Uorc.Wire.OwnedRead::step arena (project reader cursor))
    | Uorc.Wire.OwnedRead::Step.advanced next => (record Load {reader := (record Reader {cursor := next, span := (project reader span), ended := (project reader ended)}), failure := (ctor Option.none [Failure])})
    | Uorc.Wire.OwnedRead::Step.borrowedSpan image offset length next => (if (prim equal length (0 : UInt64) : Bool) then (record Load {reader := reader, failure := (ctor Option.some [Failure] (ctor Failure.invalidSpan))}) else (record Load {reader := (record Reader {cursor := next, span := (ctor Option.some [Span] (record Span {image := image, offset := offset, length := length, used := (0 : UInt64)})), ended := (project reader ended)}), failure := (ctor Option.none [Failure])}))
    | Uorc.Wire.OwnedRead::Step.complete => (record Load {reader := (record Reader {cursor := (project reader cursor), span := (project reader span), ended := true}), failure := (ctor Option.none [Failure])})
    | Uorc.Wire.OwnedRead::Step.failed failure => (record Load {reader := reader, failure := (ctor Option.some [Failure] (ctor Failure.read side failure))})),
  axioms: [Quot.sound, propext]
}

definition{
  name: prepareByte,
  parameters: [s : State, leftSpan : Span, rightSpan : Span],
  result: (Result BytePlan Failure),
  body: (match (prim checked_add (project leftSpan offset) (project leftSpan used) : (Option UInt64))
    | Option.none => (ctor Result.error [BytePlan, Failure] (ctor Failure.counterExhausted))
    | Option.some leftIndex => (match (prim checked_add (project rightSpan offset) (project rightSpan used) : (Option UInt64))
      | Option.none => (ctor Result.error [BytePlan, Failure] (ctor Failure.counterExhausted))
      | Option.some rightIndex => (match (prim checked_add (project leftSpan used) (1 : UInt64) : (Option UInt64))
        | Option.none => (ctor Result.error [BytePlan, Failure] (ctor Failure.counterExhausted))
        | Option.some leftUsed => (match (prim checked_add (project rightSpan used) (1 : UInt64) : (Option UInt64))
          | Option.none => (ctor Result.error [BytePlan, Failure] (ctor Failure.counterExhausted))
          | Option.some rightUsed => (match (prim checked_add (project s compared) (1 : UInt64) : (Option UInt64))
            | Option.none => (ctor Result.error [BytePlan, Failure] (ctor Failure.counterExhausted))
            | Option.some compared => (ctor Result.ok [BytePlan, Failure] (record BytePlan {leftSpan := leftSpan, rightSpan := rightSpan, leftIndex := leftIndex, rightIndex := rightIndex, leftUsed := leftUsed, rightUsed := rightUsed, nextCompared := compared}))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: completeByte,
  parameters: [s : State, observation : PendingByte, rightByte : UInt8],
  result: State,
  body: (if (prim equal (project observation leftByte) rightByte : Bool) then (record State {left := (advanceSpan (project s left) (project (project observation plan) leftSpan) (project (project observation plan) leftUsed)), right := (advanceSpan (project s right) (project (project observation plan) rightSpan) (project (project observation plan) rightUsed)), compared := (project (project observation plan) nextCompared), result := (project s result), pending := (ctor Option.none [PendingByte])}) else (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.different (project s compared)), pending := (ctor Option.none [PendingByte])}))
}

definition{
  name: finishByte,
  parameters: [arena : Uorc.Core.Heap::Arena, s : State, observation : PendingByte],
  result: State,
  body: (match (Uorc.Core.RuntimeImage::readByte arena (project (project (project observation plan) rightSpan) image) (project (project observation plan) rightIndex))
    | Result.error failure => (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.failed (ctor Failure.byte (ctor Side.right) failure)), pending := (ctor Option.none [PendingByte])})
    | Result.ok rightByte => (completeByte s observation rightByte)),
  axioms: [Quot.sound, propext]
}

definition{
  name: compareByte,
  parameters: [arena : Uorc.Core.Heap::Arena, s : State, leftSpan : Span, rightSpan : Span],
  result: State,
  body: (match (prepareByte s leftSpan rightSpan)
    | Result.error failure => (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.failed failure), pending := (ctor Option.none [PendingByte])})
    | Result.ok plan => (match (Uorc.Core.RuntimeImage::readByte arena (project (project plan leftSpan) image) (project plan leftIndex))
      | Result.error failure => (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.failed (ctor Failure.byte (ctor Side.left) failure)), pending := (ctor Option.none [PendingByte])})
      | Result.ok leftByte => (finishByte arena s (record PendingByte {plan := plan, leftByte := leftByte})))),
  axioms: [Quot.sound, propext]
}

definition{
  name: step,
  parameters: [arena : Uorc.Core.Heap::Arena, s : State],
  result: State,
  body: (match (project s result)
    | Disposition.working => (match (project s pending)
      | Option.none => (match (project (project s left) span)
        | Option.none => (if (project (project s left) ended) then (match (project (project s right) span)
          | Option.some rightSpan => (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.lengthMismatch (project (project (project s left) cursor) length) (project (project (project s right) cursor) length)), pending := (project s pending)})
          | Option.none => (if (project (project s right) ended) then (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.equal), pending := (project s pending)}) else (loaded s (ctor Side.right) (load arena (ctor Side.right) (project s right))))) else (loaded s (ctor Side.left) (load arena (ctor Side.left) (project s left))))
        | Option.some leftSpan => (match (project (project s right) span)
          | Option.none => (if (project (project s right) ended) then (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.lengthMismatch (project (project (project s left) cursor) length) (project (project (project s right) cursor) length)), pending := (project s pending)}) else (loaded s (ctor Side.right) (load arena (ctor Side.right) (project s right))))
          | Option.some rightSpan => (compareByte arena s leftSpan rightSpan)))
      | Option.some observation => (finishByte arena s observation))
    | Disposition.equal => s
    | Disposition.different index => s
    | Disposition.lengthMismatch left right => s
    | Disposition.failed failure => s
    | Disposition.cancelled => s),
  axioms: [Quot.sound, propext]
}

definition{
  name: paidReadRight,
  parameters: [
    budget : Uorc.Core.RuntimeResources::Budget,
    usage : Uorc.Core.RuntimeResources::Usage,
    arena : Uorc.Core.Heap::Arena,
    s : State,
    observation : PendingByte
  ],
  result: Tick,
  body: (match (Uorc.Core.RuntimeResources::charge budget usage (record Uorc.Core.RuntimeResources::Event {work := (Uorc.Core.RuntimeImage::readByteCost arena (project (project (project observation plan) rightSpan) image) (project (project observation plan) rightIndex)), emittedBytes := (0 : UInt64), enteredFrames := (0 : UInt64), exitedFrames := (0 : UInt64)}))
    | Result.error resourceFailure => (ctor Tick.refused resourceFailure s usage)
    | Result.ok rightUsage => (match (Uorc.Core.RuntimeImage::readByte arena (project (project (project observation plan) rightSpan) image) (project (project observation plan) rightIndex))
      | Result.error failure => (ctor Tick.advanced (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.failed (ctor Failure.byte (ctor Side.right) failure)), pending := (ctor Option.none [PendingByte])}) rightUsage)
      | Result.ok rightByte => (ctor Tick.advanced (completeByte s observation rightByte) rightUsage))),
  axioms: [Quot.sound, propext]
}

      | Result.ok leftByte => (paidReadRight budget leftUsage arena (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (project s result), pending := (ctor Option.some [PendingByte] (record PendingByte {plan := plan, leftByte := leftByte}))}) (record PendingByte {plan := plan, leftByte := leftByte})))),

definition{
  name: paidStep,
  parameters: [
    budget : Uorc.Core.RuntimeResources::Budget,
    usage : Uorc.Core.RuntimeResources::Usage,
    arena : Uorc.Core.Heap::Arena,
    s : State
  ],
  result: Tick,
  body: (match (project (project s left) span)
    | Option.none => (if (project (project s left) ended) then (match (project (project s right) span)
      | Option.some rightSpan => (ctor Tick.advanced (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.lengthMismatch (project (project (project s left) cursor) length) (project (project (project s right) cursor) length)), pending := (project s pending)}) usage)
      | Option.none => (if (project (project s right) ended) then (ctor Tick.advanced (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.equal), pending := (project s pending)}) usage) else (ctor Tick.advanced (loaded s (ctor Side.right) (load arena (ctor Side.right) (project s right))) usage))) else (ctor Tick.advanced (loaded s (ctor Side.left) (load arena (ctor Side.left) (project s left))) usage))
    | Option.some leftSpan => (match (project (project s right) span)
      | Option.none => (if (project (project s right) ended) then (ctor Tick.advanced (record State {left := (project s left), right := (project s right), compared := (project s compared), result := (ctor Disposition.lengthMismatch (project (project (project s left) cursor) length) (project (project (project s right) cursor) length)), pending := (project s pending)}) usage) else (ctor Tick.advanced (loaded s (ctor Side.right) (load arena (ctor Side.right) (project s right))) usage))
      | Option.some rightSpan => (paidCompareByte budget usage arena s leftSpan rightSpan))),
  axioms: [Quot.sound, propext]
}

definition{
  name: tickWork,
  parameters: [s : State],
  result: UInt64,
  body: (match (project (project s left) span)
    | Option.none => (if (project (project s left) ended) then (match (project (project s right) span)
      | Option.some rightSpan => (stepWork)
      | Option.none => (if (project (project s right) ended) then (stepWork) else (traversalWork))) else (traversalWork))
    | Option.some leftSpan => (match (project (project s right) span)
      | Option.some rightSpan => (stepWork)
      | Option.none => (if (project (project s right) ended) then (stepWork) else (traversalWork))))
}

definition{
  name: tick,
  parameters: [
    budget : Uorc.Core.RuntimeResources::Budget,
    usage : Uorc.Core.RuntimeResources::Usage,
    arena : Uorc.Core.Heap::Arena,
    s : State
  ],
  result: Tick,
  body: (match (project s result)
    | Disposition.working => (match (project s pending)
      | Option.none => (match (Uorc.Core.RuntimeResources::charge budget usage (record Uorc.Core.RuntimeResources::Event {work := (tickWork s), emittedBytes := (0 : UInt64), enteredFrames := (0 : UInt64), exitedFrames := (0 : UInt64)}))
        | Result.error failure => (ctor Tick.refused failure s usage)
        | Result.ok paidUsage => (paidStep budget paidUsage arena s))
      | Option.some observation => (paidReadRight budget usage arena s observation))
    | Disposition.equal => (ctor Tick.advanced s usage)
    | Disposition.different index => (ctor Tick.advanced s usage)
    | Disposition.lengthMismatch left right => (ctor Tick.advanced s usage)
    | Disposition.failed failure => (ctor Tick.advanced s usage)
    | Disposition.cancelled => (ctor Tick.advanced s usage)),
  axioms: [Quot.sound, propext]
}
