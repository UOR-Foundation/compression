definition{
  name: accept,
  parameters: [s : State, piece : Piece],
  result: (Result State Error),
  body: (if (project s ended) then (ctor Result.error [State, Error] (ctor Error.afterEnd)) else (match piece
    | Piece.beginObject index => (if (project s inObject) then (ctor Result.error [State, Error] (ctor Error.objectAlreadyOpen)) else (if (index =? (project s nextObject)) then (ctor Result.ok [State, Error] (record State {nextObject := (project s nextObject), inObject := true, ended := (project s ended), totalBytes := (project s totalBytes)})) else (ctor Result.error [State, Error] (ctor Error.wrongObject))))
    | Piece.chunk span => (if (project s inObject) then (ctor Result.ok [State, Error] (record State {nextObject := (project s nextObject), inObject := (project s inObject), ended := (project s ended), totalBytes := ((project s totalBytes) + (project span length))})) else (ctor Result.error [State, Error] (ctor Error.noObject)))
    | Piece.endObject index => (if (project s inObject) then (if (index =? (project s nextObject)) then (ctor Result.ok [State, Error] (record State {nextObject := ((project s nextObject) + 1), inObject := false, ended := (project s ended), totalBytes := (project s totalBytes)})) else (ctor Result.error [State, Error] (ctor Error.wrongObject))) else (ctor Result.error [State, Error] (ctor Error.noObject)))
    | Piece.endInput => (if (project s inObject) then (ctor Result.error [State, Error] (ctor Error.objectStillOpen)) else (ctor Result.ok [State, Error] (record State {nextObject := (project s nextObject), inObject := (project s inObject), ended := true, totalBytes := (project s totalBytes)})))))
}
