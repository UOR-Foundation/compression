definition{
  name: countEffect,
  parameters: [trace : Trace, effect : Uorc.Product.Control::Effect],
  result: Trace,
  body: (match effect
    | Uorc.Product.Control::Effect.start run request => (record Trace {state := (project trace state), starts := ((project trace starts) + 1), cancels := (project trace cancels), discards := (project trace discards), releases := (project trace releases), publishes := (project trace publishes), extensions := (project trace extensions)})
    | Uorc.Product.Control::Effect.feed run piece => trace
    | Uorc.Product.Control::Effect.tick run credit => trace
    | Uorc.Product.Control::Effect.extend run credit => (record Trace {state := (project trace state), starts := (project trace starts), cancels := (project trace cancels), discards := (project trace discards), releases := (project trace releases), publishes := (project trace publishes), extensions := ((project trace extensions) + 1)})
    | Uorc.Product.Control::Effect.cancel run => (record Trace {state := (project trace state), starts := (project trace starts), cancels := ((project trace cancels) + 1), discards := (project trace discards), releases := (project trace releases), publishes := (project trace publishes), extensions := (project trace extensions)})
    | Uorc.Product.Control::Effect.discard run => (record Trace {state := (project trace state), starts := (project trace starts), cancels := (project trace cancels), discards := ((project trace discards) + 1), releases := (project trace releases), publishes := (project trace publishes), extensions := (project trace extensions)})
    | Uorc.Product.Control::Effect.release resource => (record Trace {state := (project trace state), starts := (project trace starts), cancels := (project trace cancels), discards := (project trace discards), releases := ((project trace releases) + 1), publishes := (project trace publishes), extensions := (project trace extensions)})
    | Uorc.Product.Control::Effect.publish run resource => (record Trace {state := (project trace state), starts := (project trace starts), cancels := (project trace cancels), discards := (project trace discards), releases := (project trace releases), publishes := ((project trace publishes) + 1), extensions := (project trace extensions)})
    | Uorc.Product.Control::Effect.requestInstanceRelease instance => trace
    | Uorc.Product.Control::Effect.cancelPublication run => trace)
}

definition{
  name: countEffects,
  parameters: [trace : Trace, effects : (List Uorc.Product.Control::Effect)],
  result: Trace,
  body: (match effects
    | List.nil => trace
    | List.cons head tail => (countEffects (countEffect trace head) tail)),
  recursive_argument: effects
}

definition{
  name: advance,
  parameters: [trace : Trace, transition : Uorc.Product.Control::Transition],
  result: Trace,
  body: (countEffects (record Trace {state := (project transition state), starts := (project trace starts), cancels := (project trace cancels), discards := (project trace discards), releases := (project trace releases), publishes := (project trace publishes), extensions := (project trace extensions)}) (project transition effects))
}

definition{
  name: runTrace,
  parameters: [trace : Trace, events : (List Uorc.Product.Control::Event)],
  result: Trace,
  body: (match events
    | List.nil => trace
    | List.cons head tail => (runTrace (advance trace (Uorc.Product.Control::step (project trace state) head)) tail)),
  recursive_argument: events
}
