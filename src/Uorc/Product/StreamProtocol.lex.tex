definition{
  name: acceptInput,
  parameters: [s : State, input : Input],
  result: Decision,
  body: (match input
    | Input.tick => (match (project s phase)
      | Phase.working => (ctor Decision.accepted (ctor InputAction.run) s)
      | Phase.awaitingInput => (ctor Decision.rejected (ctor ProtocolError.notWorking))
      | Phase.complete => (ctor Decision.rejected (ctor ProtocolError.notWorking))
      | Phase.cancelled => (ctor Decision.rejected (ctor ProtocolError.notWorking))
      | Phase.failed => (ctor Decision.rejected (ctor ProtocolError.notWorking)))
    | Input.chunk => (match (project s phase)
      | Phase.working => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested))
      | Phase.awaitingInput => (if (project s ended) then (ctor Decision.rejected (ctor ProtocolError.endAlreadySeen)) else (ctor Decision.accepted (ctor InputAction.admitChunk) (record State {phase := (ctor Phase.working), ended := (project s ended), producedBytes := (project s producedBytes), verified := (project s verified), publication := (project s publication)})))
      | Phase.complete => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested))
      | Phase.cancelled => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested))
      | Phase.failed => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested)))
    | Input.endOfInput => (match (project s phase)
      | Phase.working => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested))
      | Phase.awaitingInput => (if (project s ended) then (ctor Decision.rejected (ctor ProtocolError.endAlreadySeen)) else (ctor Decision.accepted (ctor InputAction.sealInput) (record State {phase := (ctor Phase.working), ended := true, producedBytes := (project s producedBytes), verified := (project s verified), publication := (project s publication)})))
      | Phase.complete => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested))
      | Phase.cancelled => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested))
      | Phase.failed => (ctor Decision.rejected (ctor ProtocolError.inputNotRequested)))
    | Input.cancel => (match (project s phase)
      | Phase.working => (ctor Decision.accepted (ctor InputAction.cancel) (record State {phase := (ctor Phase.cancelled), ended := (project s ended), producedBytes := (project s producedBytes), verified := false, publication := (ctor Publication.notPublished)}))
      | Phase.awaitingInput => (ctor Decision.accepted (ctor InputAction.cancel) (record State {phase := (ctor Phase.cancelled), ended := (project s ended), producedBytes := (project s producedBytes), verified := false, publication := (ctor Publication.notPublished)}))
      | Phase.complete => (ctor Decision.accepted (ctor InputAction.ignoreTerminal) s)
      | Phase.cancelled => (ctor Decision.accepted (ctor InputAction.ignoreTerminal) s)
      | Phase.failed => (ctor Decision.accepted (ctor InputAction.ignoreTerminal) s)))
}

definition{
  name: requestInput,
  parameters: [s : State],
  result: (Result State ProtocolError),
  body: (match (project s phase)
    | Phase.working => (if (project s ended) then (ctor Result.error [State, ProtocolError] (ctor ProtocolError.endAlreadySeen)) else (ctor Result.ok [State, ProtocolError] (record State {phase := (ctor Phase.awaitingInput), ended := (project s ended), producedBytes := (project s producedBytes), verified := (project s verified), publication := (project s publication)})))
    | Phase.awaitingInput => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.complete => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.cancelled => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.failed => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking)))
}

definition{
  name: stageOutput,
  parameters: [s : State, bytesProduced : Nat, limit : Nat],
  result: (Result State ProtocolError),
  body: (match (project s phase)
    | Phase.working => (if (((project s producedBytes) <=? limit) && ((bytesProduced <=? (prim subtract limit (project s producedBytes) : Nat)) && true)) then (ctor Result.ok [State, ProtocolError] (record State {phase := (project s phase), ended := (project s ended), producedBytes := ((project s producedBytes) + bytesProduced), verified := (project s verified), publication := (project s publication)})) else (ctor Result.error [State, ProtocolError] (ctor ProtocolError.outputLimit)))
    | Phase.awaitingInput => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.complete => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.cancelled => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.failed => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking)))
}

definition{
  name: completeVerified,
  parameters: [s : State],
  result: (Result State ProtocolError),
  body: (match (project s phase)
    | Phase.working => (if (project s ended) then (ctor Result.ok [State, ProtocolError] (record State {phase := (ctor Phase.complete), ended := (project s ended), producedBytes := (project s producedBytes), verified := true, publication := (project s publication)})) else (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notReady)))
    | Phase.awaitingInput => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.complete => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.cancelled => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking))
    | Phase.failed => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notWorking)))
}

definition{
  name: canCommit,
  parameters: [s : State],
  result: Bool,
  body: (match (project s phase)
    | Phase.working => false
    | Phase.awaitingInput => false
    | Phase.complete => ((project s verified) && ((match (project s publication)
      | Publication.staged => true
      | Publication.published => false
      | Publication.notPublished => false
      | Publication.unknown => false) && true))
    | Phase.cancelled => false
    | Phase.failed => false)
}

definition{
  name: recordPublication,
  parameters: [s : State, outcome : Publication],
  result: (Result State ProtocolError),
  body: (if (canCommit s) then (match outcome
    | Publication.staged => (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notReady))
    | Publication.published => (ctor Result.ok [State, ProtocolError] (record State {phase := (project s phase), ended := (project s ended), producedBytes := (project s producedBytes), verified := (project s verified), publication := (ctor Publication.published)}))
    | Publication.notPublished => (ctor Result.ok [State, ProtocolError] (record State {phase := (project s phase), ended := (project s ended), producedBytes := (project s producedBytes), verified := (project s verified), publication := (ctor Publication.notPublished)}))
    | Publication.unknown => (ctor Result.ok [State, ProtocolError] (record State {phase := (project s phase), ended := (project s ended), producedBytes := (project s producedBytes), verified := (project s verified), publication := (ctor Publication.unknown)}))) else (ctor Result.error [State, ProtocolError] (ctor ProtocolError.notReady)))
}

definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "The source controller uses this protocol around its actual operation state. An input chunk is accepted only on demand; EOF is one-shot; output remains staged until a verified completed operation permits host publication. completeVerified is called only by the independently verified operation path, never by an untrusted request flag. Atomic no-replace file publication, stdout partial-write risk, cancellation cleanup, and host outcome unknown remain distinct adapter obligations. This protocol does not itself verify archive content or prove host filesystem behavior."
