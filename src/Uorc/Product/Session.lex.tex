\begin{lexlean}{Uorc.Product.Session}
\useglossary{lexlean.std.bool@1.2.0}
\title{Boolean}
\begin{semanticmodule}
inductive{
  name: Phase,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: idle, fields: []},
    {name: admitting, fields: []},
    {name: searching, fields: []},
    {name: verifying, fields: []},
    {name: complete, fields: []},
    {name: cancelled, fields: []},
    {name: failed, fields: []}
  ]
}

inductive{
  name: ResultKind,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: none, fields: []},
    {name: inspection, fields: []},
    {name: checkedArchive, fields: []},
    {name: checkedReconstruction, fields: []},
    {name: checkedMinimum, fields: []},
    {name: failure, fields: []}
  ]
}

structure{
  name: Session,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: revision, type: Nat},
    {name: serial, type: Nat},
    {name: active, type: Bool},
    {name: hasResult, type: Bool},
    {name: phase, type: Phase},
    {name: result, type: ResultKind}
  ]
}

definition{
  name: initial,
  parameters: [],
  result: Session,
  body: (record Session {revision := 0, serial := 0, active := false, hasResult := false, phase := (ctor Phase.idle), result := (ctor ResultKind.none)})
}

definition{
  name: begin,
  parameters: [s : Session],
  result: Session,
  body: (record Session {revision := (project s revision), serial := ((project s serial) + 1), active := true, hasResult := false, phase := (ctor Phase.admitting), result := (ctor ResultKind.none)})
}

definition{
  name: edit,
  parameters: [s : Session],
  result: Session,
  body: (record Session {revision := ((project s revision) + 1), serial := (project s serial), active := false, hasResult := false, phase := (ctor Phase.idle), result := (ctor ResultKind.none)})
}

definition{
  name: cancel,
  parameters: [s : Session],
  result: Session,
  body: (if (project s active) then (record Session {revision := (project s revision), serial := (project s serial), active := false, hasResult := false, phase := (ctor Phase.cancelled), result := (ctor ResultKind.none)}) else s)
}

definition{
  name: ownsRun,
  parameters: [s : Session, runSerial : Nat, runRevision : Nat],
  result: Bool,
  body: ((project s active) && (((project s serial) =? runSerial) && (((project s revision) =? runRevision) && true))),
  executable: true
}

definition{
  name: resultPhase,
  parameters: [result : ResultKind],
  result: Phase,
  body: (match result
    | ResultKind.none => (ctor Phase.failed)
    | ResultKind.inspection => (ctor Phase.complete)
    | ResultKind.checkedArchive => (ctor Phase.complete)
    | ResultKind.checkedReconstruction => (ctor Phase.complete)
    | ResultKind.checkedMinimum => (ctor Phase.complete)
    | ResultKind.failure => (ctor Phase.failed)),
  executable: true
}

definition{
  name: finish,
  parameters: [s : Session, runSerial : Nat, runRevision : Nat, result : ResultKind],
  result: Session,
  body: (match (ownsRun s runSerial runRevision)
    | Bool.false => s
    | Bool.true => (record Session {revision := (project s revision), serial := (project s serial), active := false, hasResult := true, phase := (resultPhase result), result := result})),
  executable: true
}

definition{
  name: hasDownload,
  parameters: [s : Session],
  result: Bool,
  body: ((project s hasResult) && ((match (project s result)
    | ResultKind.none => false
    | ResultKind.inspection => false
    | ResultKind.checkedArchive => true
    | ResultKind.checkedReconstruction => true
    | ResultKind.checkedMinimum => true
    | ResultKind.failure => false) && true))
}

theorem{
  name: initial_has_no_download,
  parameters: [],
  statement: ((hasDownload (initial)) = false),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: first_run_serial,
  parameters: [],
  statement: ((project (begin (initial)) serial) = 1),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: edit_invalidates_running_result,
  parameters: [],
  statement: ((finish (edit (begin (initial))) 1 0 (ctor ResultKind.checkedArchive)) = (edit (begin (initial)))),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: cancel_invalidates_running_result,
  parameters: [],
  statement: ((finish (cancel (begin (initial))) 1 0 (ctor ResultKind.checkedArchive)) = (cancel (begin (initial)))),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: retry_invalidates_previous_result,
  parameters: [],
  statement: ((finish (begin (begin (initial))) 1 0 (ctor ResultKind.checkedArchive)) = (begin (begin (initial)))),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: current_checked_result_has_download,
  parameters: [],
  statement: ((hasDownload (finish (begin (initial)) 1 0 (ctor ResultKind.checkedArchive))) = true),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: inspection_has_no_archive_download,
  parameters: [],
  statement: ((hasDownload (finish (begin (initial)) 1 0 (ctor ResultKind.inspection))) = false),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: failure_has_no_archive_download,
  parameters: [],
  statement: ((hasDownload (finish (begin (initial)) 1 0 (ctor ResultKind.failure))) = false),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: late_wrong_revision_is_ignored,
  parameters: [],
  statement: ((finish (begin (initial)) 1 1 (ctor ResultKind.checkedArchive)) = (begin (initial))),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: completion_is_idempotent,
  parameters: [],
  statement: ((finish (finish (begin (initial)) 1 0 (ctor ResultKind.checkedArchive)) 1 0 (ctor ResultKind.failure)) = (finish (begin (initial)) 1 0 (ctor ResultKind.checkedArchive))),
  proof: reflexivity{},
  axioms: []
}
\end{semanticmodule}
\end{lexlean}
