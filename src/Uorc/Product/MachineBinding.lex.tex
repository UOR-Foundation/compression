definition{
  name: finishDrain,
  parameters: [
    s : State,
    core : Uorc.Core.RuntimeMachineEngine::State,
    output : (Option Uorc.Core.RuntimeValue::Handle)
  ],
  result: State,
  body: (match (project core disposition)
    | Uorc.Core.RuntimeMachine::Disposition.running => (record State {run := (project s run), owner := (ctor Owner.draining core output), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.suspended reason => (record State {run := (project s run), owner := (ctor Owner.draining core output), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.completed handle => (record State {run := (project s run), owner := (ctor Owner.draining core output), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.failed reason => (record State {run := (project s run), owner := (ctor Owner.draining core output), stopped := (project s stopped), failure := (ctor Option.some [Uorc.Core.RuntimeMachine::Failure] reason), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.cancelling => (record State {run := (project s run), owner := (ctor Owner.draining core output), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.cleaned => (match (Uorc.Core.RuntimeMachineEngine::takeReady core)
      | Result.error failure => (record State {run := (project s run), owner := (ctor Owner.draining core output), stopped := (project s stopped), failure := (ctor Option.some [Uorc.Core.RuntimeMachine::Failure] failure), receipt := (project s receipt)})
      | Result.ok ready => (record State {run := (project s run), owner := (ctor Owner.terminal ready output), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)}))),
  axioms: []
}

definition{
  name: afterTick,
  parameters: [s : State, core : Uorc.Core.RuntimeMachineEngine::State],
  result: State,
  body: (match (project core disposition)
    | Uorc.Core.RuntimeMachine::Disposition.running => (record State {run := (project s run), owner := (ctor Owner.running core), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.suspended reason => (record State {run := (project s run), owner := (ctor Owner.running core), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.completed handle => (match (Uorc.Core.RuntimeMachineEngine::takeOutput core)
      | Result.error reason => (record State {run := (project s run), owner := (ctor Owner.draining (Uorc.Core.RuntimeMachineEngine::cancel core) (ctor Option.none [Uorc.Core.RuntimeValue::Handle])), stopped := (project s stopped), failure := (ctor Option.some [Uorc.Core.RuntimeMachine::Failure] reason), receipt := (project s receipt)})
      | Result.ok transfer => (record State {run := (project s run), owner := (ctor Owner.draining (Uorc.Core.RuntimeMachineEngine::cancel (project transfer state)) (ctor Option.some [Uorc.Core.RuntimeValue::Handle] (project transfer handle))), stopped := (project s stopped), failure := (project s failure), receipt := (project s receipt)}))
    | Uorc.Core.RuntimeMachine::Disposition.failed reason => (record State {run := (project s run), owner := (ctor Owner.draining (Uorc.Core.RuntimeMachineEngine::cancel core) (ctor Option.none [Uorc.Core.RuntimeValue::Handle])), stopped := (project s stopped), failure := (ctor Option.some [Uorc.Core.RuntimeMachine::Failure] reason), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.cancelling => (record State {run := (project s run), owner := (ctor Owner.draining core (ctor Option.none [Uorc.Core.RuntimeValue::Handle])), stopped := true, failure := (project s failure), receipt := (project s receipt)})
    | Uorc.Core.RuntimeMachine::Disposition.cleaned => (finishDrain s core (ctor Option.none [Uorc.Core.RuntimeValue::Handle]))),
  axioms: []
}

definition{
  name: tick,
  parameters: [s : State, run : Uorc.Product.Api::Run],
  result: State,
  body: (if (sameRun (project s run) run) then (match (project s owner)
    | Owner.running core => (afterTick s (Uorc.Core.RuntimeMachineEngine::tick core))
    | Owner.draining core output => (finishDrain s (Uorc.Core.RuntimeMachineEngine::tick core) output)
    | Owner.terminal ready output => s
    | Owner.handedOff bindingUsage => s) else s),
  axioms: [Quot.sound, propext]
}

definition{
  name: takeCoreReceipt,

definition{

  parameters: [s : State, run : Uorc.Product.Api::Run],

    | Owner.running core => (ctor Movement.notReady s)
    | Owner.draining core output => (ctor Movement.notReady s)
    | Owner.terminal ready output => (ctor Movement.moved (record State {run := (project s run), owner := (ctor Owner.handedOff (project ready usage)), stopped := (project s stopped), failure := (project s failure), receipt := (ctor Option.none [OverflowReceipt])}) (record Terminal {run := (project s run), ready := ready, output := output, cancelled := (project s stopped), failure := (project s failure), receipt := (project s receipt)}))
    | Owner.handedOff bindingUsage => (ctor Movement.notReady s)) else (ctor Movement.stale s)),

definition{
  name: outputAvailable,
  parameters: [s : State],
  result: Bool,
  body: ((not (project s stopped)) && ((match (project s failure)
    | Option.some failure => false
    | Option.none => true) && ((match (project s owner)
    | Owner.running core => false
    | Owner.draining core output => false
    | Owner.terminal ready output => (match output
      | Option.none => false
      | Option.some handle => true)
    | Owner.handedOff achieved => false) && true))),
  axioms: []
}

definition{
  name: status,
  parameters: [s : State],
  result: Status,
  body: (match (project s owner)
    | Owner.handedOff achieved => (ctor Status.handedOff)
    | Owner.terminal ready output => (if (project s stopped) then (ctor Status.cancelled) else (match (project s failure)
      | Option.some failure => (ctor Status.failed failure)
      | Option.none => (if (outputAvailable s) then (ctor Status.decoded) else (ctor Status.cancelled))))
    | Owner.running core => (match (project core disposition)
      | Uorc.Core.RuntimeMachine::Disposition.running => (ctor Status.running)
      | Uorc.Core.RuntimeMachine::Disposition.suspended reason => (ctor Status.suspended reason)
      | Uorc.Core.RuntimeMachine::Disposition.completed handle => (ctor Status.running)
      | Uorc.Core.RuntimeMachine::Disposition.failed reason => (ctor Status.failed reason)
      | Uorc.Core.RuntimeMachine::Disposition.cancelling => (ctor Status.running)
      | Uorc.Core.RuntimeMachine::Disposition.cleaned => (ctor Status.running))
    | Owner.draining core output => (match (project core disposition)
      | Uorc.Core.RuntimeMachine::Disposition.running => (ctor Status.draining)
      | Uorc.Core.RuntimeMachine::Disposition.suspended reason => (ctor Status.suspended reason)
      | Uorc.Core.RuntimeMachine::Disposition.completed handle => (ctor Status.draining)
      | Uorc.Core.RuntimeMachine::Disposition.failed reason => (ctor Status.failed reason)
      | Uorc.Core.RuntimeMachine::Disposition.cancelling => (ctor Status.draining)
      | Uorc.Core.RuntimeMachine::Disposition.cleaned => (ctor Status.draining))),
  axioms: []
}

theorem{
  name: tick_guard_binding,
  parameters: [s : State, run : Uorc.Product.Api::Run],
  statement: ((tick s run) = (if (sameRun (project s run) run) then (match (project s owner)
    | Owner.running core => (afterTick s (Uorc.Core.RuntimeMachineEngine::tick core))
    | Owner.draining core output => (finishDrain s (Uorc.Core.RuntimeMachineEngine::tick core) output)
    | Owner.terminal ready output => s
    | Owner.handedOff achieved => s) else s)),
  proof: reflexivity{},
  axioms: [Quot.sound, propext]
}

definition{
  name: releaseDisposition,
  parameters: [s : State],
  result: Uorc.Product.Api::ReleaseDisposition,
  body: (match (project s owner)
    | Owner.running core => (ctor Uorc.Product.Api::ReleaseDisposition.retain (ctor Uorc.Product.Api::CustodyReason.arena))
    | Owner.draining core output => (ctor Uorc.Product.Api::ReleaseDisposition.retain (ctor Uorc.Product.Api::CustodyReason.arena))
    | Owner.terminal ready output => (ctor Uorc.Product.Api::ReleaseDisposition.retain (ctor Uorc.Product.Api::CustodyReason.arena))
    | Owner.handedOff achieved => (if (match (project s receipt)
      | Option.none => false
      | Option.some present => true) then (ctor Uorc.Product.Api::ReleaseDisposition.retain (ctor Uorc.Product.Api::CustodyReason.receipt)) else (ctor Uorc.Product.Api::ReleaseDisposition.releasable)))
