definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "One source-owned closed application root shares the generated Workspace, View, action, callback and effect constructor closure in a single package. This avoids passing package-local ADTs between independent frontend exports. Generic host code retains typed state under opaque instance identity, authorizes Command through its source Authority, and renders only generated View and BoundUserIntent values. Resource/runtime/publisher variants are trusted internal calls, never arbitrary browser JSON or caller state snapshots. Frame.retirement covers this Workspace and unconsumed effects only; actual child arenas, I/O, leases, achieved storage and publication owners remain independently retained. This facade does not prove generated outer ComputeError retains state, does not clone or reconstruct an Arena, and does not qualify operational root ABI, browser execution or compression completion."
}

definition{
  name: dispatchCommand,
  parameters: [command : Command],
  result: UpdateFrame,
  body: (match command
    | Command.create instance => (updateFrame (Uorc.Product.Workspace::initial instance) (nil Uorc.Product.Workspace::Effect) (ctor Disposition.observed) "" (ctor Option.none [Bool]))
    | Command.user s intent => (fromInvocation (Uorc.Product.Workspace::invokeUser s intent))
    | Command.resource s notice => (fromUpdate (Uorc.Product.Workspace::dispatch s (resourceIntent notice)))
    | Command.runtime s notice => (fromUpdate (Uorc.Product.Workspace::receiveRuntime s notice))
    | Command.number s bound text => (fromNumber (Uorc.Product.Workspace::editLimitText s bound text))
    | Command.present s => (updateFrame s (nil Uorc.Product.Workspace::Effect) (ctor Disposition.observed) "" (ctor Option.none [Bool]))
    | Command.permit s argRun destination => (updateFrame s (nil Uorc.Product.Workspace::Effect) (ctor Disposition.permitted) "" (ctor Option.some [Bool] (Uorc.Product.Control::authorizePublication (project s control) argRun destination)))),
  axioms: []
}
