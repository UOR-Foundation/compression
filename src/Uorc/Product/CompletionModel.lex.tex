\begin{lexlean}{Uorc.Product.CompletionModel}
\importmodule{Uorc.Product.Session}
\title{Parameters}
\begin{semanticmodule}
structure{
  name: CompletionRequest,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: session, type: Uorc.Product.Session::Session},
    {name: runSerial, type: Nat},
    {name: runRevision, type: Nat},
    {name: result, type: Uorc.Product.Session::ResultKind}
  ]
}

definition{
  name: isStaleCompletion,
  parameters: [request : CompletionRequest],
  result: Prop,
  body: ((Uorc.Product.Session::ownsRun (project request session) (project request runSerial) (project request runRevision)) = false)
}

definition{
  name: preservesSession,
  parameters: [request : CompletionRequest, output : Uorc.Product.Session::Session],
  result: Prop,
  body: (output = (project request session))
}

contract{
  name: StaleCompletionContract,
  axioms: [],
  input: {name: request, type: CompletionRequest},
  output: {name: output, type: Uorc.Product.Session::Session},
  postcondition: preservesSession,
  precondition: isStaleCompletion,
  validators: []
}

realization{
  name: CompletionTransition,
  axioms: [],
  descriptor: deterministic{
    body: (Uorc.Product.Session::finish (project request session) (project request runSerial) (project request runRevision) (project request result))
  },
  executable: true,
  input: {name: request, type: CompletionRequest},
  output: Uorc.Product.Session::Session
}

definition{
  name: select_session,
  parameters: [
    owned : Bool,
    updated : Uorc.Product.Session::Session,
    current : Uorc.Product.Session::Session
  ],
  result: Uorc.Product.Session::Session,
  body: (match owned | Bool.false => current | Bool.true => updated)
}

theorem{
  name: select_false_preserves,
  parameters: [request : CompletionRequest, updated : Uorc.Product.Session::Session],
  statement: (preservesSession request (select_session false updated (project request session))),
  proof: reflexivity{},
  axioms: []
}

theorem{
  name: stale_completion_branches,
  parameters: [request : CompletionRequest, owned : Bool, updated : Uorc.Product.Session::Session],
  statement: ((owned = false) -> (preservesSession request (select_session owned updated (project request session)))),
  proof: linear_arithmetic{definitions: [preservesSession, select_session]},
  axioms: [propext]
}

theorem{
  name: stale_completion_preserves_session,
  parameters: [request : CompletionRequest],
  statement: ((isStaleCompletion request) -> (preservesSession request (CompletionTransition request))),
  proof: apply{
    arguments: [
      request,
      (Uorc.Product.Session::ownsRun (project request session) (project request runSerial) (project request runRevision)),
      (record Uorc.Product.Session::Session {revision := (project (project request session) revision), serial := (project (project request session) serial), active := false, hasResult := true, phase := (Uorc.Product.Session::resultPhase (project request result)), result := (project request result)})
    ],
    theorem: stale_completion_branches
  },
  axioms: [propext]
}

evidence{
  name: CompletionEvidence,
  axioms: [propext],
  claims: [satisfies_contract{theorem: stale_completion_preserves_session}],
  contract: {member: StaleCompletionContract},
  realization: {member: CompletionTransition}
}

model{
  name: CompletionModel,
  axioms: [],
  contract: {member: StaleCompletionContract},
  evidence: [{member: CompletionEvidence}],
  realization: {member: CompletionTransition}
}

definition{
  name: boundary,
  parameters: [],
  result: String,
  body: "This Prism model binds the actual session-completion transition to the stale-run preservation contract. Its theorem obligation is universal over the request, not a finite UI example. Kernel acceptance of the theorem/evidence remains required; source declaration does not claim it has passed. This component contract does not substitute for wire, decoding, search, ownership, publication, or full-product contracts."
}
\end{semanticmodule}
\end{lexlean}
