definition{
  name: keep,
  parameters: [left : Trial, right : Trial],
  result: Trial,
  body: (match left
    | Trial.failed => right
    | Trial.coded lb ls ld => (match right
      | Trial.failed => left
      | Trial.coded rb rs rd => (ctor Trial.coded (prim subtract lb (prim subtract lb rb : Nat) : Nat) (if (rb <? lb) then rs else ls) (if (rb <? lb) then rd else ld))))
