definition{
  name: buildStep,
  parameters: [build : Build],
  result: Build,
  body: (match build
    | Build.reverse rest forward => (match rest
      | List.nil => (ctor Build.emit forward 0x)
      | List.cons digit tail => (if (digit <? 256) then (ctor Build.reverse tail (cons digit forward)) else (ctor Build.invalidDigit)))
    | Build.emit rest output => (match rest
      | List.nil => (ctor Build.done output)
      | List.cons digit tail => (if (digit <? 256) then (match (natByteSlice digit)
        | Option.none => (ctor Build.invalidDigit)
        | Option.some singleton => (ctor Build.emit tail (prim append output singleton : Bytes))) else (ctor Build.invalidDigit)))
    | Build.done output => (ctor Build.done output)
    | Build.invalidDigit => (ctor Build.invalidDigit))
}
