definition{
  name: step,
  parameters: [maximumLength : Nat, cursor : Cursor],
  result: Step,
  body: (match cursor
    | Cursor.ready word => (if ((project word length) <=? maximumLength) then (ctor Step.word word (ctor Cursor.carry (project word reversedDigits) (nil Nat) (project word length))) else (ctor Step.finished))
    | Cursor.carry rest zeroPrefix wordLength => (match rest
      | List.nil => (if (wordLength <? maximumLength) then (ctor Step.pending (ctor Cursor.ready (record Word {reversedDigits := (cons 0 zeroPrefix), length := (wordLength + 1)}))) else (ctor Step.finished))
      | List.cons digit tail => (if (digit <? 255) then (ctor Step.pending (ctor Cursor.restore zeroPrefix (cons (digit + 1) tail) wordLength)) else (if (digit =? 255) then (ctor Step.pending (ctor Cursor.carry tail (cons 0 zeroPrefix) wordLength)) else (ctor Step.invalidDigit))))
    | Cursor.restore zeroPrefix suffix wordLength => (match zeroPrefix
      | List.nil => (ctor Step.pending (ctor Cursor.ready (record Word {reversedDigits := suffix, length := wordLength})))
      | List.cons zero tail => (ctor Step.pending (ctor Cursor.restore tail (cons zero suffix) wordLength)))
    | Cursor.finished => (ctor Step.finished))
}

definition{
  name: beginBuild,
  parameters: [word : Word],
  result: Build,
  body: (ctor Build.reverse (project word reversedDigits) (nil Nat))

definition{
  name: octetCell,
  parameters: [octet : (Option UInt8)],
  result: (Option Bytes),
  body: (match octet
    | Option.none => (ctor Option.none [Bytes])
    | Option.some byte => (Uorc.Core.Types::byteCell byte)),
  axioms: []
}

definition{
  name: digitCell,
  parameters: [digit : (Option UInt64)],
  result: (Option Bytes),
  body: (match digit
    | Option.none => (ctor Option.none [Bytes])
    | Option.some word => (octetCell (prim checked_convert word : (Option UInt8)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: natByteSlice,
  parameters: [n : Nat],
  result: (Option Bytes),
  body: (digitCell (Uorc.Core.Types::wordOfNat n)),
  axioms: [Quot.sound, propext]
}

definition{
  name: buildStep,
  parameters: [build : Build],
  result: Build,
  body: (match build
    | Build.reverse rest forward => (match rest
      | List.nil => (ctor Build.emit forward 0x)
      | List.cons digit tail => (if (digit <? 256) then (ctor Build.reverse tail (cons digit forward)) else (ctor Build.invalidDigit)))
    | Build.emit rest output => (match rest
      | List.nil => (ctor Build.done output)
      | List.cons digit tail => (if (digit <? 256) then (match (natByteSlice digit)
        | Option.none => (ctor Build.invalidDigit)
        | Option.some singleton => (ctor Build.emit tail (prim append output singleton : Bytes))) else (ctor Build.invalidDigit)))
    | Build.done output => (ctor Build.done output)
    | Build.invalidDigit => (ctor Build.invalidDigit)),
  axioms: [Quot.sound, propext]
}

definition{
  name: runBuild,
  parameters: [fuel : Nat, build : Build],
  result: Build,
  body: (match fuel
    | Nat.zero => build
    | Nat.succ restFuel => (runBuild restFuel (buildStep build))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}
