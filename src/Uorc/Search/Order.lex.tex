structure{
  name: Cursor,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: incumbent, type: Uorc.Search.Verify::CheckedArchive},
    {name: challenger, type: Uorc.Search.Verify::CheckedArchive},
    {name: offset, type: Nat}
  ]
}

inductive{
  name: Step,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: pending, fields: [Cursor]},
    {name: selected, fields: [Uorc.Search.Verify::CheckedArchive]},
    {name: invalidCursor, fields: []}
  ]
}

definition{
  name: byteLess,
  parameters: [left : UInt8, right : UInt8],
  result: Bool,
  body: (match (prim checked_subtract left right : (Option UInt8))
    | Option.none => true
    | Option.some difference => false),
  axioms: [Quot.sound, propext]
}

definition{
  name: start,
  parameters: [
    incumbent : Uorc.Search.Verify::CheckedArchive,
    challenger : Uorc.Search.Verify::CheckedArchive
  ],
  result: Step,
  body: (if ((prim length (project challenger wire) : Nat) <? (prim length (project incumbent wire) : Nat)) then (ctor Step.selected challenger) else (if ((prim length (project incumbent wire) : Nat) <? (prim length (project challenger wire) : Nat)) then (ctor Step.selected incumbent) else (ctor Step.pending (record Cursor {incumbent := incumbent, challenger := challenger, offset := 0}))))
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (if (prim equal (prim length (project (project cursor incumbent) wire) : Nat) (prim length (project (project cursor challenger) wire) : Nat) : Bool) then (if ((project cursor offset) <? (prim length (project (project cursor incumbent) wire) : Nat)) then (match (prim index (project (project cursor incumbent) wire) (project cursor offset) : (Option UInt8))
    | Option.none => (ctor Step.invalidCursor)
    | Option.some oldByte => (match (prim index (project (project cursor challenger) wire) (project cursor offset) : (Option UInt8))
      | Option.none => (ctor Step.invalidCursor)
      | Option.some newByte => (if (prim equal oldByte newByte : Bool) then (ctor Step.pending (record Cursor {incumbent := (project cursor incumbent), challenger := (project cursor challenger), offset := ((project cursor offset) + 1)})) else (if (byteLess newByte oldByte) then (ctor Step.selected (project cursor challenger)) else (ctor Step.selected (project cursor incumbent)))))) else (if ((project cursor offset) =? (prim length (project (project cursor incumbent) wire) : Nat)) then (ctor Step.selected (project cursor incumbent)) else (ctor Step.invalidCursor))) else (ctor Step.invalidCursor)),
  axioms: [Quot.sound, propext]
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ restFuel => match{
      branches: [
        {body: (run restFuel (step cursor)), binders: [cursor], constructor: Step.pending},
        {
          body: (ctor Step.selected var{name: checked}),
          binders: [checked],
          constructor: Step.selected
        },
        {body: (ctor Step.invalidCursor), binders: [], constructor: Step.invalidCursor}
      ],
      scrutinee: state
    }),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}
