definition{
  name: meaningFits,
  parameters: [graph : Uorc.Core.Graph::Graph, expected : Bytes],
  result: Bool,
  body: (match (Uorc.Core.ReferenceSemantics::evaluateGraph graph (nil Uorc.Core.ReferenceSemantics::Sequence))
    | Result.error failure => false
    | Result.ok value => (match value
      | Uorc.Core.ReferenceSemantics::Value.natural naturalValue => false
      | Uorc.Core.ReferenceSemantics::Value.byte byteValue => false
      | Uorc.Core.ReferenceSemantics::Value.predicate predicateValue => false
      | Uorc.Core.ReferenceSemantics::Value.sequence sequenceValue => (Uorc.Core.ReferenceSemantics::equalSequences sequenceValue (Uorc.Core.ReferenceSemantics::sequenceFromBytes expected))
      | Uorc.Core.ReferenceSemantics::Value.store storeValue => false))
}

definition{
  name: learnedFits,
  parameters: [state : Uorc.Search.PackedFit::Step, expected : Bytes],
  result: Bool,
  body: (match (Uorc.Search.PackedFit::proposal state)
    | Option.none => false
    | Option.some graph => (meaningFits graph expected))
}
