definition{
  name: eventWork,
  parameters: [normalization : Uorc.Search.Normalize::Cursor],
  result: Nat,
  body: (match normalization
    | Uorc.Search.Normalize::Cursor.reversing workState => 1
    | Uorc.Search.Normalize::Cursor.building workState => 1
    | Uorc.Search.Normalize::Cursor.initial workState => 1
    | Uorc.Search.Normalize::Cursor.blockReading workState => 1
    | Uorc.Search.Normalize::Cursor.blockAppending workState => (match workState
      | Uorc.Search.Normalize::BlockAppend.state b0 work b2 b3 b4 => (Uorc.Core.ByteBuilder::eventWork work))
    | Uorc.Search.Normalize::Cursor.blockFinishing workState => (match workState
      | Uorc.Search.Normalize::BlockAppend.state b0 work b2 b3 b4 => (Uorc.Core.ByteBuilder::eventWork work)))
}

definition{
  name: tick,
  parameters: [policy : Uorc.Search.Normalize::Policy, cursor : Cursor],
  result: Step,
  body: (if (((project cursor eventCredit) + 1) <? (eventWork (project cursor normalization))) then (ctor Step.pending (record Cursor {normalization := (project cursor normalization), eventCredit := ((project cursor eventCredit) + 1), consumedCredits := ((project cursor consumedCredits) + 1)})) else (let observation : Uorc.Search.Normalize::Observation := (Uorc.Search.Normalize::observe policy (project cursor normalization)); (match (project observation result)
    | Uorc.Search.Normalize::Step.pending next => (ctor Step.pending (record Cursor {normalization := next, eventCredit := 0, consumedCredits := ((project cursor consumedCredits) + 1)}))

    | Uorc.Search.Normalize::Step.rejected failure => (ctor Step.rejected failure (project observation prefix) ((project cursor consumedCredits) + 1))
    | Uorc.Search.Normalize::Step.ropeFailure failure => (ctor Step.ropeFailure failure (project observation prefix) ((project cursor consumedCredits) + 1))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: runCursor,
  parameters: [
    work : Nat,
    policy : Uorc.Search.Normalize::Policy,
    normalization : Uorc.Search.Normalize::Cursor,
    eventCredit : Nat,
    consumedCredits : Nat
  ],
  result: Step,
  body: (match work
    | Nat.zero => (ctor Step.pending (record Cursor {normalization := normalization, eventCredit := eventCredit, consumedCredits := consumedCredits}))
    | Nat.succ remainingWork => (if ((eventCredit + 1) <? (eventWork normalization)) then (runCursor remainingWork policy normalization (eventCredit + 1) (consumedCredits + 1)) else (let observation : Uorc.Search.Normalize::Observation := (Uorc.Search.Normalize::observe policy normalization); (match (project observation result)
      | Uorc.Search.Normalize::Step.pending next => (runCursor remainingWork policy next 0 (consumedCredits + 1))
      | Uorc.Search.Normalize::Step.done normalized physical => (ctor Step.done normalized physical (consumedCredits + 1))
      | Uorc.Search.Normalize::Step.rejected failure => (ctor Step.rejected failure (project observation prefix) (consumedCredits + 1))
      | Uorc.Search.Normalize::Step.ropeFailure failure => (ctor Step.ropeFailure failure (project observation prefix) (consumedCredits + 1)))))),
  axioms: [Quot.sound, propext],
  recursive_argument: work
}

definition{
  name: runWork,
  parameters: [work : Nat, policy : Uorc.Search.Normalize::Policy, state : Step],
  result: Step,
  body: (match state
    | Step.pending cursor => (runCursor work policy (project cursor normalization) (project cursor eventCredit) (project cursor consumedCredits))
    | Step.done normalized physical credits => (ctor Step.done normalized physical credits)
    | Step.rejected failure prefix credits => (ctor Step.rejected failure prefix credits)
    | Step.ropeFailure failure prefix credits => (ctor Step.ropeFailure failure prefix credits)),
  axioms: [Quot.sound, propext]
}
