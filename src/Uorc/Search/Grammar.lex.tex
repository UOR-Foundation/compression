definition{
  name: checkChoice,
  parameters: [
    prior : (List Uorc.Core.Graph::Block),
    slots : (List Uorc.Core.Types::ValueType),
    ordinal : Nat,
    arguments : Arguments
  ],
  result: (Result TypedProposal Uorc.Core.Types::StaticError),
  body: (match (instructionChoice ordinal arguments)
    | Option.none => (ctor Result.error [TypedProposal, Uorc.Core.Types::StaticError] (ctor Uorc.Core.Types::StaticError.undefinedOpcode))
    | Option.some instruction => (match (Uorc.Core.Graph::inferInstruction prior slots instruction)
      | Result.error typingError => (ctor Result.error [TypedProposal, Uorc.Core.Types::StaticError] typingError)
      | Result.ok resultType => (ctor Result.ok [TypedProposal, Uorc.Core.Types::StaticError] (record TypedProposal {instruction := instruction, resultType := resultType}))))
}
