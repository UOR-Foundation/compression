  result: Step,
  body: (match (project cursor phase)

(project (project (project cursor right) budget) maxSteps)) && true)))) then (match (project (project cursor left) scope)
      | Uorc.Wire.Frame::Scope.selfContained => (match (project (project cursor right) scope)
        | Uorc.Wire.Frame::Scope.selfContained => (ctor Step.pending (record Cursor {left := (project cursor left), right := (project cursor right), phase := (ctor Phase.boundaries (Uorc.Search.Boundaries::start (project (project cursor left) objectLengths) (project (project cursor right) objectLengths)))}))
        | Uorc.Wire.Frame::Scope.conditional rightBasisId => (ctor Step.different (ctor Field.scope)))
      | Uorc.Wire.Frame::Scope.conditional leftBasisId => (match (project (project cursor right) scope)
        | Uorc.Wire.Frame::Scope.selfContained => (ctor Step.different (ctor Field.scope))

    | Phase.boundaries comparison => (match (Uorc.Search.Boundaries::step comparison)
      | Uorc.Search.Boundaries::Step.pending next => (ctor Step.pending (record Cursor {left := (project cursor left), right := (project cursor right), phase := (ctor Phase.boundaries next)}))

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ remainingFuel => (match state
      | Step.pending cursor => (run remainingFuel (step cursor))
      | Step.matched => (ctor Step.matched)
      | Step.different field => (ctor Step.different field)
      | Step.invalidScope => (ctor Step.invalidScope)
      | Step.ropeFailure error => (ctor Step.ropeFailure error))),
  recursive_argument: fuel
}
