definition{
  name: start,
  parameters: [
    incumbent : Uorc.Search.VerifyRope::CheckedArchive,
    challenger : Uorc.Search.VerifyRope::CheckedArchive
  ],
  result: Step,
  body: (if ((Uorc.Core.Bytes::length (project challenger wire)) <? (Uorc.Core.Bytes::length (project incumbent wire))) then (ctor Step.selected challenger) else (if ((Uorc.Core.Bytes::length (project incumbent wire)) <? (Uorc.Core.Bytes::length (project challenger wire))) then (ctor Step.selected incumbent) else (ctor Step.pending (record Cursor {incumbent := incumbent, challenger := challenger, comparison := (Uorc.Search.VerifyRope::start (cons (project incumbent wire) (nil Uorc.Core.Bytes::Rope)) (cons (project challenger wire) (nil Uorc.Core.Bytes::Rope)))}))))
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (Uorc.Search.VerifyRope::step (project cursor comparison))
    | Uorc.Search.VerifyRope::Step.pending next => (ctor Step.pending (record Cursor {incumbent := (project cursor incumbent), challenger := (project cursor challenger), comparison := next}))
    | Uorc.Search.VerifyRope::Step.matched compared boundaries => (ctor Step.selected (project cursor incumbent))
    | Uorc.Search.VerifyRope::Step.mismatch mismatch => (match (project mismatch kind)
      | Uorc.Search.Verify::MismatchKind.objectCount => (ctor Step.invalidCursor)
      | Uorc.Search.Verify::MismatchKind.objectLength => (ctor Step.invalidCursor)
      | Uorc.Search.Verify::MismatchKind.byteValue => (match (project (project cursor comparison) expectedByte)
        | Option.none => (ctor Step.invalidCursor)
        | Option.some oldByte => (match (project (project cursor comparison) actualByte)
          | Option.none => (ctor Step.invalidCursor)
          | Option.some newByte => (if (Uorc.Search.Order::byteLess newByte oldByte) then (ctor Step.selected (project cursor challenger)) else (ctor Step.selected (project cursor incumbent)))))
      | Uorc.Search.Verify::MismatchKind.invalidCursor => (ctor Step.invalidCursor))
    | Uorc.Search.VerifyRope::Step.ropeFailure ropeError => (ctor Step.ropeFailure ropeError))
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ restFuel => match{
      branches: [
        {body: (run restFuel (step cursor)), binders: [cursor], constructor: Step.pending},
        {
          body: (ctor Step.selected var{name: checked}),
          binders: [checked],
          constructor: Step.selected
        },
        {body: (ctor Step.invalidCursor), binders: [], constructor: Step.invalidCursor},
        {
          body: (ctor Step.ropeFailure ropeError),
          binders: [ropeError],
          constructor: Step.ropeFailure
        }
      ],
      scrutinee: state
    }),
  recursive_argument: fuel
}
