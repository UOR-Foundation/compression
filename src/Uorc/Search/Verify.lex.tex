definition{
  name: start,
  parameters: [expected : (List Bytes), actual : (List Bytes)],
  result: Cursor,
  body: (record Cursor {expected := expected, actual := actual, objectIndex := 0, byteOffset := 0, bytesCompared := 0, boundariesChecked := 0})
}

definition{
  name: failure,
  parameters: [cursor : Cursor, kind : MismatchKind],
  result: Step,
  body: (ctor Step.mismatch (record Mismatch {kind := kind, objectIndex := (project cursor objectIndex), byteOffset := (project cursor byteOffset)}))
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor expected)
    | List.nil => (match (project cursor actual)
      | List.nil => (if ((project cursor byteOffset) =? 0) then (ctor Step.matched (project cursor bytesCompared) (project cursor boundariesChecked)) else (failure cursor (ctor MismatchKind.invalidCursor)))
      | List.cons unexpectedHead unexpectedTail => (failure cursor (ctor MismatchKind.objectCount)))
    | List.cons expectedHead expectedRest => (match (project cursor actual)
      | List.nil => (failure cursor (ctor MismatchKind.objectCount))
      | List.cons actualHead actualRest => (if ((prim length expectedHead : Nat) =? (prim length actualHead : Nat)) then (if ((project cursor byteOffset) <? (prim length expectedHead : Nat)) then (match (prim index expectedHead (project cursor byteOffset) : (Option UInt8))
        | Option.none => (failure cursor (ctor MismatchKind.invalidCursor))
        | Option.some expectedByte => (match (prim index actualHead (project cursor byteOffset) : (Option UInt8))
          | Option.none => (failure cursor (ctor MismatchKind.invalidCursor))
          | Option.some actualByte => (if (prim equal expectedByte actualByte : Bool) then (ctor Step.pending (record Cursor {expected := (project cursor expected), actual := (project cursor actual), objectIndex := (project cursor objectIndex), byteOffset := ((project cursor byteOffset) + 1), bytesCompared := ((project cursor bytesCompared) + 1), boundariesChecked := (project cursor boundariesChecked)})) else (failure cursor (ctor MismatchKind.byteValue))))) else (if ((project cursor byteOffset) =? (prim length expectedHead : Nat)) then (ctor Step.pending (record Cursor {expected := expectedRest, actual := actualRest, objectIndex := ((project cursor objectIndex) + 1), byteOffset := 0, bytesCompared := (project cursor bytesCompared), boundariesChecked := ((project cursor boundariesChecked) + 1)})) else (failure cursor (ctor MismatchKind.invalidCursor)))) else (failure cursor (ctor MismatchKind.objectLength)))))
}

definition{
  name: run,
  parameters: [fuel : Nat, cursor : Cursor],
  result: Step,
  body: (match fuel
    | Nat.zero => (ctor Step.pending cursor)
    | Nat.succ restFuel => (match (step cursor)
      | Step.pending next => (run restFuel next)
      | Step.matched bytes boundaries => (ctor Step.matched bytes boundaries)
      | Step.mismatch mismatch => (ctor Step.mismatch mismatch))),
  recursive_argument: fuel
}

definition{
  name: startTrial,
  parameters: [wire : Bytes, expected : (List Bytes), actual : (List Bytes)],
  result: Trial,
  body: (record Trial {wire := wire, cursor := (start expected actual)})
}

definition{
  name: stepTrial,
  parameters: [trial : Trial],
  result: TrialStep,
  body: (match (step (project trial cursor))
    | Step.pending next => (ctor TrialStep.pending (record Trial {wire := (project trial wire), cursor := next}))
    | Step.matched bytes boundaries => (ctor TrialStep.eligible (record CheckedArchive {wire := (project trial wire), bytesCompared := bytes, boundariesChecked := boundaries}))
    | Step.mismatch mismatch => (ctor TrialStep.rejected mismatch))
}

definition{
  name: runTrial,
  parameters: [fuel : Nat, trial : Trial],
  result: TrialStep,
  body: (match fuel
    | Nat.zero => (ctor TrialStep.pending trial)
    | Nat.succ restFuel => match{
      branches: [
        {body: (runTrial restFuel next), binders: [next], constructor: TrialStep.pending},
        {
          body: (ctor TrialStep.eligible var{name: checked}),
          binders: [checked],
          constructor: TrialStep.eligible
        },
        {
          body: (ctor TrialStep.rejected mismatch),
          binders: [mismatch],
          constructor: TrialStep.rejected
        }
      ],
      scrutinee: (stepTrial trial)
    }),
  recursive_argument: fuel
}
