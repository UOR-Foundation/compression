definition{
  name: readStep,
  parameters: [
    arena : Uorc.Core.Heap::Arena,
    walking : Uorc.Core.RuntimeValue::Handle,
    index : UInt64
  ],
  result: Read,
  body: (match (Uorc.Core.Heap::lookup arena walking)
    | Result.error error => (ctor Read.failed (ctor Error.heap error))
    | Result.ok entry => (match (project entry node)
      | Uorc.Core.Heap::Node.image field0 field1 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.sequenceLeaf field0 field1 field2 => (if (Uorc.Core.Heap::less index field2) then (match (prim checked_add field1 index : (Option UInt64))
        | Option.none => (ctor Read.failed (ctor Error.addressSpaceExhausted))
        | Option.some absolute => (match (Uorc.Core.RuntimeImage::readByte arena field0 absolute)
          | Result.error error => (ctor Read.failed (ctor Error.image error))
          | Result.ok value => (ctor Read.byte value))) else (ctor Read.failed (ctor Error.invalidShape)))
      | Uorc.Core.Heap::Node.sequenceBranch field0 field1 field2 field3 => (match (Uorc.Core.Heap::sequenceInfo arena field2)
        | Result.error error => (ctor Read.failed (ctor Error.heap error))
        | Result.ok info => (if (Uorc.Core.Heap::less (project (if (Uorc.Core.Heap::less index (project info length)) then field2 else field3) id) (project walking id)) then (ctor Read.descend (if (Uorc.Core.Heap::less index (project info length)) then field2 else field3) (if (Uorc.Core.Heap::less index (project info length)) then index else (Uorc.Core.Heap::wordSub index (project info length)))) else (ctor Read.failed (ctor Error.invalidShape))))
      | Uorc.Core.Heap::Node.storeLeaf field0 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.storeBranch field0 field1 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.slot field0 field1 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.rootBranch field0 field1 field2 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.metadataWord field0 field1 field2 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.metadataLink field0 field1 field2 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.metadataBranch field0 field1 field2 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.ownerRoot field0 field1 => (ctor Read.failed (ctor Error.invalidShape))
      | Uorc.Core.Heap::Node.imageReservation field0 field1 => (ctor Read.failed (ctor Error.invalidShape)))),
  axioms: [Quot.sound, propext],
  executable: true
}
