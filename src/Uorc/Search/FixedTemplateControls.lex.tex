definition{
  name: advance,
  parameters: [work : Nat, cursor : Uorc.Search.FixedTemplates::Cursor],
  result: Uorc.Search.FixedTemplates::Cursor,
  body: (match work
    | Nat.zero => cursor
    | Nat.succ remaining => (match (Uorc.Search.FixedTemplates::step cursor)
      | Uorc.Search.FixedTemplates::EventStep.complete => cursor
      | Uorc.Search.FixedTemplates::EventStep.invalidPosition => cursor
      | Uorc.Search.FixedTemplates::EventStep.produced offered continuation => (advance remaining continuation)
      | Uorc.Search.FixedTemplates::EventStep.literal sequenceRoot continuation => (advance remaining continuation))),
  axioms: [Quot.sound, propext],
  recursive_argument: work
}
