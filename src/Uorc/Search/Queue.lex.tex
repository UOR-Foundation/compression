\importmodule{Uorc.Search.Schedule}

\title{Parameters}
\begin{semanticmodule}
structure{
  name: Task,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: phase, type: Uorc.Search.Schedule::Phase},
    {name: round, type: Nat},
    {name: wire, type: Bytes},
    {name: canonicalTask, type: Bytes}
  ]
}

inductive{
  name: Cursor,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: scan, fields: [Task, (List Task), (List Task), Nat, Bool]},
    {name: rebuild, fields: [(List Task), (List Task)]}
  ]
}

inductive{
  name: Step,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: pending, fields: [Cursor]},
    {name: done, fields: [(List Task)]},
    {name: invalidCursor, fields: []}
  ]
}

definition{
  name: beginInsert,
  parameters: [incoming : Task, queue : (List Task)],
  result: Cursor,
  body: (ctor Cursor.scan incoming (nil Task) queue 0 false)
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match cursor
    | Cursor.scan incoming leftReversed right offset wireStage => (match right
      | List.nil => (ctor Step.pending (ctor Cursor.rebuild leftReversed (cons incoming (nil Task))))
      | List.cons head tail => (if ((project incoming round) <? (project head round)) then (ctor Step.pending (ctor Cursor.rebuild leftReversed (cons incoming right))) else (if ((project head round) <? (project incoming round)) then (ctor Step.pending (ctor Cursor.scan incoming (cons head leftReversed) tail 0 false)) else (if ((prim length (project incoming wire) : Nat) <? (prim length (project head wire) : Nat)) then (ctor Step.pending (ctor Cursor.rebuild leftReversed (cons incoming right))) else (if ((prim length (project head wire) : Nat) <? (prim length (project incoming wire) : Nat)) then (ctor Step.pending (ctor Cursor.scan incoming (cons head leftReversed) tail 0 false)) else (if ((offset <? (prim length (if wireStage then (project incoming wire) else (project incoming canonicalTask)) : Nat)) && (offset <? (prim length (if wireStage then (project head wire) else (project head canonicalTask)) : Nat))) then (match (prim index (if wireStage then (project incoming wire) else (project incoming canonicalTask)) offset : (Option UInt8))
        | Option.none => (ctor Step.invalidCursor)
        | Option.some newByte => (match (prim index (if wireStage then (project head wire) else (project head canonicalTask)) offset : (Option UInt8))
          | Option.none => (ctor Step.invalidCursor)
          | Option.some oldByte => (if (prim equal newByte oldByte : Bool) then (ctor Step.pending (ctor Cursor.scan incoming leftReversed right (offset + 1) wireStage)) else (if (Uorc.Search.Order::byteLess newByte oldByte) then (ctor Step.pending (ctor Cursor.rebuild leftReversed (cons incoming right))) else (ctor Step.pending (ctor Cursor.scan incoming (cons head leftReversed) tail 0 false)))))) else (if ((offset =? (prim length (if wireStage then (project incoming wire) else (project incoming canonicalTask)) : Nat)) && (offset =? (prim length (if wireStage then (project head wire) else (project head canonicalTask)) : Nat))) then (if wireStage then (ctor Step.pending (ctor Cursor.rebuild leftReversed right)) else (ctor Step.pending (ctor Cursor.scan incoming leftReversed right 0 true))) else (if (offset =? (prim length (if wireStage then (project incoming wire) else (project incoming canonicalTask)) : Nat)) then (ctor Step.pending (ctor Cursor.rebuild leftReversed (cons incoming right))) else (if (offset =? (prim length (if wireStage then (project head wire) else (project head canonicalTask)) : Nat)) then (ctor Step.pending (ctor Cursor.scan incoming (cons head leftReversed) tail 0 false)) else (ctor Step.invalidCursor))))))))))
    | Cursor.rebuild leftReversed right => (match leftReversed
      | List.nil => (ctor Step.done right)
      | List.cons head tail => (ctor Step.pending (ctor Cursor.rebuild tail (cons head right))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ restFuel => (match state
      | Step.pending cursor => (run restFuel (step cursor))
      | Step.done queue => (ctor Step.done queue)
      | Step.invalidCursor => (ctor Step.invalidCursor))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

theorem{
  name: exact_archive_size_precedes_task_key,
  parameters: [],
  statement: ((run 30 (ctor Step.pending (beginInsert (record Task {phase := (ctor Uorc.Search.Schedule::Phase.typedGeneration), round := 0, wire := 0xff, canonicalTask := 0xff}) (cons (record Task {phase := (ctor Uorc.Search.Schedule::Phase.typedGeneration), round := 0, wire := 0x0000, canonicalTask := 0x02}) (nil Task))))) = (ctor Step.done (cons (record Task {phase := (ctor Uorc.Search.Schedule::Phase.typedGeneration), round := 0, wire := 0xff, canonicalTask := 0xff}) (cons (record Task {phase := (ctor Uorc.Search.Schedule::Phase.typedGeneration), round := 0, wire := 0x0000, canonicalTask := 0x02}) (nil Task))))),
  proof: reflexivity{},
  axioms: [Quot.sound, propext]
}
