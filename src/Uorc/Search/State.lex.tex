definition{
  name: infer,
  parameters: [input : Bytes, multiplier : UInt64],
  result: (Result Parameters Error),
  body: match{
    branches: [
      {
        body: (ctor Result.error [Parameters, Error] (ctor Error.insufficientObservations)),
        binders: [],
        constructor: Option.none
      },
      {
        body: match{
          branches: [
            {
              body: (ctor Result.error [Parameters, Error] (ctor Error.insufficientObservations)),
              binders: [],
              constructor: Option.none
            },
            {
              body: (match (prim checked_multiply multiplier (Uorc.Search.Charts::byteNatural var{
                  name: first
                }) : (Option UInt64))
                | Option.none => (ctor Result.error [Parameters, Error] (ctor Error.arithmeticOverflow))
                | Option.some product => (match (prim checked_quotient product (256 : UInt64) : (Option UInt64))
                  | Option.none => (ctor Result.error [Parameters, Error] (ctor Error.arithmeticOverflow))
                  | Option.some quotient => (match (prim checked_multiply quotient (256 : UInt64) : (Option UInt64))
                    | Option.none => (ctor Result.error [Parameters, Error] (ctor Error.arithmeticOverflow))
                    | Option.some multiple => (match (prim checked_subtract product multiple : (Option UInt64))
                      | Option.none => (ctor Result.error [Parameters, Error] (ctor Error.arithmeticOverflow))
                      | Option.some reduced => (match (Uorc.Search.Charts::ringDifference (Uorc.Search.Charts::byteNatural var{
                          name: second
                        }) reduced)
                        | Result.error chartError => (ctor Result.error [Parameters, Error] (ctor Error.arithmeticOverflow))
                        | Result.ok increment => (ctor Result.ok [Parameters, Error] (record Parameters {initial := (Uorc.Search.Charts::byteNatural var{
                          name: first
                        }), multiplier := multiplier, increment := increment}))))))),
              binders: [second],
              constructor: Option.some
            }
          ],
          scrutinee: (prim index input 1 : (Option UInt8))
        },
        binders: [first],
        constructor: Option.some
      }
    ],
    scrutinee: (prim index input 0 : (Option UInt8))
  },
  axioms: [Quot.sound, propext]
}

definition{
  name: step,
  parameters: [input : Bytes, cursor : Cursor],
  result: Step,
  body: (if (256 <=? (project cursor nextMultiplier)) then (ctor Step.finished) else (match (Uorc.Search.Effort::checkedU64 (project cursor nextMultiplier))
    | Result.error conversionError => (ctor Step.rejected (ctor Error.arithmeticOverflow) (record Cursor {nextMultiplier := ((project cursor nextMultiplier) + 1)}))
    | Result.ok multiplier => (match (infer input multiplier)
      | Result.error inferenceError => (ctor Step.rejected inferenceError (record Cursor {nextMultiplier := ((project cursor nextMultiplier) + 1)}))
      | Result.ok parameters => (match (Uorc.Search.Effort::checkedU64 (prim length input : Nat))
        | Result.error sizeError => (ctor Step.rejected (ctor Error.arithmeticOverflow) (record Cursor {nextMultiplier := ((project cursor nextMultiplier) + 1)}))
        | Result.ok count => (ctor Step.proposal (stateGraph parameters count) (record Cursor {nextMultiplier := ((project cursor nextMultiplier) + 1)})))))),
  axioms: [Quot.sound, propext]
}
