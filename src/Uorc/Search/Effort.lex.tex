definition{
  name: appendBit,
  parameters: [rest : UInt64, bit : Nat],
  result: (Result UInt64 Error),
  body: (match (prim checked_add rest rest : (Option UInt64))
    | Option.none => (ctor Result.error [UInt64, Error] (ctor Error.defaultOverflow))
    | Option.some doubled => (if (bit =? 0) then (ctor Result.ok [UInt64, Error] doubled) else (match (prim checked_add doubled (1 : UInt64) : (Option UInt64))
      | Option.none => (ctor Result.error [UInt64, Error] (ctor Error.defaultOverflow))
      | Option.some incremented => (ctor Result.ok [UInt64, Error] incremented)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: natToU64Bits,
  parameters: [fuel : Nat, value : Nat],
  result: (Result UInt64 Error),
  body: (match fuel
    | Nat.zero => (if (value =? 0) then (ctor Result.ok [UInt64, Error] (0 : UInt64)) else (ctor Result.error [UInt64, Error] (ctor Error.defaultOverflow)))
    | Nat.succ restFuel => (match (natToU64Bits restFuel (prim quotient value 2 0 : Nat))
      | Result.error error => (ctor Result.error [UInt64, Error] error)
      | Result.ok higher => (appendBit higher (prim remainder value 2 0 : Nat)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: checkedDefault,
  parameters: [
    bytes : Nat,
    objects : Nat,
    base : UInt64,

  ],
  result: (Result UInt64 Error),
  body: (match (checkedU64 bytes)
    | Result.error error => (ctor Result.error [UInt64, Error] error)

      | Result.ok objectWord => (match (prim checked_multiply byteWord byteFactor : (Option UInt64))

            | Option.none => (ctor Result.error [UInt64, Error] (ctor Error.defaultOverflow))

              | Option.none => (ctor Result.error [UInt64, Error] (ctor Error.defaultOverflow))
              | Option.some totalWork => (ctor Result.ok [UInt64, Error] totalWork))))))),
