inductive{
  name: Phase,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: validating, fields: [Uorc.Core.Validation::Cursor]},
    {
      name: executing,
      fields: [Uorc.Core.Validation::CheckedGraph, Uorc.Core.ExecutionCredit::Cursor]
    },
    {name: fitting, fields: [Uorc.Search.ResidualRope::Cursor]},
    {name: building, fields: [Uorc.Search.Residual::Finish]}
  ]
}

structure{
  name: Cursor,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: target, type: Uorc.Core.Bytes::Rope},
    {name: generator, type: Uorc.Core.Graph::Block},
    {name: count, type: UInt64},
    {name: budget, type: Uorc.Core.Resources::Budget},
    {name: phase, type: Phase},
    {name: completedEvents, type: Nat}
  ]
}

inductive{
  name: Error,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: coordinateOverflow, fields: []},
    {name: static, fields: [Uorc.Core.Types::StaticError]},
    {name: table, fields: [Uorc.Core.Validation::TableError]},
    {name: execution, fields: [Uorc.Core.StepMachine::Failure]},
    {name: machine, fields: [Uorc.Core.StepMachine::Failure, Uorc.Core.Resources::Usage]},
    {name: credit, fields: [Uorc.Core.ExecutionCredit::CreditError]},
    {name: outputType, fields: []},

      name: machineAccounted,
      fields: [
        Uorc.Core.StepMachine::Failure,

      ]

      name: creditAccounted,
      fields: [Uorc.Core.ExecutionCredit::CreditError, Uorc.Core.ExecutionCredit::Ledger]
    },
    {name: outputTypeAccounted, fields: [Uorc.Core.ExecutionCredit::Ledger]}
  ]
}

inductive{
  name: Step,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: pending, fields: [Cursor]},
    {name: proposal, fields: [Uorc.Core.Graph::Graph, Nat]},
    {name: rejected, fields: [Error]}
  ]
}

definition{
  name: start,
  parameters: [
    target : Uorc.Core.Bytes::Rope,
    generator : Uorc.Core.Graph::Block,
    budget : Uorc.Core.Resources::Budget
  ],
  result: Step,
  body: (match (Uorc.Search.Effort::checkedU64 (Uorc.Core.Bytes::length target))
    | Result.error error => (ctor Step.rejected (ctor Error.coordinateOverflow))
    | Result.ok count => (ctor Step.pending (record Cursor {target := target, generator := generator, count := count, budget := budget, phase := (ctor Phase.validating (Uorc.Core.Validation::start (record Uorc.Core.Graph::Graph {blocks := (cons generator (cons (record Uorc.Core.Graph::Block {parameters := (nil Uorc.Core.Types::ValueType), result := (ctor Uorc.Core.Types::ValueType.sequence), instructions := (cons (ctor Uorc.Core.Graph::Instruction.naturalConstant count) (cons (ctor Uorc.Core.Graph::Instruction.tabulate 1 1 (nil Nat)) (nil Uorc.Core.Graph::Instruction))), returnDistance := 1}) (nil Uorc.Core.Graph::Block))), entry := 1}))), completedEvents := 0})))
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor phase)
    | Phase.validating validation => (match (Uorc.Core.Validation::tick validation)
      | Uorc.Core.Validation::Step.pending next => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.validating next), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Core.Validation::Step.validated checkedGraph => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.executing checkedGraph (Uorc.Core.ExecutionCredit::startInitialization checkedGraph (Uorc.Core.Basis::empty) (project cursor budget) (Uorc.Core.Resources::zeroUsage))), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Core.Validation::Step.rejected error events visits => (ctor Step.rejected (ctor Error.static error))
      | Uorc.Core.Validation::Step.tableFailure error events visits => (ctor Step.rejected (ctor Error.table error)))
    | Phase.executing checkedGraph executionCredit => (match (Uorc.Core.ExecutionCredit::tick (Uorc.Core.StepMachine::contextForChecked checkedGraph (nil Bytes) (project cursor budget)) executionCredit)
      | Uorc.Core.ExecutionCredit::Tick.pending nextCredit => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.executing checkedGraph nextCredit), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Core.ExecutionCredit::Tick.done execution executionLedger => (match (project execution value)
        | Uorc.Core.Types::Value.sequence bytes => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.fitting (Uorc.Search.ResidualRope::start (project cursor target) (Uorc.Core.Bytes::fromBytes bytes))), completedEvents := ((project cursor completedEvents) + 1)}))
        | Uorc.Core.Types::Value.segmentedSequence rope => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.fitting (Uorc.Search.ResidualRope::start (project cursor target) rope)), completedEvents := ((project cursor completedEvents) + 1)}))
        | Uorc.Core.Types::Value.natural payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger))
        | Uorc.Core.Types::Value.byte payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger))
        | Uorc.Core.Types::Value.predicate payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger))
        | Uorc.Core.Types::Value.store payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger)))
      | Uorc.Core.ExecutionCredit::Tick.failed error usage executionLedger => (ctor Step.rejected (ctor Error.machineAccounted error usage executionLedger))
      | Uorc.Core.ExecutionCredit::Tick.rejected error executionLedger => (ctor Step.rejected (ctor Error.creditAccounted error executionLedger)))
    | Phase.fitting fitCursor => (match (Uorc.Search.ResidualRope::step fitCursor)
      | Uorc.Search.ResidualRope::Step.pending next => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.fitting next), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.ResidualRope::Step.complete exceptions count events => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.building (Uorc.Search.Residual::beginBuild (project cursor generator) (project cursor count) exceptions)), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.ResidualRope::Step.rejected error => (ctor Step.rejected (ctor Error.residual error)))
    | Phase.building builder => (match (Uorc.Search.Residual::stepFinish builder)
      | Uorc.Search.Residual::Finish.building nextBuilder => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.building (ctor Uorc.Search.Residual::Finish.building nextBuilder)), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.Residual::Finish.reversing generator remaining forward => (ctor Step.pending (record Cursor {target := (project cursor target), generator := (project cursor generator), count := (project cursor count), budget := (project cursor budget), phase := (ctor Phase.building (ctor Uorc.Search.Residual::Finish.reversing generator remaining forward)), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.Residual::Finish.done proposal => (ctor Step.proposal proposal ((project cursor completedEvents) + 1))))
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ remainingFuel => (match state
      | Step.pending cursor => (run remainingFuel (step cursor))
      | Step.proposal proposal events => (ctor Step.proposal proposal events)
      | Step.rejected error => (ctor Step.rejected error))),
  recursive_argument: fuel
}

\end{lexlean}
