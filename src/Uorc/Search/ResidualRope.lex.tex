definition{
  name: start,
  parameters: [target : Uorc.Core.Bytes::Rope, prediction : Uorc.Core.Bytes::Rope],

}

definition{
  name: readTarget,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor target)
    | Reader.state targetSource targetCached targetEnded => (match (Uorc.Core.Bytes::nextChunk targetSource 1)
      | Result.error ropeError => (ctor Step.rejected (ctor Error.rope ropeError))
      | Result.ok event => (match event
        | Uorc.Core.Bytes::CursorEvent.advanced next => (ctor Step.pending (record Cursor {target := (ctor Reader.state next targetCached targetEnded), prediction := (project cursor prediction), coordinate := (project cursor coordinate), exceptionsReversed := (project cursor exceptionsReversed), completedEvents := ((project cursor completedEvents) + 1)}))
        | Uorc.Core.Bytes::CursorEvent.produced chunk next => (if ((prim length chunk : Nat) =? 1) then (match (prim index chunk 0 : (Option UInt8))
          | Option.none => (ctor Step.rejected (ctor Error.invalidCursor))
          | Option.some byte => (ctor Step.pending (record Cursor {target := (ctor Reader.state next (ctor Option.some [UInt8] byte) targetEnded), prediction := (project cursor prediction), coordinate := (project cursor coordinate), exceptionsReversed := (project cursor exceptionsReversed), completedEvents := ((project cursor completedEvents) + 1)}))) else (ctor Step.rejected (ctor Error.invalidCursor)))
        | Uorc.Core.Bytes::CursorEvent.done => (ctor Step.pending (record Cursor {target := (ctor Reader.state targetSource targetCached true), prediction := (project cursor prediction), coordinate := (project cursor coordinate), exceptionsReversed := (project cursor exceptionsReversed), completedEvents := ((project cursor completedEvents) + 1)})))))
}

definition{
  name: readPrediction,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor prediction)
    | Reader.state predictionSource predictionCached predictionEnded => (match (Uorc.Core.Bytes::nextChunk predictionSource 1)
      | Result.error ropeError => (ctor Step.rejected (ctor Error.rope ropeError))
      | Result.ok event => (match event
        | Uorc.Core.Bytes::CursorEvent.advanced next => (ctor Step.pending (record Cursor {target := (project cursor target), prediction := (ctor Reader.state next predictionCached predictionEnded), coordinate := (project cursor coordinate), exceptionsReversed := (project cursor exceptionsReversed), completedEvents := ((project cursor completedEvents) + 1)}))
        | Uorc.Core.Bytes::CursorEvent.produced chunk next => (if ((prim length chunk : Nat) =? 1) then (match (prim index chunk 0 : (Option UInt8))
          | Option.none => (ctor Step.rejected (ctor Error.invalidCursor))
          | Option.some byte => (ctor Step.pending (record Cursor {target := (project cursor target), prediction := (ctor Reader.state next (ctor Option.some [UInt8] byte) predictionEnded), coordinate := (project cursor coordinate), exceptionsReversed := (project cursor exceptionsReversed), completedEvents := ((project cursor completedEvents) + 1)}))) else (ctor Step.rejected (ctor Error.invalidCursor)))
        | Uorc.Core.Bytes::CursorEvent.done => (ctor Step.pending (record Cursor {target := (project cursor target), prediction := (ctor Reader.state predictionSource predictionCached true), coordinate := (project cursor coordinate), exceptionsReversed := (project cursor exceptionsReversed), completedEvents := ((project cursor completedEvents) + 1)})))))
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor target)
    | Reader.state targetSource targetCached targetEnded => (match (project cursor prediction)
      | Reader.state predictionSource predictionCached predictionEnded => (match targetCached
        | Option.none => (if targetEnded then (match predictionCached
          | Option.none => (if predictionEnded then (ctor Step.complete (project cursor exceptionsReversed) (project cursor coordinate) ((project cursor completedEvents) + 1)) else (readPrediction cursor))
          | Option.some unusedByte => (ctor Step.rejected (ctor Error.lengthMismatch))) else (readTarget cursor))
        | Option.some targetByte => (match predictionCached
          | Option.none => (if predictionEnded then (ctor Step.rejected (ctor Error.lengthMismatch)) else (readPrediction cursor))
          | Option.some predictionByte => (if (prim equal targetByte predictionByte : Bool) then (ctor Step.pending (record Cursor {target := (ctor Reader.state targetSource (ctor Option.none [UInt8]) targetEnded), prediction := (ctor Reader.state predictionSource (ctor Option.none [UInt8]) predictionEnded), coordinate := ((project cursor coordinate) + 1), exceptionsReversed := (project cursor exceptionsReversed), completedEvents := ((project cursor completedEvents) + 1)})) else (match (Uorc.Search.Effort::checkedU64 (project cursor coordinate))
            | Result.error unusedError => (ctor Step.rejected (ctor Error.coordinateOverflow))
            | Result.ok coordinate => (ctor Step.pending (record Cursor {target := (ctor Reader.state targetSource (ctor Option.none [UInt8]) targetEnded), prediction := (ctor Reader.state predictionSource (ctor Option.none [UInt8]) predictionEnded), coordinate := ((project cursor coordinate) + 1), exceptionsReversed := (cons (record Uorc.Search.Residual::Exception {coordinate := coordinate, value := targetByte}) (project cursor exceptionsReversed)), completedEvents := ((project cursor completedEvents) + 1)})))))))),
  axioms: [Quot.sound, propext]
}

definition{
  name: runCursor,
  parameters: [fuel : Nat, cursor : Cursor],
  result: Step,
  body: (match fuel
    | Nat.zero => (ctor Step.pending cursor)
    | Nat.succ remainingFuel => (match (step cursor)
      | Step.pending nextCursor => (runCursor remainingFuel nextCursor)
      | Step.complete exceptionsValue count events => (ctor Step.complete exceptionsValue count events)
      | Step.rejected error => (ctor Step.rejected error))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match state
    | Step.pending cursor => (runCursor fuel cursor)
    | Step.complete exceptionsValue count events => (ctor Step.complete exceptionsValue count events)
    | Step.rejected error => (ctor Step.rejected error)),
  axioms: [Quot.sound, propext]
}

definition{
  name: exceptions,
  parameters: [state : Step],
  result: (Option (List Uorc.Search.Residual::Exception)),
  body: (match state
    | Step.pending cursor => (ctor Option.none [(List Uorc.Search.Residual::Exception)])
    | Step.complete exceptionsValue count events => (ctor Option.some [(List Uorc.Search.Residual::Exception)] exceptionsValue)
    | Step.rejected error => (ctor Option.none [(List Uorc.Search.Residual::Exception)]))
}
