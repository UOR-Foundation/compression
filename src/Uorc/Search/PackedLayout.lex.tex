definition{
  name: start,
  parameters: [payload : Uorc.Core.Bytes::Rope, wordCount : Nat, bitsPerIndex : Nat],
  result: Step,
  body: (match (Uorc.Search.Effort::checkedU64 wordCount)
    | Result.error failure => (ctor Step.rejected (ctor Error.addressSpaceExhausted) 0)
    | Result.ok words => (match (Uorc.Search.Effort::checkedU64 bitsPerIndex)
      | Result.error failure => (ctor Step.rejected (ctor Error.addressSpaceExhausted) 0)
      | Result.ok bits => (match (prim checked_multiply words bits : (Option UInt64))
        | Option.none => (ctor Step.rejected (ctor Error.addressSpaceExhausted) 0)
        | Option.some bitWord => (let bitCount : Nat := (Uorc.Core.Types::wordToNat bitWord); (let layoutValue : Layout := (record Layout {payload := payload, wordCount := wordCount, bitsPerIndex := bitsPerIndex, bitCount := bitCount, payloadBytes := ((prim quotient bitCount 8 0 : Nat) + (if ((prim remainder bitCount 8 0 : Nat) =? 0) then 0 else 1)), unusedHighBits := (if ((prim remainder bitCount 8 0 : Nat) =? 0) then 0 else (prim subtract 8 (prim remainder bitCount 8 0 : Nat) : Nat))}); (ctor Step.pending (record Cursor {layout := layoutValue, phase := (ctor Phase.validating (cons payload (nil Uorc.Core.Bytes::Rope))), observedBytes := 0, lastByte := (ctor Option.none [UInt8]), completedEvents := 0})))))))
}

definition{
  name: powerTwo,
  parameters: [exponent : Nat],
  result: Nat,
  body: (match exponent
    | Nat.zero => 1
    | Nat.succ remaining => (prim multiply 2 (powerTwo remaining) : Nat)),
  recursive_argument: exponent
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor phase)
    | Phase.validating remaining => (match remaining
      | List.nil => (ctor Step.pending (record Cursor {layout := (project cursor layout), phase := (ctor Phase.reading (Uorc.Core.Bytes::cursor (project (project cursor layout) payload))), observedBytes := (project cursor observedBytes), lastByte := (project cursor lastByte), completedEvents := ((project cursor completedEvents) + 1)}))
      | List.cons node rest => (match (Uorc.Core.Bytes::checkNode node)
        | Uorc.Core.Bytes::NodeStatus.valid => (match node
          | Uorc.Core.Bytes::Rope.empty => (ctor Step.pending (record Cursor {layout := (project cursor layout), phase := (ctor Phase.validating rest), observedBytes := (project cursor observedBytes), lastByte := (project cursor lastByte), completedEvents := ((project cursor completedEvents) + 1)}))
          | Uorc.Core.Bytes::Rope.leaf leafBytes => (ctor Step.pending (record Cursor {layout := (project cursor layout), phase := (ctor Phase.validating rest), observedBytes := (project cursor observedBytes), lastByte := (project cursor lastByte), completedEvents := ((project cursor completedEvents) + 1)}))
          | Uorc.Core.Bytes::Rope.view backing offset size => (ctor Step.pending (record Cursor {layout := (project cursor layout), phase := (ctor Phase.validating rest), observedBytes := (project cursor observedBytes), lastByte := (project cursor lastByte), completedEvents := ((project cursor completedEvents) + 1)}))
          | Uorc.Core.Bytes::Rope.branch cachedLength cachedHeight leftChild rightChild => (ctor Step.pending (record Cursor {layout := (project cursor layout), phase := (ctor Phase.validating (cons leftChild (cons rightChild rest))), observedBytes := (project cursor observedBytes), lastByte := (project cursor lastByte), completedEvents := ((project cursor completedEvents) + 1)})))
        | Uorc.Core.Bytes::NodeStatus.invalid => (ctor Step.rejected (ctor Error.invalidRope) ((project cursor completedEvents) + 1))
        | Uorc.Core.Bytes::NodeStatus.addressSpaceExhausted => (ctor Step.rejected (ctor Error.addressSpaceExhausted) ((project cursor completedEvents) + 1))))
    | Phase.reading reader => (match (Uorc.Core.Bytes::nextChunk reader 1)
      | Result.error failure => (ctor Step.rejected (ctor Error.rope failure) ((project cursor completedEvents) + 1))
      | Result.ok event => (match event
        | Uorc.Core.Bytes::CursorEvent.advanced next => (ctor Step.pending (record Cursor {layout := (project cursor layout), phase := (ctor Phase.reading next), observedBytes := (project cursor observedBytes), lastByte := (project cursor lastByte), completedEvents := ((project cursor completedEvents) + 1)}))
        | Uorc.Core.Bytes::CursorEvent.produced chunk next => (if ((prim length chunk : Nat) =? 1) then (match (prim index chunk 0 : (Option UInt8))
          | Option.none => (ctor Step.rejected (ctor Error.invalidCursor) ((project cursor completedEvents) + 1))
          | Option.some byte => (if ((project cursor observedBytes) <? (project (project cursor layout) payloadBytes)) then (ctor Step.pending (record Cursor {layout := (project cursor layout), phase := (ctor Phase.reading next), observedBytes := ((project cursor observedBytes) + 1), lastByte := (ctor Option.some [UInt8] byte), completedEvents := ((project cursor completedEvents) + 1)})) else (ctor Step.rejected (ctor Error.trailingPayload) ((project cursor completedEvents) + 1)))) else (ctor Step.rejected (ctor Error.invalidCursor) ((project cursor completedEvents) + 1)))
        | Uorc.Core.Bytes::CursorEvent.done => (if ((project cursor observedBytes) <? (project (project cursor layout) payloadBytes)) then (ctor Step.rejected (ctor Error.truncatedPayload) ((project cursor completedEvents) + 1)) else (if ((project cursor observedBytes) =? (project (project cursor layout) payloadBytes)) then (if ((project (project cursor layout) unusedHighBits) =? 0) then (ctor Step.complete (project cursor layout) ((project cursor completedEvents) + 1)) else (match (project cursor lastByte)
          | Option.none => (ctor Step.rejected (ctor Error.invalidCursor) ((project cursor completedEvents) + 1))
          | Option.some last => (if ((Uorc.Core.Types::wordToNat (Uorc.Core.Types::byteToNatural last)) <? (powerTwo (prim subtract 8 (project (project cursor layout) unusedHighBits) : Nat))) then (ctor Step.complete (project cursor layout) ((project cursor completedEvents) + 1)) else (ctor Step.rejected (ctor Error.nonzeroPadding) ((project cursor completedEvents) + 1))))) else (ctor Step.rejected (ctor Error.trailingPayload) ((project cursor completedEvents) + 1)))))))
}

definition{
  name: runWork,
  parameters: [work : Nat, state : Step],
  result: Step,
  body: (match work
    | Nat.zero => state
    | Nat.succ remainingWork => (match state
      | Step.pending cursor => (runWork remainingWork (step cursor))
      | Step.complete layout events => (ctor Step.complete layout events)
      | Step.rejected failure events => (ctor Step.rejected failure events))),
  recursive_argument: work
}

definition{
  name: accepted,
  parameters: [state : Step],
  result: Bool,
  body: (match state
    | Step.pending cursor => false
    | Step.complete layout events => true
    | Step.rejected failure events => false)
}

definition{
  name: rejection,
  parameters: [state : Step],
  result: (Option Error),
  body: (match state
    | Step.pending cursor => (ctor Option.none [Error])
    | Step.complete layout events => (ctor Option.none [Error])
    | Step.rejected failure events => (ctor Option.some [Error] failure))
}
