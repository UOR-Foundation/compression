definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor phase)
    | Phase.emitting emitter builder => (match (Uorc.Wire.GraphEmit::step emitter 1)
      | Uorc.Wire.GraphEmit::Event.advanced nextEmitter => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.emitting nextEmitter builder), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Wire.GraphEmit::Event.produced chunk nextEmitter => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.emitting nextEmitter (Uorc.Search.RopeBuilder::appendChunk builder chunk)), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Wire.GraphEmit::Event.complete size emitWork => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.building (Uorc.Search.RopeBuilder::finish builder)), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Wire.GraphEmit::Event.failed emitError position emitWork => (ctor Step.rejected (ctor Error.emit emitError) ((project cursor completedEvents) + 1)))
    | Phase.building builderCursor => (match (Uorc.Search.Normalize::step (record Uorc.Search.Normalize::Policy {blockBytes := 1}) builderCursor)
      | Uorc.Search.Normalize::Step.pending nextBuilder => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.building nextBuilder), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.Normalize::Step.done normalized physical => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project normalized rope), phase := (ctor Phase.parsing (ctor Uorc.Wire.GraphParse::Event.advanced (Uorc.Wire.GraphParse::start)) (Uorc.Core.Bytes::cursor (project normalized rope)) false), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.Normalize::Step.rejected normalizeError => (ctor Step.rejected (ctor Error.normalize normalizeError) ((project cursor completedEvents) + 1))
      | Uorc.Search.Normalize::Step.ropeFailure ropeError => (ctor Step.rejected (ctor Error.rope ropeError) ((project cursor completedEvents) + 1)))
    | Phase.parsing parserEvent reader ended => (match parserEvent
      | Uorc.Wire.GraphParse::Event.advanced parser => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.parsing (Uorc.Wire.GraphParse::step parser 1) reader ended), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Wire.GraphParse::Event.needInput parser => (if ended then (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.parsing (Uorc.Wire.GraphParse::finishInput parser) reader true), completedEvents := ((project cursor completedEvents) + 1)})) else (match (Uorc.Core.Bytes::nextChunk reader 1)
        | Result.error ropeError => (ctor Step.rejected (ctor Error.rope ropeError) ((project cursor completedEvents) + 1))
        | Result.ok readEvent => (match readEvent
          | Uorc.Core.Bytes::CursorEvent.advanced nextReader => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.parsing parserEvent nextReader ended), completedEvents := ((project cursor completedEvents) + 1)}))
          | Uorc.Core.Bytes::CursorEvent.produced chunk nextReader => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.parsing (Uorc.Wire.GraphParse::feed parser chunk) nextReader ended), completedEvents := ((project cursor completedEvents) + 1)}))
          | Uorc.Core.Bytes::CursorEvent.done => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.parsing (Uorc.Wire.GraphParse::finishInput parser) reader true), completedEvents := ((project cursor completedEvents) + 1)})))))
      | Uorc.Wire.GraphParse::Event.complete parsedGraph position parseWork => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.validating (Uorc.Core.Validation::start parsedGraph)), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Wire.GraphParse::Event.failed parseError position parseWork => (ctor Step.rejected (ctor Error.parse parseError) ((project cursor completedEvents) + 1)))
    | Phase.validating validation => (match (Uorc.Core.Validation::tick validation)
      | Uorc.Core.Validation::Step.pending nextValidation => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.validating nextValidation), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Core.Validation::Step.validated checkedGraph => (match (Uorc.Core.Resources::charge (project (project cursor query) budget) (project (project cursor query) initialUsage) (record Uorc.Core.Resources::Event {kind := (ctor Uorc.Core.Resources::EventKind.control), work := (project checkedGraph logicalWork), allocatedCells := 0, releasedCells := 0, emittedBytes := 0, enteredFrames := 0, exitedFrames := 0}))
        | Result.error error => (ctor Step.rejected (ctor Error.resource error) ((project cursor completedEvents) + 1))
        | Result.ok validationUsage => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.admitting checkedGraph (Uorc.Core.Basis::start (project (project cursor query) basis) validationUsage)), completedEvents := ((project cursor completedEvents) + 1)})))
      | Uorc.Core.Validation::Step.rejected staticError events visits => (ctor Step.rejected (ctor Error.static staticError) ((project cursor completedEvents) + 1))
      | Uorc.Core.Validation::Step.tableFailure tableError events visits => (ctor Step.rejected (ctor Error.table tableError) ((project cursor completedEvents) + 1)))
    | Phase.admitting checkedGraph basisCursor => (match (Uorc.Core.Basis::tick (project (project cursor query) budget) basisCursor)
      | Uorc.Core.Basis::Step.pending nextBasis => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.admitting checkedGraph nextBasis), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Core.Basis::Step.ready basisTable usage => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.executing checkedGraph basisTable (Uorc.Core.ExecutionCredit::startInitialization checkedGraph basisTable (project (project cursor query) budget) usage)), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Core.Basis::Step.failed error usage => (ctor Step.rejected (ctor Error.admission error) ((project cursor completedEvents) + 1)))
    | Phase.executing checkedGraph basisTable executionCredit => (match (Uorc.Core.ExecutionCredit::tick (Uorc.Core.StepMachine::contextForAdmitted checkedGraph basisTable (project (project cursor query) budget)) executionCredit)
      | Uorc.Core.ExecutionCredit::Tick.pending nextCredit => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.executing checkedGraph basisTable nextCredit), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Core.ExecutionCredit::Tick.done execution executionLedger => (match (project execution value)
        | Uorc.Core.Types::Value.sequence bytes => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.checking (project checkedGraph graph) (project execution usage) executionLedger (Uorc.Search.VerifyRope::start (cons (project (project cursor query) target) (nil Uorc.Core.Bytes::Rope)) (cons (Uorc.Core.Bytes::fromBytes bytes) (nil Uorc.Core.Bytes::Rope)))), completedEvents := ((project cursor completedEvents) + 1)}))
        | Uorc.Core.Types::Value.segmentedSequence rope => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.checking (project checkedGraph graph) (project execution usage) executionLedger (Uorc.Search.VerifyRope::start (cons (project (project cursor query) target) (nil Uorc.Core.Bytes::Rope)) (cons rope (nil Uorc.Core.Bytes::Rope)))), completedEvents := ((project cursor completedEvents) + 1)}))
        | Uorc.Core.Types::Value.natural payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger) ((project cursor completedEvents) + 1))
        | Uorc.Core.Types::Value.byte payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger) ((project cursor completedEvents) + 1))
        | Uorc.Core.Types::Value.predicate payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger) ((project cursor completedEvents) + 1))
        | Uorc.Core.Types::Value.store payload => (ctor Step.rejected (ctor Error.outputTypeAccounted executionLedger) ((project cursor completedEvents) + 1)))
      | Uorc.Core.ExecutionCredit::Tick.failed executionError failedUsage executionLedger => (ctor Step.rejected (ctor Error.executionAccounted executionError failedUsage executionLedger) ((project cursor completedEvents) + 1))
      | Uorc.Core.ExecutionCredit::Tick.rejected creditError executionLedger => (ctor Step.rejected (ctor Error.creditAccounted creditError executionLedger) ((project cursor completedEvents) + 1)))
    | Phase.checking parsedGraph usage executionLedger comparison => (match (Uorc.Search.VerifyRope::step comparison)
      | Uorc.Search.VerifyRope::Step.pending nextComparison => (ctor Step.pending (record Cursor {query := (project cursor query), wire := (project cursor wire), phase := (ctor Phase.checking parsedGraph usage executionLedger nextComparison), completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.VerifyRope::Step.matched bytesCompared boundaries => (ctor Step.complete (record CheckedProgram {graph := parsedGraph, wire := (project cursor wire), usage := usage, executionLedger := executionLedger, bytesCompared := bytesCompared, completedEvents := ((project cursor completedEvents) + 1)}))
      | Uorc.Search.VerifyRope::Step.mismatch mismatch => (ctor Step.rejected (ctor Error.mismatchAccounted mismatch executionLedger) ((project cursor completedEvents) + 1))
      | Uorc.Search.VerifyRope::Step.ropeFailure ropeError => (ctor Step.rejected (ctor Error.verificationRopeAccounted ropeError executionLedger) ((project cursor completedEvents) + 1)))),
  axioms: [Quot.sound, propext]
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ remainingFuel => (match state
      | Step.pending cursor => (run remainingFuel (step cursor))
      | Step.complete checkedProgram => (ctor Step.complete checkedProgram)
      | Step.rejected error events => (ctor Step.rejected error events))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: accepted,
  parameters: [state : Step],
  result: Bool,
  body: (match state
    | Step.pending cursor => false
    | Step.complete checkedProgram => true
    | Step.rejected error events => false)
}
