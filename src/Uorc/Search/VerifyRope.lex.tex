definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor expectedCursor)
    | Option.none => (match (project cursor actualCursor)
      | Option.none => (match (project cursor expected)
        | List.nil => (match (project cursor actual)
          | List.nil => (ctor Step.matched (project cursor bytesCompared) (project cursor boundariesChecked))
          | List.cons extraObject extraRest => (failure cursor (ctor Uorc.Search.Verify::MismatchKind.objectCount)))
        | List.cons expectedObject expectedRest => (match (project cursor actual)
          | List.nil => (failure cursor (ctor Uorc.Search.Verify::MismatchKind.objectCount))
          | List.cons actualObject actualRest => (ctor Step.pending (record Cursor {expected := expectedRest, actual := actualRest, expectedCursor := (ctor Option.some [Uorc.Core.Bytes::Cursor] (Uorc.Core.Bytes::cursor expectedObject)), actualCursor := (ctor Option.some [Uorc.Core.Bytes::Cursor] (Uorc.Core.Bytes::cursor actualObject)), expectedByte := (project cursor expectedByte), actualByte := (project cursor actualByte), expectedEnded := (project cursor expectedEnded), actualEnded := (project cursor actualEnded), objectIndex := (project cursor objectIndex), byteOffset := (project cursor byteOffset), bytesCompared := (project cursor bytesCompared), boundariesChecked := (project cursor boundariesChecked)}))))
      | Option.some unexpectedActualReader => (failure cursor (ctor Uorc.Search.Verify::MismatchKind.invalidCursor)))
    | Option.some expectedReader => (match (project cursor actualCursor)
      | Option.none => (failure cursor (ctor Uorc.Search.Verify::MismatchKind.invalidCursor))
      | Option.some actualReader => (compareCurrent cursor expectedReader actualReader)))
}

definition{
  name: run,
  parameters: [fuel : Nat, cursor : Cursor],
  result: Step,
  body: (match fuel
    | Nat.zero => (ctor Step.pending cursor)
    | Nat.succ restFuel => (match (step cursor)
      | Step.pending next => (run restFuel next)
      | Step.matched compared boundaries => (ctor Step.matched compared boundaries)
      | Step.mismatch mismatch => (ctor Step.mismatch mismatch)
      | Step.ropeFailure ropeError => (ctor Step.ropeFailure ropeError))),
  recursive_argument: fuel
}

definition{
  name: stepTrial,
  parameters: [trial : Trial],
  result: TrialStep,
  body: (match (step (project trial cursor))
    | Step.pending next => (ctor TrialStep.pending (record Trial {wire := (project trial wire), cursor := next}))
    | Step.matched compared boundaries => (ctor TrialStep.eligible (record CheckedArchive {wire := (project trial wire), bytesCompared := compared, boundariesChecked := boundaries}))
    | Step.mismatch mismatch => (ctor TrialStep.rejected mismatch)
    | Step.ropeFailure ropeError => (ctor TrialStep.ropeFailure ropeError))
}
