\begin{lexlean}{Uorc.Search.Boundaries}
\title{Parameters}
\begin{semanticmodule}
structure{
  name: Cursor,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: expected, type: (List Nat)},
    {name: actual, type: (List Nat)},
    {name: objectIndex, type: Nat}
  ]
}

inductive{
  name: Step,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: pending, fields: [Cursor]},
    {name: matched, fields: [Nat]},
    {name: mismatch, fields: [Nat]}
  ]
}

definition{
  name: start,
  parameters: [expected : (List Nat), actual : (List Nat)],
  result: Cursor,
  body: (record Cursor {expected := expected, actual := actual, objectIndex := 0})
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor expected)
    | List.nil => (match (project cursor actual)
      | List.nil => (ctor Step.matched (project cursor objectIndex))
      | List.cons extra remainingExtra => (ctor Step.mismatch (project cursor objectIndex)))
    | List.cons expectedLength remainingExpected => (match (project cursor actual)
      | List.nil => (ctor Step.mismatch (project cursor objectIndex))
      | List.cons actualLength remainingActual => (if (expectedLength =? actualLength) then (ctor Step.pending (record Cursor {expected := remainingExpected, actual := remainingActual, objectIndex := ((project cursor objectIndex) + 1)})) else (ctor Step.mismatch (project cursor objectIndex)))))
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ remainingFuel => (match state
      | Step.pending cursor => (run remainingFuel (step cursor))
      | Step.matched objects => (ctor Step.matched objects)
      | Step.mismatch index => (ctor Step.mismatch index))),
  recursive_argument: fuel
}

theorem{
  name: empty_objects_remain_ordered_boundaries,
  parameters: [],
  statement: ((run 20 (ctor Step.pending (start (cons 0 (cons 2 (cons 0 (nil Nat)))) (cons 0 (cons 2 (cons 0 (nil Nat))))))) = (ctor Step.matched 3)),
  proof: reflexivity{}
}

theorem{
  name: same_flat_bytes_do_not_hide_different_partition,
  parameters: [],
  statement: ((run 20 (ctor Step.pending (start (cons 1 (cons 2 (nil Nat))) (cons 2 (cons 1 (nil Nat)))))) = (ctor Step.mismatch 0)),
  proof: reflexivity{}
}

theorem{
  name: empty_collection_is_not_one_empty_object,
  parameters: [],
  statement: ((run 20 (ctor Step.pending (start (nil Nat) (cons 0 (nil Nat))))) = (ctor Step.mismatch 0)),
  proof: reflexivity{}
}
\end{semanticmodule}
\end{lexlean}
