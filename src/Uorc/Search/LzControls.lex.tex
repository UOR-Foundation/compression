definition{
  name: instructionLiterals,
  parameters: [instructions : (List Uorc.Core.Graph::Instruction)],
  result: Nat,
  body: (match instructions
    | List.nil => 0
    | List.cons instruction rest => ((instructionLiteralBytes instruction) + (instructionLiterals rest))),
  recursive_argument: instructions
}

definition{
  name: blockLiterals,
  parameters: [blocks : (List Uorc.Core.Graph::Block)],
  result: Nat,
  body: (match blocks
    | List.nil => 0
    | List.cons block rest => ((instructionLiterals (project block instructions)) + (blockLiterals rest))),
  recursive_argument: blocks
}

definition{
  name: fitsExactly,
  parameters: [state : Uorc.Search.LzFit::Step, expected : Bytes],
  result: Bool,
  body: (match state
    | Uorc.Search.LzFit::Step.pending cursor => false
    | Uorc.Search.LzFit::Step.proposal graph events table literals references => ((Uorc.Search.PackedControls::meaningFits graph expected) && (((blockLiterals (project graph blocks)) =? literals) && ((literals + references) =? (prim length expected : Nat))))
    | Uorc.Search.LzFit::Step.rejected failure events => false),
  axioms: [Quot.sound, propext]
}

definition{
  name: usedReference,
  parameters: [state : Uorc.Search.LzFit::Step],

  body: (match state

    | Uorc.Search.LzFit::Step.proposal graph events table literals references => (0 <? references)
    | Uorc.Search.LzFit::Step.rejected failure events => false)
}
