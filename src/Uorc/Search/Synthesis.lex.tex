definition{
  name: cursorAfter,
  parameters: [state : Step],
  result: (Option Cursor),
  body: (match state
    | Step.advanced cursor => (ctor Option.some [Cursor] cursor)
    | Step.proposal phase round candidate cursor => (ctor Option.some [Cursor] cursor)
    | Step.familyFailure phase error cursor => (ctor Option.some [Cursor] cursor)
    | Step.finished cursor => (ctor Option.some [Cursor] cursor)
    | Step.invalidPolicy => (ctor Option.none [Cursor]))

definition{
  name: advance,
  parameters: [fuel : Nat, cursor : Cursor],
  result: (Option Cursor),
  body: (match fuel
    | Nat.zero => (ctor Option.some [Cursor] cursor)
    | Nat.succ remainingFuel => (match (cursorAfter (step cursor))
      | Option.none => (ctor Option.none [Cursor])
      | Option.some next => (advance remainingFuel next))),
  recursive_argument: fuel
}

  result: Bool,

    | Step.advanced next => false
    | Step.proposal phase round candidate next => false
    | Step.familyFailure phase error next => false
    | Step.finished next => true
    | Step.invalidPolicy => false)
}
