definition{
  name: activeStep,
  parameters: [
    cursor : Cursor,
    available : UInt64,
    activeArena : Uorc.Core.Heap::Arena,
    header : Header,
    stage : Stage
  ],
  result: Transition,
  body: (match stage
    | Stage.splitOriginal => (match (Uorc.Core.RuntimeOwnerRoots::startSplit activeArena (ctor Uorc.Core.RuntimeOwnerRoots::Role.search) (project (project header caller) search))
      | Result.error ownerFailure => (record Transition {cursor := (record Cursor {budget := (project cursor budget), phase := (project cursor phase), evidence := (project cursor evidence), paid := (project cursor paid), consumedCredits := (project cursor consumedCredits), attemptedOperations := (project cursor attemptedOperations), failure := (ctor Option.some [Error] (ctor Error.owner ownerFailure))}), remainingCredits := available, outcome := (ctor Outcome.failed (ctor Error.owner ownerFailure))})
      | Result.ok splitCursor => (observeSplit cursor available splitCursor (record Header {program := (project header program), caller := (record Uorc.Core.RuntimeProgram::CallerRoots {input := (project (project header caller) input), archive := (project (project header caller) archive), search := (Uorc.Core.RuntimeValue::nullHandle), checkpoint := (project (project header caller) checkpoint), certificate := (project (project header caller) certificate), verification := (project (project header caller) verification), wire := (project (project header caller) wire)}), usage := (project header usage), producer := (project header producer), control := (project header control), walk := (project header walk), walkDepth := (project header walkDepth), originalSearch := (project header originalSearch)})))
    | Stage.findControl => (findControl cursor available activeArena header)
    | Stage.retainControl => (match (Uorc.Core.Heap::retain activeArena (project header control))
      | Result.error heapFailure => (record Transition {cursor := (record Cursor {budget := (project cursor budget), phase := (project cursor phase), evidence := (project cursor evidence), paid := (project cursor paid), consumedCredits := (project cursor consumedCredits), attemptedOperations := (project cursor attemptedOperations), failure := (ctor Option.some [Error] (ctor Error.heap heapFailure))}), remainingCredits := available, outcome := (ctor Outcome.failed (ctor Error.heap heapFailure))})
      | Result.ok mutation => (observe cursor available (project mutation arena) header (ctor Stage.releaseProducer)))
    | Stage.releaseProducer => (heapResult cursor available (record Header {program := (project header program), caller := (project header caller), usage := (project header usage), producer := (Uorc.Core.RuntimeValue::nullHandle), control := (project header control), walk := (project header walk), walkDepth := (project header walkDepth), originalSearch := (project header originalSearch)}) (ctor Stage.drainProducer) (Uorc.Core.Heap::release activeArena (project header producer)))
    | Stage.drainProducer => (match (Uorc.Core.Heap::releaseStep activeArena)
      | Result.error heapFailure => (record Transition {cursor := (record Cursor {budget := (project cursor budget), phase := (project cursor phase), evidence := (project cursor evidence), paid := (project cursor paid), consumedCredits := (project cursor consumedCredits), attemptedOperations := (project cursor attemptedOperations), failure := (ctor Option.some [Error] (ctor Error.heap heapFailure))}), remainingCredits := available, outcome := (ctor Outcome.failed (ctor Error.heap heapFailure))})
      | Result.ok mutation => (observe cursor available (project mutation arena) header (if (project mutation complete) then (ctor Stage.releaseControl) else stage)))
    | Stage.releaseControl => (heapResult cursor available (record Header {program := (project header program), caller := (project header caller), usage := (project header usage), producer := (project header producer), control := (Uorc.Core.RuntimeValue::nullHandle), walk := (project header walk), walkDepth := (project header walkDepth), originalSearch := (project header originalSearch)}) (ctor Stage.drainControl) (Uorc.Core.Heap::release activeArena (project header control)))
    | Stage.drainControl => (match (Uorc.Core.Heap::releaseStep activeArena)
      | Result.error heapFailure => (record Transition {cursor := (record Cursor {budget := (project cursor budget), phase := (project cursor phase), evidence := (project cursor evidence), paid := (project cursor paid), consumedCredits := (project cursor consumedCredits), attemptedOperations := (project cursor attemptedOperations), failure := (ctor Option.some [Error] (ctor Error.heap heapFailure))}), remainingCredits := available, outcome := (ctor Outcome.failed (ctor Error.heap heapFailure))})
      | Result.ok mutation => (observe cursor available (project mutation arena) header (if (project mutation complete) then (ctor Stage.complete) else stage)))
    | Stage.complete => (record Transition {cursor := cursor, remainingCredits := available, outcome := (ctor Outcome.released)}))
}

    | Phase.active activeArena header stage => (match stage
      | Stage.splitOriginal => (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.ownerSplitStart))
      | Stage.findControl => (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.metadataMutation))

      | Stage.releaseControl => (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.heapRelease))
      | Stage.drainControl => (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.heapReleaseStep))
      | Stage.complete => (0 : UInt64)))
}
