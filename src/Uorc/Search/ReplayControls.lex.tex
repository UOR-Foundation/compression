definition{
  name: seekWords,
  parameters: [work : Nat, state : Uorc.Search.Replay::Step],
  result: ProbeSetup,
  body: (match work
    | Nat.zero => (ctor ProbeSetup.pending state)
    | Nat.succ remaining => (match state
      | Uorc.Search.Replay::Step.pending cursor => (match (project cursor phase)
        | Uorc.Search.Replay::Phase.words nextWords => (ctor ProbeSetup.ready cursor nextWords)
        | Uorc.Search.Replay::Phase.queryIdentity c => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.sealIdentity c => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.winnerTrial s => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.winnerPreparation c => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.rawTrial s => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.ordering r e o w => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.parsing b e r f w => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.validating b c w => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.measuring g c w => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.predecessorTrial s w => (seekWords remaining (Uorc.Search.Replay::step cursor)))
      | Uorc.Search.Replay::Step.counterexample c => (ctor ProbeSetup.unavailable)
      | Uorc.Search.Replay::Step.completeBodyReplay c => (ctor ProbeSetup.unavailable)
      | Uorc.Search.Replay::Step.incomplete e c => (ctor ProbeSetup.unavailable))),
  recursive_argument: work
}

definition{
  name: seekBlocks,
  parameters: [blocks : Nat, setup : ProbeSetup],
  result: ProbeSetup,
  body: (match blocks
    | Nat.zero => setup
    | Nat.succ remaining => (match setup
      | ProbeSetup.pending state => (seekBlocks remaining (seekWords 1000 state))

}
