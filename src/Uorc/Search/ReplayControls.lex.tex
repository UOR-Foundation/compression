  parameters: [work : Nat, state : Uorc.Search.Replay::Step],
  result: ProbeSetup,
  body: (match work
    | Nat.zero => (ctor ProbeSetup.pending state)
    | Nat.succ remaining => (match state
      | Uorc.Search.Replay::Step.pending cursor => (match (project cursor phase)
        | Uorc.Search.Replay::Phase.words nextWords => (ctor ProbeSetup.ready cursor nextWords)
        | Uorc.Search.Replay::Phase.queryIdentity c => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.sealIdentity c => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.winnerTrial s => (seekWords remaining (Uorc.Search.Replay::step cursor))
        | Uorc.Search.Replay::Phase.winnerPreparation c => (seekWords remaining (Uorc.Search.Replay::step cursor))
