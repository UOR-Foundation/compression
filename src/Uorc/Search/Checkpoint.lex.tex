definition{
  name: start,
  parameters: [record : Record, query : Uorc.Search.Identity::Snapshot, policy : PolicySnapshot],
  result: Cursor,
  body: (record Cursor {record := var{
    name: record
  }, query := query, policy := policy, phase := (ctor Phase.query (Uorc.Search.Identity::start (project var{
    name: record
  } query) query))})
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor phase)
    | Phase.query comparison => (match (Uorc.Search.Identity::step comparison)
      | Uorc.Search.Identity::Step.pending next => (ctor Step.pending (record Cursor {record := (project cursor record), query := (project cursor query), policy := (project cursor policy), phase := (ctor Phase.query next)}))
      | Uorc.Search.Identity::Step.matched => (if ((project (project cursor policy) quantum) =? 0) then (ctor Step.rejected (ctor Error.invalidQuantum)) else (if ((project (project cursor policy) quantum) =? (project (project (project cursor record) policy) quantum)) then (ctor Step.pending (record Cursor {record := (project cursor record), query := (project cursor query), policy := (project cursor policy), phase := (ctor Phase.source (Uorc.Search.VerifyRope::start (cons (project (project (project cursor record) policy) sourceIdentity) (nil Uorc.Core.Bytes::Rope)) (cons (project (project cursor policy) sourceIdentity) (nil Uorc.Core.Bytes::Rope))))})) else (ctor Step.rejected (ctor Error.policyMismatch))))
      | Uorc.Search.Identity::Step.different field => (ctor Step.rejected (ctor Error.queryMismatch field))
      | Uorc.Search.Identity::Step.invalidScope => (ctor Step.rejected (ctor Error.invalidScope))
      | Uorc.Search.Identity::Step.ropeFailure error => (ctor Step.rejected (ctor Error.rope error)))
    | Phase.source comparison => (match (Uorc.Search.VerifyRope::step comparison)
      | Uorc.Search.VerifyRope::Step.pending next => (ctor Step.pending (record Cursor {record := (project cursor record), query := (project cursor query), policy := (project cursor policy), phase := (ctor Phase.source next)}))
      | Uorc.Search.VerifyRope::Step.matched bytes boundaries => (ctor Step.pending (record Cursor {record := (project cursor record), query := (project cursor query), policy := (project cursor policy), phase := (ctor Phase.seeds (Uorc.Search.VerifyRope::start (cons (project (project (project cursor record) policy) seedInventory) (nil Uorc.Core.Bytes::Rope)) (cons (project (project cursor policy) seedInventory) (nil Uorc.Core.Bytes::Rope))))}))
      | Uorc.Search.VerifyRope::Step.mismatch mismatch => (ctor Step.rejected (ctor Error.policyMismatch))
      | Uorc.Search.VerifyRope::Step.ropeFailure error => (ctor Step.rejected (ctor Error.rope error)))
    | Phase.seeds comparison => (match (Uorc.Search.VerifyRope::step comparison)
      | Uorc.Search.VerifyRope::Step.pending next => (ctor Step.pending (record Cursor {record := (project cursor record), query := (project cursor query), policy := (project cursor policy), phase := (ctor Phase.seeds next)}))
      | Uorc.Search.VerifyRope::Step.matched bytes boundaries => (ctor Step.pending (record Cursor {record := (project cursor record), query := (project cursor query), policy := (project cursor policy), phase := (ctor Phase.sharing (Uorc.Search.VerifyRope::start (cons (project (project (project cursor record) policy) sharingInventory) (nil Uorc.Core.Bytes::Rope)) (cons (project (project cursor policy) sharingInventory) (nil Uorc.Core.Bytes::Rope))))}))
      | Uorc.Search.VerifyRope::Step.mismatch mismatch => (ctor Step.rejected (ctor Error.policyMismatch))
      | Uorc.Search.VerifyRope::Step.ropeFailure error => (ctor Step.rejected (ctor Error.rope error)))
    | Phase.sharing comparison => (match (Uorc.Search.VerifyRope::step comparison)
      | Uorc.Search.VerifyRope::Step.pending next => (ctor Step.pending (record Cursor {record := (project cursor record), query := (project cursor query), policy := (project cursor policy), phase := (ctor Phase.sharing next)}))
      | Uorc.Search.VerifyRope::Step.matched bytes boundaries => (ctor Step.replayRequired (record ReplayRequest {query := (project cursor query), policy := (project cursor policy), canonicalPrefix := (project (project cursor record) effortPrefix), incumbentWireToRecheck := (project (project cursor record) incumbentWireToRecheck)}))
      | Uorc.Search.VerifyRope::Step.mismatch mismatch => (ctor Step.rejected (ctor Error.policyMismatch))
      | Uorc.Search.VerifyRope::Step.ropeFailure error => (ctor Step.rejected (ctor Error.rope error))))
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ remainingFuel => (match state
      | Step.pending cursor => (run remainingFuel (step cursor))
      | Step.replayRequired request => (ctor Step.replayRequired request)
      | Step.rejected error => (ctor Step.rejected error))),
  recursive_argument: fuel
}
