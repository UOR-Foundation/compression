definition{
  name: requiredWork,
  parameters: [cursor : Cursor],
  result: UInt64,
  body: (match (project cursor phase)
    | Phase.allocateControl => (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.controlAllocate))
    | Phase.pinPayload => (match (project cursor payloads)
      | List.nil => (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.control))
      | List.cons payload remainingPayloads => (if (Uorc.Core.RuntimeValue::isNull payload) then (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.control)) else (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.ownerPin))))
    | Phase.install => (if (Uorc.Core.RuntimeValue::isNull (project cursor originalSearch)) then (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.control)) else (Uorc.Core.RuntimeWork::fee (ctor Uorc.Core.RuntimeWork::Operation.ownerJoin)))
    | Phase.ready => (0 : UInt64))
}

  result: Transition,
  body: (match (project cursor phase)

    | Option.some needed => (let taken : UInt64 := (if (Uorc.Core.Heap::less available needed) then available else needed); (match (prim checked_subtract available taken : (Option UInt64))
      | Option.none => (record Transition {cursor := (record Cursor {arena := (project cursor arena), basis := (project cursor basis), caller := (project cursor caller), usage := (project cursor usage), source := (project cursor source), payloads := (project cursor payloads), owner := (project cursor owner), originalSearch := (project cursor originalSearch), joinedOriginal := (project cursor joinedOriginal), phase := (project cursor phase), paid := (project cursor paid), consumedCredits := (project cursor consumedCredits), attemptedOperations := (project cursor attemptedOperations), failure := (ctor Option.some [Error] (ctor Error.invalidFunding))}), remainingCredits := available, outcome := (ctor Outcome.failed (ctor Error.invalidFunding))})

        | Option.none => (record Transition {cursor := (record Cursor {arena := (project cursor arena), basis := (project cursor basis), caller := (project cursor caller), usage := (project cursor usage), source := (project cursor source), payloads := (project cursor payloads), owner := (project cursor owner), originalSearch := (project cursor originalSearch), joinedOriginal := (project cursor joinedOriginal), phase := (project cursor phase), paid := (project cursor paid), consumedCredits := (project cursor consumedCredits), attemptedOperations := (project cursor attemptedOperations), failure := (ctor Option.some [Error] (ctor Error.counterExhausted))}), remainingCredits := available, outcome := (ctor Outcome.failed (ctor Error.counterExhausted))})
        | Option.some consumed => (match (prim checked_add (project cursor paid) taken : (Option UInt64))
          | Option.none => (record Transition {cursor := (record Cursor {arena := (project cursor arena), basis := (project cursor basis), caller := (project cursor caller), usage := (project cursor usage), source := (project cursor source), payloads := (project cursor payloads), owner := (project cursor owner), originalSearch := (project cursor originalSearch), joinedOriginal := (project cursor joinedOriginal), phase := (project cursor phase), paid := (project cursor paid), consumedCredits := (project cursor consumedCredits), attemptedOperations := (project cursor attemptedOperations), failure := (ctor Option.some [Error] (ctor Error.counterExhausted))}), remainingCredits := available, outcome := (ctor Outcome.failed (ctor Error.counterExhausted))})

definition{
  name: step,
  parameters: [budget : Uorc.Core.RuntimeResources::Budget, cursor : Cursor, available : UInt64],
  result: Transition,
  body: (match (project cursor failure)
    | Option.some retainedFailure => (record Transition {cursor := cursor, remainingCredits := available, outcome := (ctor Outcome.failed retainedFailure)})
    | Option.none => (let fee : UInt64 := (requiredWork cursor); (if (prim equal fee (0 : UInt64) : Bool) then (record Transition {cursor := cursor, remainingCredits := available, outcome := (ctor Outcome.ready)}) else (if (prim equal available (0 : UInt64) : Bool) then (record Transition {cursor := cursor, remainingCredits := available, outcome := (ctor Outcome.pending)}) else (fund budget cursor available fee)))))
}

definition{
  name: contract,
  parameters: [],
  result: String,
  body: "Preparation allocates the shared 4096-byte fixed control reservation and pins each of the at-most-three fixed-template logical sequence dependencies through real arena metadata edges, one bounded pin per step. It never copies or coalesces literal payloads, never retains a Graph AST, and never treats the reservation as variable-data backing. The original search ticket is moved into the registered search-owner aggregate and its borrowed identifier is retained for later restoration; the caller search field carries the aggregate ticket. Actual shared RuntimeWork task fees are completely paid and RuntimeResources usage charged before owner calls, and the resulting arena is observed. Partial payment and all failure roots stay explicit. The initial basis/caller root inventory and any borrowed payload dependencies must be live in the supplied arena. Only phase ready can be moved into admission; the caller must eventually split/restore original search ownership and stepwise release the producer aggregate after full admission acknowledgment or cancellation. This preparation token itself is not an admitted program or archive incumbent."
}
