definition{
  name: nextByte,
  parameters: [value : UInt8],
  result: UInt8,
  body: (match (prim checked_add value (1 : UInt8) : (Option UInt8))
    | Option.none => (0 : UInt8)
    | Option.some converted => converted),
  axioms: [Quot.sound, propext]
}

definition{
  name: fromNat,
  parameters: [remaining : Nat],
  result: UInt8,
  body: (match remaining
    | Nat.zero => (0 : UInt8)
    | Nat.succ previous => (nextByte (fromNat previous))),
  axioms: [Quot.sound, propext],
  recursive_argument: remaining
}
