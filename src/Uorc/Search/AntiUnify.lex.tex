definition{
  name: tasksStep,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor tasks)
    | List.nil => (match (project cursor values)
      | List.nil => (ctor Step.invalidCursor)
      | List.cons expressionValue otherValues => (match otherValues
        | List.nil => (ctor Step.complete (record Generalization {expression := expressionValue, bindings := (project cursor bindings), completedEvents := ((project cursor completedEvents) + 1)}))
        | List.cons extraValue extraValues => (ctor Step.invalidCursor)))
    | List.cons task restTasks => (match task
      | Task.generalize leftExpression rightExpression => (if (Uorc.Core.Types::sameType (Uorc.Search.Expression::resultType leftExpression) (Uorc.Search.Expression::resultType rightExpression)) then (ctor Step.pending (record Cursor {mode := (ctor Mode.compareEqual), tasks := restTasks, values := (project cursor values), bindings := (project cursor bindings), nextParameter := (project cursor nextParameter), activeLeft := leftExpression, activeRight := rightExpression, equality := (ctor Option.some [Uorc.Search.Expression::EqualityCursor] (Uorc.Search.Expression::startEquality leftExpression rightExpression)), remainingBindings := (project cursor remainingBindings), candidateBinding := (project cursor candidateBinding), arityLeft := (project cursor arityLeft), arityRight := (project cursor arityRight), completedEvents := ((project cursor completedEvents) + 1)})) else (ctor Step.typeMismatch (Uorc.Search.Expression::resultType leftExpression) (Uorc.Search.Expression::resultType rightExpression)))
      | Task.arguments leftArguments rightArguments reverseArguments functionIndex type => (match leftArguments
        | List.nil => (match rightArguments
          | List.nil => (ctor Step.pending (record Cursor {mode := (project cursor mode), tasks := (cons (ctor Task.reverseArguments reverseArguments (nil Uorc.Search.Expression::Expression) functionIndex type) restTasks), values := (project cursor values), bindings := (project cursor bindings), nextParameter := (project cursor nextParameter), activeLeft := (project cursor activeLeft), activeRight := (project cursor activeRight), equality := (project cursor equality), remainingBindings := (project cursor remainingBindings), candidateBinding := (project cursor candidateBinding), arityLeft := (project cursor arityLeft), arityRight := (project cursor arityRight), completedEvents := ((project cursor completedEvents) + 1)}))
          | List.cons extraRight otherRight => (ctor Step.invalidCursor))
        | List.cons leftHead leftTail => (match rightArguments
          | List.nil => (ctor Step.invalidCursor)
          | List.cons rightHead rightTail => (ctor Step.pending (record Cursor {mode := (project cursor mode), tasks := (cons (ctor Task.generalize leftHead rightHead) (cons (ctor Task.collectArgument leftTail rightTail reverseArguments functionIndex type) restTasks)), values := (project cursor values), bindings := (project cursor bindings), nextParameter := (project cursor nextParameter), activeLeft := (project cursor activeLeft), activeRight := (project cursor activeRight), equality := (project cursor equality), remainingBindings := (project cursor remainingBindings), candidateBinding := (project cursor candidateBinding), arityLeft := (project cursor arityLeft), arityRight := (project cursor arityRight), completedEvents := ((project cursor completedEvents) + 1)}))))
      | Task.collectArgument leftArguments rightArguments reverseArguments functionIndex type => (match (project cursor values)
        | List.nil => (ctor Step.invalidCursor)
        | List.cons argument otherValues => (ctor Step.pending (record Cursor {mode := (project cursor mode), tasks := (cons (ctor Task.arguments leftArguments rightArguments (cons argument reverseArguments) functionIndex type) restTasks), values := otherValues, bindings := (project cursor bindings), nextParameter := (project cursor nextParameter), activeLeft := (project cursor activeLeft), activeRight := (project cursor activeRight), equality := (project cursor equality), remainingBindings := (project cursor remainingBindings), candidateBinding := (project cursor candidateBinding), arityLeft := (project cursor arityLeft), arityRight := (project cursor arityRight), completedEvents := ((project cursor completedEvents) + 1)})))
      | Task.reverseArguments remaining forward functionIndex type => (match remaining
        | List.nil => (ctor Step.pending (record Cursor {mode := (project cursor mode), tasks := restTasks, values := (cons (ctor Uorc.Search.Expression::Expression.call functionIndex type forward) (project cursor values)), bindings := (project cursor bindings), nextParameter := (project cursor nextParameter), activeLeft := (project cursor activeLeft), activeRight := (project cursor activeRight), equality := (project cursor equality), remainingBindings := (project cursor remainingBindings), candidateBinding := (project cursor candidateBinding), arityLeft := (project cursor arityLeft), arityRight := (project cursor arityRight), completedEvents := ((project cursor completedEvents) + 1)}))
        | List.cons argument rest => (ctor Step.pending (record Cursor {mode := (project cursor mode), tasks := (cons (ctor Task.reverseArguments rest (cons argument forward) functionIndex type) restTasks), values := (project cursor values), bindings := (project cursor bindings), nextParameter := (project cursor nextParameter), activeLeft := (project cursor activeLeft), activeRight := (project cursor activeRight), equality := (project cursor equality), remainingBindings := (project cursor remainingBindings), candidateBinding := (project cursor candidateBinding), arityLeft := (project cursor arityLeft), arityRight := (project cursor arityRight), completedEvents := ((project cursor completedEvents) + 1)})))))
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor mode)
    | Mode.tasks => (tasksStep cursor)
    | Mode.compareEqual => (compareEqualStep cursor)
    | Mode.arity => (arityStep cursor)
    | Mode.scanBindings => (scanBindingsStep cursor)
    | Mode.compareBindingLeft => (compareBindingLeftStep cursor)
    | Mode.compareBindingRight => (compareBindingRightStep cursor))
}

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ remainingFuel => (match state
      | Step.pending cursor => (run remainingFuel (step cursor))
      | Step.complete generalization => (ctor Step.complete generalization)
      | Step.typeMismatch leftType rightType => (ctor Step.typeMismatch leftType rightType)
      | Step.invalidCursor => (ctor Step.invalidCursor)
      | Step.ropeFailure ropeError => (ctor Step.ropeFailure ropeError))),
  recursive_argument: fuel
}

definition{
  name: result,
  parameters: [state : Step],
  result: (Option Generalization),
  body: (match state
    | Step.pending cursor => (ctor Option.none [Generalization])
    | Step.complete generalization => (ctor Option.some [Generalization] generalization)
    | Step.typeMismatch leftType rightType => (ctor Option.none [Generalization])
    | Step.invalidCursor => (ctor Option.none [Generalization])
    | Step.ropeFailure ropeError => (ctor Option.none [Generalization]))
}

definition{
  name: expression,
  parameters: [state : Step],
  result: (Option Uorc.Search.Expression::Expression),
  body: (match (result state)
    | Option.none => (ctor Option.none [Uorc.Search.Expression::Expression])
    | Option.some generalization => (ctor Option.some [Uorc.Search.Expression::Expression] (project generalization expression)))
}

definition{
  name: parameterCount,
  parameters: [state : Step],
  result: (Option Nat),
  body: (match (result state)
    | Option.none => (ctor Option.none [Nat])
    | Option.some generalization => (ctor Option.some [Nat] (prim length (project generalization bindings) : Nat)))
}
