definition{
  name: beginCollect,
  parameters: [target : Bytes, prediction : Bytes],
  result: CollectStep,
  body: (if ((prim length target : Nat) =? (prim length prediction : Nat)) then (ctor CollectStep.pending (record CollectCursor {target := target, prediction := prediction, offset := 0, exceptionsReversed := (nil Exception), bytesCompared := 0})) else (ctor CollectStep.rejected (ctor Error.lengthMismatch)))
}

definition{
  name: collectStep,
  parameters: [cursor : CollectCursor],
  result: CollectStep,
  body: (if ((prim length (project cursor target) : Nat) =? (prim length (project cursor prediction) : Nat)) then (if ((project cursor offset) <? (prim length (project cursor target) : Nat)) then (match (prim index (project cursor target) (project cursor offset) : (Option UInt8))
    | Option.none => (ctor CollectStep.rejected (ctor Error.invalidCursor))
    | Option.some targetByte => (match (prim index (project cursor prediction) (project cursor offset) : (Option UInt8))
      | Option.none => (ctor CollectStep.rejected (ctor Error.invalidCursor))
      | Option.some predictedByte => (if (prim equal targetByte predictedByte : Bool) then (ctor CollectStep.pending (record CollectCursor {target := (project cursor target), prediction := (project cursor prediction), offset := ((project cursor offset) + 1), exceptionsReversed := (project cursor exceptionsReversed), bytesCompared := ((project cursor bytesCompared) + 1)})) else (match (Uorc.Search.Effort::checkedU64 (project cursor offset))
        | Result.error coordinateError => (ctor CollectStep.rejected (ctor Error.coordinateOverflow))
        | Result.ok coordinate => (ctor CollectStep.pending (record CollectCursor {target := (project cursor target), prediction := (project cursor prediction), offset := ((project cursor offset) + 1), exceptionsReversed := (cons (record Exception {coordinate := coordinate, value := targetByte}) (project cursor exceptionsReversed)), bytesCompared := ((project cursor bytesCompared) + 1)})))))) else (if ((project cursor offset) =? (prim length (project cursor target) : Nat)) then (ctor CollectStep.done (project cursor exceptionsReversed) (project cursor bytesCompared)) else (ctor CollectStep.rejected (ctor Error.invalidCursor)))) else (ctor CollectStep.rejected (ctor Error.lengthMismatch))),
  axioms: [Quot.sound, propext]
}

definition{
  name: runCollect,
  parameters: [fuel : Nat, state : CollectStep],
  result: CollectStep,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ restFuel => (match state
      | CollectStep.pending cursor => (runCollect restFuel (collectStep cursor))
      | CollectStep.done exceptions compared => (ctor CollectStep.done exceptions compared)
      | CollectStep.rejected error => (ctor CollectStep.rejected error))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: repairBlock,
  parameters: [],
  result: Uorc.Core.Graph::Block,
  body: (record Uorc.Core.Graph::Block {parameters := (cons (ctor Uorc.Core.Types::ValueType.natural) (cons (ctor Uorc.Core.Types::ValueType.store) (cons (ctor Uorc.Core.Types::ValueType.store) (nil Uorc.Core.Types::ValueType)))), result := (ctor Uorc.Core.Types::ValueType.byte), instructions := (cons (ctor Uorc.Core.Graph::Instruction.byteConstant (0 : UInt8)) (cons (ctor Uorc.Core.Graph::Instruction.storeRead 3 4) (cons (ctor Uorc.Core.Graph::Instruction.byteEqual 1 2) (cons (ctor Uorc.Core.Graph::Instruction.storeRead 4 6) (cons (ctor Uorc.Core.Graph::Instruction.apply 1 (cons 7 (nil Nat))) (cons (ctor Uorc.Core.Graph::Instruction.choose 3 1 2) (nil Uorc.Core.Graph::Instruction))))))), returnDistance := 1})
}

definition{
  name: beginBuild,
  parameters: [generator : Uorc.Core.Graph::Block, count : UInt64, exceptions : (List Exception)],
  result: Finish,
  body: (ctor Finish.building (record Build {generator := generator, count := count, remaining := exceptions, reversedInstructions := (cons (ctor Uorc.Core.Graph::Instruction.storeNew 2) (cons (ctor Uorc.Core.Graph::Instruction.storeNew 1) (cons (ctor Uorc.Core.Graph::Instruction.byteConstant (0 : UInt8)) (cons (ctor Uorc.Core.Graph::Instruction.naturalConstant count) (nil Uorc.Core.Graph::Instruction))))), slots := 4, presenceSlot := 2, valueSlot := 3}))
}

definition{
  name: buildStep,
  parameters: [build : Build],
  result: Finish,
  body: (match (project build remaining)
    | List.nil => (ctor Finish.reversing (project build generator) (cons (ctor Uorc.Core.Graph::Instruction.tabulate (project build slots) 1 (cons (prim subtract (project build slots) (project build presenceSlot) : Nat) (cons (prim subtract (project build slots) (project build valueSlot) : Nat) (nil Nat)))) (project build reversedInstructions)) (nil Uorc.Core.Graph::Instruction))
    | List.cons exception rest => (ctor Finish.building (record Build {generator := (project build generator), count := (project build count), remaining := rest, reversedInstructions := (cons (ctor Uorc.Core.Graph::Instruction.storeWrite (prim subtract ((project build slots) + 4) (project build valueSlot) : Nat) 4 1) (cons (ctor Uorc.Core.Graph::Instruction.byteConstant (project exception value)) (cons (ctor Uorc.Core.Graph::Instruction.storeWrite (prim subtract ((project build slots) + 2) (project build presenceSlot) : Nat) 2 1) (cons (ctor Uorc.Core.Graph::Instruction.byteConstant (1 : UInt8)) (cons (ctor Uorc.Core.Graph::Instruction.naturalConstant (project exception coordinate)) (project build reversedInstructions)))))), slots := ((project build slots) + 5), presenceSlot := ((project build slots) + 2), valueSlot := ((project build slots) + 4)})))
}

definition{
  name: stepFinish,
  parameters: [finish : Finish],
  result: Finish,
  body: (match finish
    | Finish.building build => (buildStep build)
    | Finish.reversing generator remaining forward => (match remaining
      | List.nil => (ctor Finish.done (record Uorc.Core.Graph::Graph {blocks := (cons generator (cons (repairBlock) (cons (record Uorc.Core.Graph::Block {parameters := (nil Uorc.Core.Types::ValueType), result := (ctor Uorc.Core.Types::ValueType.sequence), instructions := forward, returnDistance := 1}) (nil Uorc.Core.Graph::Block)))), entry := 2}))
      | List.cons instruction rest => (ctor Finish.reversing generator rest (cons instruction forward)))
    | Finish.done graph => (ctor Finish.done graph))
}

definition{
  name: runBuild,
  parameters: [fuel : Nat, finish : Finish],
  result: Finish,
  body: (match fuel
    | Nat.zero => finish
    | Nat.succ restFuel => (runBuild restFuel (stepFinish finish))),
  recursive_argument: fuel
}

  parameters: [],

    | Finish.done graph => (Uorc.Core.Graph::validateGraph graph)) = (ctor Result.ok [Unit, Uorc.Core.Types::StaticError] ())),
  proof: reflexivity{}
}
