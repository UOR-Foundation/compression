  name: sameStorage,

  result: Bool,
  body: (match var{name: first}
    | Uorc.Core.CanonicalCost::StorageScope.unconstrained => (match var{name: second}
      | Uorc.Core.CanonicalCost::StorageScope.unconstrained => true
      | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadBytes secondBound => false
      | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadUnits secondBound => false)
    | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadBytes firstBound => (match var{
        name: second
      }
      | Uorc.Core.CanonicalCost::StorageScope.unconstrained => false
      | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadBytes secondBound => (firstBound =? secondBound)
      | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadUnits secondBound => false)
    | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadUnits firstBound => (match var{
        name: second
      }
      | Uorc.Core.CanonicalCost::StorageScope.unconstrained => false
      | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadBytes secondBound => false
      | Uorc.Core.CanonicalCost::StorageScope.liveBindingPayloadUnits secondBound => (firstBound =? secondBound)))
}

definition{
  name: step,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor phase)
    | Phase.header => (if ((canonicalSchema (project (project cursor first) schema)) && (canonicalSchema (project (project cursor second) schema))) then (if (((Uorc.Core.CanonicalCost::profileSupported (project (project cursor first) costProfile)) && (Uorc.Core.CanonicalCost::budgetSupported (project (project cursor first) costProfile) (project (project (project cursor first) query) membership))) && ((Uorc.Core.CanonicalCost::profileSupported (project (project cursor second) costProfile)) && (Uorc.Core.CanonicalCost::budgetSupported (project (project cursor second) costProfile) (project (project (project cursor second) query) membership)))) then (if (Uorc.Core.CanonicalCost::sameProfile (project (project cursor first) costProfile) (project (project cursor second) costProfile)) then (if (sameMembership (project (project (project cursor first) query) membership) (project (project (project cursor second) query) membership)) then (ctor Step.pending (record Cursor {first := (project cursor first), second := (project cursor second), phase := (ctor Phase.inputs (Uorc.Search.Identity::start (inputComparisonView (project (project cursor first) query)) (inputComparisonView (project (project cursor second) query))))})) else (ctor Step.different (ctor Difference.membership))) else (ctor Step.different (ctor Difference.profile))) else (ctor Step.invalidScope)) else (ctor Step.legacyInterpretationRequired))
    | Phase.inputs inputCursor => (match (Uorc.Search.Identity::step inputCursor)
      | Uorc.Search.Identity::Step.pending continuedInput => (ctor Step.pending (record Cursor {first := (project cursor first), second := (project cursor second), phase := (ctor Phase.inputs continuedInput)}))
      | Uorc.Search.Identity::Step.matched => (ctor Step.pending (record Cursor {first := (project cursor first), second := (project cursor second), phase := (ctor Phase.identities (Uorc.Search.VerifyRope::start (identityRoots (project cursor first)) (identityRoots (project cursor second))))}))
      | Uorc.Search.Identity::Step.different fieldDifference => (ctor Step.different (ctor Difference.input fieldDifference))
      | Uorc.Search.Identity::Step.invalidScope => (ctor Step.invalidScope)
      | Uorc.Search.Identity::Step.ropeFailure ropeFailure => (ctor Step.ropeFailure ropeFailure))
    | Phase.identities identityCursor => (match (Uorc.Search.VerifyRope::step identityCursor)
      | Uorc.Search.VerifyRope::Step.pending continuedIdentity => (ctor Step.pending (record Cursor {first := (project cursor first), second := (project cursor second), phase := (ctor Phase.identities continuedIdentity)}))
      | Uorc.Search.VerifyRope::Step.matched byteCount objectCount => (ctor Step.matched)
      | Uorc.Search.VerifyRope::Step.mismatch identityDifference => (ctor Step.different (ctor Difference.identities identityDifference))
      | Uorc.Search.VerifyRope::Step.ropeFailure ropeFailure => (ctor Step.ropeFailure ropeFailure)))
}

definition{
  name: run,
  parameters: [work : Nat, state : Step],
  result: Step,
  body: (match work
    | Nat.zero => state
    | Nat.succ remainingWork => (match state
      | Step.pending nextCursor => (run remainingWork (step nextCursor))
      | Step.matched => state
      | Step.different difference => state
      | Step.legacyInterpretationRequired => state
      | Step.invalidScope => state
      | Step.ropeFailure ropeFailure => state)),
  recursive_argument: work
}

    | Envelope.owned firstBudget => (match var{name: second}
      | Envelope.referenceDiagnostic secondBudget => false
      | Envelope.owned secondBudget => (((prim equal (project firstBudget maxWork) (project secondBudget maxWork) : Bool) && (prim equal (project firstBudget maxOwnedBytes) (project secondBudget maxOwnedBytes) : Bool)) && ((prim equal (project firstBudget maxOutputBytes) (project secondBudget maxOutputBytes) : Bool) && (prim equal (project firstBudget maxFrames) (project secondBudget maxFrames) : Bool))))
    | Envelope.referenceDiagnostic firstBudget => (match var{name: second}
      | Envelope.owned secondBudget => false
      | Envelope.referenceDiagnostic secondBudget => ((((project firstBudget maxSteps) =? (project secondBudget maxSteps)) && ((project firstBudget maxLiveCells) =? (project secondBudget maxLiveCells))) && (((project firstBudget maxOutputBytes) =? (project secondBudget maxOutputBytes)) && ((project firstBudget maxFrames) =? (project secondBudget maxFrames))))))
