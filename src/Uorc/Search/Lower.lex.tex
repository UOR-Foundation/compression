  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor parameterTail)
    | List.nil => (match (project cursor tasks)
      | List.nil => (match (project cursor values)
        | List.nil => (ctor Step.failed (ctor Error.invalidCursor))

            | Uorc.Search.Expression::Literal.predicate payload => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := restTasks, values := (cons (project cursor nextSlot) (project cursor values)), instructionsReversed := (cons (ctor Uorc.Core.Graph::Instruction.predicateConstant payload) (project cursor instructionsReversed)), nextSlot := ((project cursor nextSlot) + 1), completedEvents := ((project cursor completedEvents) + 1)}))
            | Uorc.Search.Expression::Literal.sequence payload => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := restTasks, values := (cons (project cursor nextSlot) (project cursor values)), instructionsReversed := (cons (ctor Uorc.Core.Graph::Instruction.sequenceRopeConstant payload) (project cursor instructionsReversed)), nextSlot := ((project cursor nextSlot) + 1), completedEvents := ((project cursor completedEvents) + 1)})))
          | Uorc.Search.Expression::Expression.parameter index type => (if (index <? (project cursor parameterCount)) then (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := restTasks, values := (cons index (project cursor values)), instructionsReversed := (project cursor instructionsReversed), nextSlot := (project cursor nextSlot), completedEvents := ((project cursor completedEvents) + 1)})) else (ctor Step.failed (ctor Error.invalidParameter)))
          | Uorc.Search.Expression::Expression.call functionIndex type arguments => (if (functionIndex <? (project cursor libraryCount)) then (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := (cons (ctor Task.arguments arguments (nil Nat) functionIndex) restTasks), values := (project cursor values), instructionsReversed := (project cursor instructionsReversed), nextSlot := (project cursor nextSlot), completedEvents := ((project cursor completedEvents) + 1)})) else (ctor Step.failed (ctor Error.invalidFunction))))

          | List.nil => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := (cons (ctor Task.references reverseAbsolute (nil Nat) functionIndex) restTasks), values := (project cursor values), instructionsReversed := (project cursor instructionsReversed), nextSlot := (project cursor nextSlot), completedEvents := ((project cursor completedEvents) + 1)}))
          | List.cons argument remainingArguments => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := (cons (ctor Task.lower argument) (cons (ctor Task.collect remainingArguments reverseAbsolute functionIndex) restTasks)), values := (project cursor values), instructionsReversed := (project cursor instructionsReversed), nextSlot := (project cursor nextSlot), completedEvents := ((project cursor completedEvents) + 1)})))
        | Task.collect remainingExpressions reverseAbsolute functionIndex => (match (project cursor values)
          | List.nil => (ctor Step.failed (ctor Error.invalidCursor))
          | List.cons absolute remainingValues => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := (cons (ctor Task.arguments remainingExpressions (cons absolute reverseAbsolute) functionIndex) restTasks), values := remainingValues, instructionsReversed := (project cursor instructionsReversed), nextSlot := (project cursor nextSlot), completedEvents := ((project cursor completedEvents) + 1)})))

          | List.nil => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := restTasks, values := (cons (project cursor nextSlot) (project cursor values)), instructionsReversed := (cons (ctor Uorc.Core.Graph::Instruction.apply (prim subtract (project cursor libraryCount) functionIndex : Nat) forwardReferences) (project cursor instructionsReversed)), nextSlot := ((project cursor nextSlot) + 1), completedEvents := ((project cursor completedEvents) + 1)}))
          | List.cons reference remainingReferencesTail => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := (cons (ctor Task.reverseReferences remainingReferencesTail (cons reference forwardReferences) functionIndex) restTasks), values := (project cursor values), instructionsReversed := (project cursor instructionsReversed), nextSlot := (project cursor nextSlot), completedEvents := ((project cursor completedEvents) + 1)})))
        | Task.reverseInstructions remainingInstructions forwardInstructions absoluteReturn => (match remainingInstructions
          | List.nil => (ctor Step.complete (record Uorc.Core.Graph::Block {parameters := (project cursor parameters), result := (project cursor resultType), instructions := forwardInstructions, returnDistance := (prim subtract (project cursor nextSlot) absoluteReturn : Nat)}) ((project cursor completedEvents) + 1))
          | List.cons instruction remainingInstructionsTail => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := (project cursor parameterTail), parameterCount := (project cursor parameterCount), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := (cons (ctor Task.reverseInstructions remainingInstructionsTail (cons instruction forwardInstructions) absoluteReturn) restTasks), values := (project cursor values), instructionsReversed := (project cursor instructionsReversed), nextSlot := (project cursor nextSlot), completedEvents := ((project cursor completedEvents) + 1)})))))
    | List.cons parameterType remainingParameterTypes => (ctor Step.pending (record Cursor {parameters := (project cursor parameters), parameterTail := remainingParameterTypes, parameterCount := ((project cursor parameterCount) + 1), libraryCount := (project cursor libraryCount), resultType := (project cursor resultType), tasks := (project cursor tasks), values := (project cursor values), instructionsReversed := (project cursor instructionsReversed), nextSlot := ((project cursor nextSlot) + 1), completedEvents := ((project cursor completedEvents) + 1)})))

definition{
  name: run,
  parameters: [fuel : Nat, state : Step],
  result: Step,
  body: (match fuel
    | Nat.zero => state
    | Nat.succ remainingFuel => (match state
      | Step.pending cursor => (run remainingFuel (step cursor))
      | Step.complete blockValue events => (ctor Step.complete blockValue events)
      | Step.failed error => (ctor Step.failed error))),
  recursive_argument: fuel
}

definition{
  name: block,
  parameters: [state : Step],
  result: (Option Uorc.Core.Graph::Block),
  body: (match state
    | Step.pending cursor => (ctor Option.none [Uorc.Core.Graph::Block])
    | Step.complete completedBlock events => (ctor Option.some [Uorc.Core.Graph::Block] completedBlock)
    | Step.failed error => (ctor Option.none [Uorc.Core.Graph::Block]))
}
