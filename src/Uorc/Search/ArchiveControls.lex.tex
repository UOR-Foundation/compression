definition{
  name: runCursor,
  parameters: [work : Nat, cursor : Uorc.Search.ArchiveTrial::Cursor],
  result: Uorc.Search.ArchiveTrial::Step,
  body: (match work
    | Nat.zero => (ctor Uorc.Search.ArchiveTrial::Step.pending cursor)
    | Nat.succ remainingWork => (match (Uorc.Search.ArchiveTrial::step cursor)
      | Uorc.Search.ArchiveTrial::Step.pending next => (runCursor remainingWork next)
      | Uorc.Search.ArchiveTrial::Step.complete eligible => (ctor Uorc.Search.ArchiveTrial::Step.complete eligible)
      | Uorc.Search.ArchiveTrial::Step.rejected failure usage ledger events => (ctor Uorc.Search.ArchiveTrial::Step.rejected failure usage ledger events))),
  recursive_argument: work
}

definition{
  name: run,
  parameters: [work : Nat, state : Uorc.Search.ArchiveTrial::Step],
  result: Uorc.Search.ArchiveTrial::Step,
  body: (match work
    | Nat.zero => state
    | Nat.succ remainingWork => (match state
      | Uorc.Search.ArchiveTrial::Step.pending cursor => (runCursor (ctor Nat.succ remainingWork) cursor)
      | Uorc.Search.ArchiveTrial::Step.complete eligible => (ctor Uorc.Search.ArchiveTrial::Step.complete eligible)
      | Uorc.Search.ArchiveTrial::Step.rejected failure usage ledger events => (ctor Uorc.Search.ArchiveTrial::Step.rejected failure usage ledger events)))
}

definition{
  name: size,
  parameters: [state : Uorc.Search.ArchiveTrial::Step],
  result: (Option Nat),
  body: (match state
    | Uorc.Search.ArchiveTrial::Step.pending cursor => (ctor Option.none [Nat])
    | Uorc.Search.ArchiveTrial::Step.complete eligible => (ctor Option.some [Nat] (Uorc.Core.Bytes::length (project (project eligible checked) wire)))
    | Uorc.Search.ArchiveTrial::Step.rejected failure usage ledger events => (ctor Option.none [Nat]))
}
