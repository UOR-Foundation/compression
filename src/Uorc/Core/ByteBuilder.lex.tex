      | List.cons headBin restBins => (ctor Event.failed (ctor Error.invalidState))) else (if ((0 <? (prim length carry : Nat)) && ((prim length carry : Nat) <? capacity)) then (match remainingBins
      | List.nil => (ctor Event.advanced (ctor Work.rebuilding completed (cons (ctor Option.some [Bytes] carry) (nil (Option Bytes))) zeros count capacity))
