definition{
  name: eventWork,
  parameters: [workValue : Work],
  result: Nat,
  body: (match workValue
    | Work.beginByte builder value => 9
    | Work.merging completed carry remainingBins zeros count capacity => (match remainingBins
      | List.nil => 1
      | List.cons headBin restBins => (match headBin
        | Option.none => 1
        | Option.some oldBlock => (if (((prim length oldBlock : Nat) <=? 4096) && ((prim length carry : Nat) <=? 4096)) then (1 + ((prim length oldBlock : Nat) + (prim length carry : Nat))) else 1)))
    | Work.rebuilding completed bins zeros count capacity => 1
    | Work.flushing ropeWork capacity => 1
    | Work.tail completed remainingBins tailBytes => (match remainingBins
      | List.nil => 1
      | List.cons headBin restBins => (match headBin
        | Option.none => 1
        | Option.some oldBlock => (if ((prim length tailBytes : Nat) =? 0) then 1 else (if (((prim length oldBlock : Nat) <=? 4096) && ((prim length tailBytes : Nat) <=? 4096)) then (1 + ((prim length oldBlock : Nat) + (prim length tailBytes : Nat))) else 1))))
    | Work.tailFlushing ropeWork => 1),
  executable: true
}

definition{
  name: step,
  parameters: [workValue : Work],
  result: Event,
  body: (match workValue
    | Work.beginByte builder value => (if ((project builder buffered) <? (project builder capacity)) then (match (byteSlice value)
      | Option.none => (ctor Event.failed (ctor Error.invalidState))
      | Option.some singleByte => (ctor Event.advanced (ctor Work.merging (project builder completed) singleByte (project builder bins) 0 ((project builder buffered) + 1) (project builder capacity)))) else (ctor Event.failed (ctor Error.invalidState)))
    | Work.merging completed carry remainingBins zeros count capacity => (if ((0 <? capacity) && (capacity <=? 4096)) then (if ((prim length carry : Nat) =? capacity) then (match remainingBins
      | List.nil => (if (count =? capacity) then (if (Uorc.Core.Resources::fitsAddition (Uorc.Wire.Domains::wordLast) (Uorc.Core.Bytes::length completed) (prim length carry : Nat)) then (ctor Event.advanced (ctor Work.flushing (Uorc.Core.Bytes::startJoin completed (Uorc.Core.Bytes::fromBytes carry)) capacity)) else (ctor Event.failed (ctor Error.addressSpaceExhausted))) else (ctor Event.failed (ctor Error.invalidState)))
      | List.cons headBin restBins => (ctor Event.failed (ctor Error.invalidState))) else (if ((0 <? (prim length carry : Nat)) && ((prim length carry : Nat) <? capacity)) then (match remainingBins
      | List.nil => (ctor Event.advanced (ctor Work.rebuilding completed (cons (ctor Option.some [Bytes] carry) (nil (Option Bytes))) zeros count capacity))
      | List.cons headBin restBins => (match headBin
        | Option.none => (ctor Event.advanced (ctor Work.rebuilding completed (cons (ctor Option.some [Bytes] carry) restBins) zeros count capacity))
        | Option.some oldBlock => (if (((prim length oldBlock : Nat) =? (prim length carry : Nat)) && ((prim length carry : Nat) <=? (prim quotient capacity (Uorc.Wire.Domains::radix) 0 : Nat))) then (ctor Event.advanced (ctor Work.merging completed (prim append oldBlock carry : Bytes) restBins (zeros + 1) count capacity)) else (ctor Event.failed (ctor Error.invalidState))))) else (ctor Event.failed (ctor Error.invalidState)))) else (ctor Event.failed (ctor Error.invalidState)))
    | Work.rebuilding completed bins zeros count capacity => (match zeros
      | Nat.zero => (ctor Event.appended (record Builder {completed := completed, bins := bins, buffered := count, capacity := capacity}))
      | Nat.succ remainingZeros => (ctor Event.advanced (ctor Work.rebuilding completed (cons (ctor Option.none [Bytes]) bins) remainingZeros count capacity)))
    | Work.flushing ropeWork capacity => (match (Uorc.Core.Bytes::workStep ropeWork)
      | Uorc.Core.Bytes::WorkEvent.advanced nextWork => (ctor Event.advanced (ctor Work.flushing nextWork capacity))
      | Uorc.Core.Bytes::WorkEvent.sequenceDone completedRope => (ctor Event.appended (record Builder {completed := completedRope, bins := (nil (Option Bytes)), buffered := 0, capacity := capacity}))
      | Uorc.Core.Bytes::WorkEvent.byteDone resultByte => (ctor Event.failed (ctor Error.invalidState))
      | Uorc.Core.Bytes::WorkEvent.failed ropeError => (ctor Event.failed (ctor Error.rope ropeError)))
    | Work.tail completed remainingBins tailBytes => (match remainingBins
      | List.nil => (if ((prim length tailBytes : Nat) =? 0) then (ctor Event.finished completed) else (if (Uorc.Core.Resources::fitsAddition (Uorc.Wire.Domains::wordLast) (Uorc.Core.Bytes::length completed) (prim length tailBytes : Nat)) then (ctor Event.advanced (ctor Work.tailFlushing (Uorc.Core.Bytes::startJoin completed (Uorc.Core.Bytes::fromBytes tailBytes)))) else (ctor Event.failed (ctor Error.addressSpaceExhausted))))
      | List.cons headBin restBins => (match headBin
        | Option.none => (ctor Event.advanced (ctor Work.tail completed restBins tailBytes))
        | Option.some oldBlock => (if (((prim length oldBlock : Nat) <=? 4096) && ((prim length tailBytes : Nat) <=? (prim subtract 4096 (prim length oldBlock : Nat) : Nat))) then (ctor Event.advanced (ctor Work.tail completed restBins (if ((prim length tailBytes : Nat) =? 0) then oldBlock else (prim append oldBlock tailBytes : Bytes)))) else (ctor Event.failed (ctor Error.invalidState)))))
    | Work.tailFlushing ropeWork => (match (Uorc.Core.Bytes::workStep ropeWork)
      | Uorc.Core.Bytes::WorkEvent.advanced nextWork => (ctor Event.advanced (ctor Work.tailFlushing nextWork))
      | Uorc.Core.Bytes::WorkEvent.sequenceDone completedRope => (ctor Event.finished completedRope)
      | Uorc.Core.Bytes::WorkEvent.byteDone resultByte => (ctor Event.failed (ctor Error.invalidState))
      | Uorc.Core.Bytes::WorkEvent.failed ropeError => (ctor Event.failed (ctor Error.rope ropeError)))),
  axioms: []
}

definition{
  name: runEvents,
  parameters: [fuel : Nat, workValue : Work],
  result: Event,
  body: (match fuel
    | Nat.zero => (ctor Event.advanced workValue)
    | Nat.succ remainingFuel => (match (step workValue)
      | Event.advanced nextWork => (runEvents remainingFuel nextWork)
      | Event.appended builder => (ctor Event.appended builder)
      | Event.finished rope => (ctor Event.finished rope)
      | Event.failed error => (ctor Event.failed error))),
  axioms: [],
  recursive_argument: fuel
}
