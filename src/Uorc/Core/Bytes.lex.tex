definition{
  name: advancePending,
  parameters: [pending : (List Rope)],
  result: CursorEvent,
  body: (match pending
    | List.nil => (ctor CursorEvent.done)
    | List.cons headRope restRopes => (match headRope
      | Rope.empty => (ctor CursorEvent.advanced (record Cursor {pending := restRopes, current := (ctor Option.none [Bytes]), offset := 0, end := 0}))
      | Rope.leaf chunk => (ctor CursorEvent.advanced (record Cursor {pending := restRopes, current := (ctor Option.some [Bytes] chunk), offset := 0, end := (prim length chunk : Nat)}))
      | Rope.branch cachedLength cachedHeight leftTree rightTree => (ctor CursorEvent.advanced (record Cursor {pending := (cons leftTree (cons rightTree restRopes)), current := (ctor Option.none [Bytes]), offset := 0, end := 0}))
      | Rope.view backing viewOffset viewLength => (ctor CursorEvent.advanced (record Cursor {pending := restRopes, current := (ctor Option.some [Bytes] backing), offset := viewOffset, end := (viewOffset + viewLength)}))))
}

}

  body: (match (project workValue task)
    | WorkTask.joining leftRope rightRope => (if ((length leftRope) =? 0) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning rightRope), continuations := (project workValue continuations)})) else (if ((length rightRope) =? 0) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning leftRope), continuations := (project workValue continuations)})) else (if (((height rightRope) + 1) <? (height leftRope)) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.joining (right leftRope) rightRope), continuations := (cons (ctor WorkContinuation.balanceRight (left leftRope)) (project workValue continuations))})) else (if (((height leftRope) + 1) <? (height rightRope)) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.joining leftRope (left rightRope)), continuations := (cons (ctor WorkContinuation.balanceLeft (right rightRope)) (project workValue continuations))})) else (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning (node leftRope rightRope)), continuations := (project workValue continuations)}))))))
    | WorkTask.slicing sourceRope start count => (if (count =? 0) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning (ctor Rope.empty)), continuations := (project workValue continuations)})) else (if ((start =? 0) && (count =? (length sourceRope))) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning sourceRope), continuations := (project workValue continuations)})) else (match sourceRope
      | Rope.empty => (ctor WorkEvent.failed (ctor RopeError.invalidCursor))
      | Rope.leaf chunk => (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning (ctor Rope.view chunk start count)), continuations := (project workValue continuations)}))
      | Rope.branch cachedLength cachedHeight leftNode rightNode => (if (start <? (length leftNode)) then (if (count <=? (prim subtract (length leftNode) start : Nat)) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.slicing leftNode start count), continuations := (project workValue continuations)})) else (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.slicing leftNode start (prim subtract (length leftNode) start : Nat)), continuations := (cons (ctor WorkContinuation.sliceSecond rightNode (prim subtract count (prim subtract (length leftNode) start : Nat) : Nat)) (project workValue continuations))}))) else (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.slicing rightNode (prim subtract start (length leftNode) : Nat) count), continuations := (project workValue continuations)})))
      | Rope.view backing viewOffset viewLength => (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning (ctor Rope.view backing (viewOffset + start) count)), continuations := (project workValue continuations)})))))
    | WorkTask.indexing sourceRope index => (match sourceRope
      | Rope.empty => (ctor WorkEvent.failed (ctor RopeError.invalidCursor))
      | Rope.leaf chunk => (match (readLeafByte chunk index)
        | LeafByteRead.missing => (ctor WorkEvent.failed (ctor RopeError.invalidCursor))
        | LeafByteRead.found selectedByte => (ctor WorkEvent.byteDone selectedByte))
      | Rope.branch cachedLength cachedHeight leftNode rightNode => (if (index <? (length leftNode)) then (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.indexing leftNode index), continuations := (project workValue continuations)})) else (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.indexing rightNode (prim subtract index (length leftNode) : Nat)), continuations := (project workValue continuations)})))
      | Rope.view backing viewOffset viewLength => (match (prim index backing (viewOffset + index) : (Option UInt8))
        | Option.none => (ctor WorkEvent.failed (ctor RopeError.invalidCursor))
        | Option.some selectedByte => (ctor WorkEvent.byteDone selectedByte)))
    | WorkTask.returning returnedRope => (match (project workValue continuations)
      | List.nil => (ctor WorkEvent.sequenceDone returnedRope)
      | List.cons continuation restContinuations => (match continuation
        | WorkContinuation.balanceRight leftRope => (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning (balance leftRope returnedRope)), continuations := restContinuations}))
        | WorkContinuation.balanceLeft rightRope => (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.returning (balance returnedRope rightRope)), continuations := restContinuations}))
        | WorkContinuation.sliceSecond rightRope rightCount => (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.slicing rightRope 0 rightCount), continuations := (cons (ctor WorkContinuation.joinFirst returnedRope) restContinuations)}))
        | WorkContinuation.joinFirst leftRope => (ctor WorkEvent.advanced (record Work {task := (ctor WorkTask.joining leftRope returnedRope), continuations := restContinuations})))))
}

definition{
  name: runWorkEvents,
  parameters: [fuel : Nat, workValue : Work],
  result: WorkEvent,
  body: (match fuel
    | Nat.zero => (ctor WorkEvent.advanced workValue)
    | Nat.succ remaining => (match (workStep workValue)
      | WorkEvent.advanced nextWork => (runWorkEvents remaining nextWork)
      | WorkEvent.sequenceDone resultRope => (ctor WorkEvent.sequenceDone resultRope)
      | WorkEvent.byteDone resultByte => (ctor WorkEvent.byteDone resultByte)
      | WorkEvent.failed error => (ctor WorkEvent.failed error))),
  recursive_argument: fuel
}
