definition{
  name: validNode,
  parameters: [arena : Arena, node : Node],
  result: Bool,
  body: (match node
    | Node.image k bs => (if (isControlKind k) then false else true)
    | Node.sequenceLeaf image offset length => (if (Uorc.Core.RuntimeValue::isNull image) then false else (match (lookup arena image)
      | Result.error e => false
      | Result.ok entry => (match (project entry node)
        | Node.image k bytes => (match (wordOfNat (prim length bytes : Nat))
          | Option.none => false
          | Option.some size => (if (if (prim equal length (0 : UInt64) : Bool) then false else true) then (if (if (less size offset) then false else true) then (if (less (wordSub size offset) length) then false else true) else false) else false))
        | Node.sequenceLeaf i o n => false
        | Node.sequenceBranch n h a b => false
        | Node.storeLeaf b => false
        | Node.storeBranch a b => false
        | Node.slot r v => false
        | Node.rootBranch r a b => false
        | Node.metadataWord meta0 meta1 meta2 => false
        | Node.metadataLink meta0 meta1 meta2 => false
        | Node.metadataBranch meta0 meta1 meta2 => false
        | Node.ownerRoot meta0 meta1 => false
        | Node.imageReservation meta0 meta1 => false)))
    | Node.sequenceBranch length height a b => (match (sequenceInfo arena a)
      | Result.error e => false
      | Result.ok leftInfo => (match (sequenceInfo arena b)
        | Result.error e => false
        | Result.ok rightInfo => (match (prim checked_add (project leftInfo length) (project rightInfo length) : (Option UInt64))
          | Option.none => false
          | Option.some size => (let maxHeight : UInt64 := (if (less (project leftInfo height) (project rightInfo height)) then (project rightInfo height) else (project leftInfo height)); (match (prim checked_add maxHeight (1 : UInt64) : (Option UInt64))
            | Option.none => false
            | Option.some h => (if (if (if (Uorc.Core.RuntimeValue::isNull a) then false else true) then (if (Uorc.Core.RuntimeValue::isNull b) then false else true) else false) then (if (prim equal length size : Bool) then (prim equal height h : Bool) else false) else false))))))
    | Node.storeLeaf b => true
    | Node.storeBranch a b => (if (if (if (Uorc.Core.RuntimeValue::isNull a) then false else true) then true else (if (Uorc.Core.RuntimeValue::isNull b) then false else true)) then (if (validStoreHandle arena a) then (validStoreHandle arena b) else false) else false)
    | Node.slot r v => (validValue arena v)
    | Node.rootBranch r a b => (if (if (if (Uorc.Core.RuntimeValue::isNull a) then false else true) then true else (if (Uorc.Core.RuntimeValue::isNull b) then false else true)) then (if (rootNode arena a) then (rootNode arena b) else false) else false)
    | Node.metadataWord meta0 meta1 meta2 => true
    | Node.metadataLink meta0 meta1 meta2 => true
    | Node.metadataBranch meta0 meta1 meta2 => true
    | Node.ownerRoot meta0 meta1 => true
    | Node.imageReservation meta0 meta1 => true),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: retainEntry,
  parameters: [arena : Arena, handle : Uorc.Core.RuntimeValue::Handle, external : Bool],
  result: (Result Arena HeapError),
  body: (if (Uorc.Core.RuntimeValue::isNull handle) then (ctor Result.ok [Arena, HeapError] arena) else (match (lookup arena handle)
    | Result.error error => (ctor Result.error [Arena, HeapError] error)
    | Result.ok entry => (if (isReservation (project entry node)) then (ctor Result.error [Arena, HeapError] (ctor HeapError.protocolFailure)) else (match (prim checked_add (project entry incoming) (1 : UInt64) : (Option UInt64))
      | Option.none => (ctor Result.error [Arena, HeapError] (ctor HeapError.countOverflow))

definition{
  name: commitImage,
  parameters: [arena : Arena, reservation : Uorc.Core.RuntimeValue::Handle, bytes : Bytes],
  result: (Result Allocation HeapError),
  body: (match (wordOfNat (prim length bytes : Nat))
    | Option.none => (ctor Result.error [Allocation, HeapError] (ctor HeapError.ownedBytesOverflow))

      | Result.error error => (ctor Result.error [Allocation, HeapError] error)
      | Result.ok charged => (match (lookup arena reservation)
        | Result.error error => (ctor Result.error [Allocation, HeapError] error)
