definition{
  name: prependLive,
  parameters: [
    selected : (Option Uorc.Core.CanonicalBindings::Binding),
    rest : (List Uorc.Core.CanonicalBindings::Binding)
  ],
  result: (List Uorc.Core.CanonicalBindings::Binding),
  body: (match selected | Option.none => rest | Option.some binding => (cons binding rest)),
  axioms: [],
  executable: false
}

definition{
  name: remainingFrom,
  parameters: [
    count : Nat,
    slot : Nat,
    returnedSlot : Nat,
    liveAt : (Fn (Nat) (Option Uorc.Core.CanonicalBindings::Binding))
  ],
  result: (List Uorc.Core.CanonicalBindings::Binding),
  body: (match count
    | Nat.zero => (nil Uorc.Core.CanonicalBindings::Binding)
    | Nat.succ countLeft => (if (slot =? returnedSlot) then (remainingFrom countLeft (slot + 1) returnedSlot liveAt) else (prependLive (apply liveAt slot) (remainingFrom countLeft (slot + 1) returnedSlot liveAt)))),
  axioms: [],
  executable: false,
  recursive_argument: count
}

definition{
  name: remaining,
  parameters: [
    count : Nat,
    returnedSlot : Nat,
    liveAt : (Fn (Nat) (Option Uorc.Core.CanonicalBindings::Binding))

  body: (remainingFrom count 0 returnedSlot liveAt),
  axioms: [],
  executable: false
}

definition{

definition{
  name: frameReturnPlan,
  parameters: [

  executable: false

definition{
  name: returnCoherent,
  parameters: [

  body: ((returnedSlot < count) /\ ((apply liveAt returnedSlot) = (ctor Option.some [Uorc.Core.CanonicalBindings::Binding] (record Uorc.Core.CanonicalBindings::Binding {role := sourceRole, value := (project (project plan binding) value)})))),
  axioms: [],
  executable: false
}
