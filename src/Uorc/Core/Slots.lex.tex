definition{
  name: readTrie,
  parameters: [depth : Nat, tree : SlotTrie, key : UInt64],
  result: (Option Uorc.Core.Types::Value),
  body: (match depth
    | Nat.zero => (match tree
      | SlotTrie.empty => (ctor Option.none [Uorc.Core.Types::Value])
      | SlotTrie.leaf leafValue => leafValue
      | SlotTrie.branch leftTree rightTree => (ctor Option.none [Uorc.Core.Types::Value]))
    | Nat.succ remaining => (match tree
      | SlotTrie.empty => (ctor Option.none [Uorc.Core.Types::Value])
      | SlotTrie.leaf leafValue => (ctor Option.none [Uorc.Core.Types::Value])
      | SlotTrie.branch leftTree rightTree => (readTrie remaining (if (Uorc.Core.Store::lowBit key) then rightTree else leftTree) (Uorc.Core.Store::nextKey key)))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: depth
}

definition{
  name: writeTrie,
  parameters: [depth : Nat, tree : SlotTrie, key : UInt64, value : (Option Uorc.Core.Types::Value)],
  result: SlotTrie,
  body: (match depth
    | Nat.zero => (ctor SlotTrie.leaf value)
    | Nat.succ remaining => (if (Uorc.Core.Store::lowBit key) then (ctor SlotTrie.branch (left tree) (writeTrie remaining (right tree) (Uorc.Core.Store::nextKey key) value)) else (ctor SlotTrie.branch (writeTrie remaining (left tree) (Uorc.Core.Store::nextKey key) value) (right tree)))),
  axioms: [Quot.sound, propext],
  recursive_argument: depth
}

definition{
  name: appendValues,
  parameters: [values : (List Uorc.Core.Types::Value), table : SlotTable],
  result: (Result SlotTable SlotError),
  body: (match values
    | List.nil => (ctor Result.ok [SlotTable, SlotError] table)
    | List.cons headValue restValues => (match (appendSlot table headValue)
      | Result.error error => (ctor Result.error [SlotTable, SlotError] error)
      | Result.ok nextTable => (appendValues restValues nextTable))),
  axioms: [Quot.sound, propext],
  recursive_argument: values
}

definition{
  name: atIndex,
  parameters: [table : SlotTable, index : Nat],
  result: (Result Uorc.Core.Types::Value SlotError),
  body: (if (index <? (project table length)) then (match (Uorc.Core.Types::wordOfNat index)
    | Option.none => (ctor Result.error [Uorc.Core.Types::Value, SlotError] (ctor SlotError.addressSpaceExhausted))
    | Option.some key => (match (readTrie 64 (project table root) key)
      | Option.none => (ctor Result.error [Uorc.Core.Types::Value, SlotError] (ctor SlotError.releasedCell))
      | Option.some selected => (ctor Result.ok [Uorc.Core.Types::Value, SlotError] selected))) else (ctor Result.error [Uorc.Core.Types::Value, SlotError] (ctor SlotError.invalidReference))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: byDistance,
  parameters: [table : SlotTable, distance : Nat],
  result: (Result Uorc.Core.Types::Value SlotError),
  body: (if (distance =? 0) then (ctor Result.error [Uorc.Core.Types::Value, SlotError] (ctor SlotError.invalidReference)) else (if ((project table length) <? distance) then (ctor Result.error [Uorc.Core.Types::Value, SlotError] (ctor SlotError.invalidReference)) else (atIndex table (prim subtract (project table length) distance : Nat)))),
  axioms: [Quot.sound, propext],
  executable: true
}
