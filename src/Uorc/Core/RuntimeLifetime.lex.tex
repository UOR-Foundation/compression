definition{
  name: readUse,
  parameters: [
    arena : Uorc.Core.Heap::Arena,
    lastUses : Uorc.Core.RuntimeTable::Table,
    slot : UInt64
  ],
  result: (Result (Option UInt64) Uorc.Core.RuntimeProgram::Failure),
  body: (match (Uorc.Core.RuntimeTable::lookup arena lastUses slot)
    | Result.error heapError => (ctor Result.error [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.heap heapError))
    | Result.ok leaf => (if (Uorc.Core.RuntimeValue::isNull leaf) then (ctor Result.ok [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] (ctor Option.none [UInt64])) else (match (Uorc.Core.RuntimeProgram::valueAtLeaf arena leaf (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.invalidLocalReference)))
      | Result.error leafError => (ctor Result.error [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] leafError)
      | Result.ok value => (match value
        | Uorc.Core.RuntimeValue::RuntimeValue.natural stamp => (ctor Result.ok [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] (ctor Option.some [UInt64] stamp))
        | Uorc.Core.RuntimeValue::RuntimeValue.byte byteValue => (ctor Result.error [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.typeMismatch)))
        | Uorc.Core.RuntimeValue::RuntimeValue.predicate predicateValue => (ctor Result.error [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.typeMismatch)))
        | Uorc.Core.RuntimeValue::RuntimeValue.sequence sequenceHandle => (ctor Result.error [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.typeMismatch)))
        | Uorc.Core.RuntimeValue::RuntimeValue.store defaultByte storeHandle => (ctor Result.error [(Option UInt64), Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.typeMismatch))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: start,
  parameters: [parameterCount : UInt64, instructionCount : UInt64, returnSlot : UInt64],
  result: (Result Cursor Uorc.Core.RuntimeProgram::Failure),
  body: (match (prim checked_add parameterCount instructionCount : (Option UInt64))
    | Option.none => (ctor Result.error [Cursor, Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.operational (ctor Uorc.Core.Resources::OperationalError.addressSpaceExhausted)))
    | Option.some slotCount => (if (Uorc.Core.Heap::less returnSlot slotCount) then (ctor Result.ok [Cursor, Uorc.Core.RuntimeProgram::Failure] (record Cursor {parameterCount := parameterCount, instructionCount := instructionCount, slotCount := slotCount, returnSlot := returnSlot, nextSlot := (0 : UInt64)})) else (ctor Result.error [Cursor, Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.invalidLocalReference))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: step,
  parameters: [
    arena : Uorc.Core.Heap::Arena,
    lastUses : Uorc.Core.RuntimeTable::Table,
    cursor : Cursor
  ],
  result: (Result Event Uorc.Core.RuntimeProgram::Failure),
  body: (if (Uorc.Core.Heap::less (project cursor nextSlot) (project cursor slotCount)) then (match (prim checked_add (project cursor nextSlot) (1 : UInt64) : (Option UInt64))
    | Option.none => (ctor Result.error [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.operational (ctor Uorc.Core.Resources::OperationalError.addressSpaceExhausted)))
    | Option.some following => (if (prim equal (project cursor nextSlot) (project cursor returnSlot) : Bool) then (ctor Result.ok [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Event.skipped (record Cursor {parameterCount := (project cursor parameterCount), instructionCount := (project cursor instructionCount), slotCount := (project cursor slotCount), returnSlot := (project cursor returnSlot), nextSlot := following}))) else (match (readUse arena lastUses (project cursor nextSlot))
      | Result.error readError => (ctor Result.error [Event, Uorc.Core.RuntimeProgram::Failure] readError)
      | Result.ok timestamp => (match timestamp
        | Option.none => (if (Uorc.Core.Heap::less (project cursor nextSlot) (project cursor parameterCount)) then (ctor Result.ok [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Event.assigned (record Cursor {parameterCount := (project cursor parameterCount), instructionCount := (project cursor instructionCount), slotCount := (project cursor slotCount), returnSlot := (project cursor returnSlot), nextSlot := following}) (ctor Bucket.initial) (project cursor nextSlot))) else (match (prim checked_subtract (project cursor nextSlot) (project cursor parameterCount) : (Option UInt64))
          | Option.none => (ctor Result.error [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.invalidLocalReference)))
          | Option.some creationIndex => (ctor Result.ok [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Event.assigned (record Cursor {parameterCount := (project cursor parameterCount), instructionCount := (project cursor instructionCount), slotCount := (project cursor slotCount), returnSlot := (project cursor returnSlot), nextSlot := following}) (ctor Bucket.after creationIndex) (project cursor nextSlot)))))
        | Option.some stamp => (if (Uorc.Core.Heap::less stamp (project cursor instructionCount)) then (if (Uorc.Core.Heap::less (project cursor nextSlot) (project cursor parameterCount)) then (ctor Result.ok [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Event.assigned (record Cursor {parameterCount := (project cursor parameterCount), instructionCount := (project cursor instructionCount), slotCount := (project cursor slotCount), returnSlot := (project cursor returnSlot), nextSlot := following}) (ctor Bucket.after stamp) (project cursor nextSlot))) else (match (prim checked_subtract (project cursor nextSlot) (project cursor parameterCount) : (Option UInt64))
          | Option.none => (ctor Result.error [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.invalidLocalReference)))
          | Option.some creationIndex => (if (Uorc.Core.Heap::less stamp creationIndex) then (ctor Result.error [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.invalidLocalReference))) else (ctor Result.ok [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Event.assigned (record Cursor {parameterCount := (project cursor parameterCount), instructionCount := (project cursor instructionCount), slotCount := (project cursor slotCount), returnSlot := (project cursor returnSlot), nextSlot := following}) (ctor Bucket.after stamp) (project cursor nextSlot)))))) else (ctor Result.error [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Uorc.Core.RuntimeProgram::Failure.static (ctor Uorc.Core.Types::StaticError.invalidLocalReference)))))))) else (ctor Result.ok [Event, Uorc.Core.RuntimeProgram::Failure] (ctor Event.done))),
  axioms: [Quot.sound, propext],
  executable: true
}
