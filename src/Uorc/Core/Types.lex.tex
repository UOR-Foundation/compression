definition{
  name: wordOfNatAux,
  parameters: [depth : Nat, value : Nat],
  result: (Option UInt64),
  body: (match depth
    | Nat.zero => (if (value =? 0) then (ctor Option.some [UInt64] (0 : UInt64)) else (ctor Option.none [UInt64]))
    | Nat.succ remaining => (match (wordOfNatAux remaining (prim quotient value (Uorc.Wire.Domains::radix) 0 : Nat))
      | Option.none => (ctor Option.none [UInt64])
      | Option.some upper => (match (prim checked_multiply upper (2 : UInt64) : (Option UInt64))
        | Option.none => (ctor Option.none [UInt64])
        | Option.some doubled => (prim checked_add doubled (if ((prim remainder value (Uorc.Wire.Domains::radix) 0 : Nat) =? 0) then (0 : UInt64) else (1 : UInt64)) : (Option UInt64))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: depth
}
