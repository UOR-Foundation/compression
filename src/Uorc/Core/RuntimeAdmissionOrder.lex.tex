definition{
  name: capturePoint,
  parameters: [
    opcode : Uorc.Core.RuntimeOpcode::Opcode,
    lane : Uorc.Core.RuntimeProgram::CaptureLane,
    scalarsSeen : UInt64
  ],
  result: Bool,
  body: (match lane
    | Uorc.Core.RuntimeProgram::CaptureLane.first => (match (project (Uorc.Core.RuntimeOpcode::spec opcode) call)
      | Uorc.Core.RuntimeOpcode::CallRule.none => false
      | Uorc.Core.RuntimeOpcode::CallRule.apply => (prim equal scalarsSeen (1 : UInt64) : Bool)
      | Uorc.Core.RuntimeOpcode::CallRule.tabulate => (prim equal scalarsSeen (2 : UInt64) : Bool)
      | Uorc.Core.RuntimeOpcode::CallRule.fold => (prim equal scalarsSeen (3 : UInt64) : Bool)
      | Uorc.Core.RuntimeOpcode::CallRule.scan => (prim equal scalarsSeen (3 : UInt64) : Bool))
    | Uorc.Core.RuntimeProgram::CaptureLane.second => (match (project (Uorc.Core.RuntimeOpcode::spec opcode) call)
      | Uorc.Core.RuntimeOpcode::CallRule.none => false
      | Uorc.Core.RuntimeOpcode::CallRule.apply => false
      | Uorc.Core.RuntimeOpcode::CallRule.tabulate => false
      | Uorc.Core.RuntimeOpcode::CallRule.fold => false
      | Uorc.Core.RuntimeOpcode::CallRule.scan => (prim equal scalarsSeen (4 : UInt64) : Bool))),
  axioms: [Quot.sound, propext],
  executable: true
}
