definition{
  name: routesDiffer,
  parameters: [depth : Nat, keyA : UInt64, keyB : UInt64],
  result: Bool,
  body: (match depth
    | Nat.zero => false
    | Nat.succ remaining => (if (Uorc.Core.Heap::highBit keyA) then (if (Uorc.Core.Heap::highBit keyB) then (routesDiffer remaining (Uorc.Core.Heap::advanceKey keyA) (Uorc.Core.Heap::advanceKey keyB)) else true) else (if (Uorc.Core.Heap::highBit keyB) then true else (routesDiffer remaining (Uorc.Core.Heap::advanceKey keyA) (Uorc.Core.Heap::advanceKey keyB))))),
  axioms: [Quot.sound, propext],
  executable: false,
  recursive_argument: depth
}
