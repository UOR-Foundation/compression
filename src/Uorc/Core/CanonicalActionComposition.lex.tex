definition{
  name: indexValue,
  parameters: [sequence : Uorc.Core.ReferenceSemantics::Sequence, index : Nat],
  result: (Result Uorc.Core.ReferenceSemantics::Value Uorc.Core.ReferenceSemantics::Failure),
  body: (match (Uorc.Core.ReferenceSemantics::indexSequence sequence index)
    | Result.error failure => (ctor Result.error [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] failure)

  axioms: [],
  executable: false
}

definition{
  name: basisValue,
  parameters: [basis : (List Uorc.Core.ReferenceSemantics::Sequence), index : Nat],
  result: (Result Uorc.Core.ReferenceSemantics::Value Uorc.Core.ReferenceSemantics::Failure),
  body: (match (prim index basis index : (Option Uorc.Core.ReferenceSemantics::Sequence))
    | Option.none => (ctor Result.error [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] (ctor Uorc.Core.ReferenceSemantics::Failure.semantic (ctor Uorc.Core.ReferenceSemantics::SemanticError.basisOutOfBounds)))
    | Option.some sequence => (ctor Result.ok [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] (ctor Uorc.Core.ReferenceSemantics::Value.sequence sequence))),
  axioms: [],
  executable: false
}

definition{
  name: finishObserved,
  parameters: [
    result : (Result Uorc.Core.ReferenceSemantics::Value Uorc.Core.ReferenceSemantics::Failure),
    usage : Uorc.Core.CanonicalCost::Usage,
    deadBindings : (List Uorc.Core.CanonicalBindings::Binding)
  ],
  result: Step,
  body: (match result
    | Result.error failure => (ctor Step.completed (ctor Result.error [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] failure) usage)
    | Result.ok value => (ctor Step.pending (record Cursor {stage := (ctor Stage.returning value), pending := (Uorc.Core.CanonicalSchedule::completedResult (record Uorc.Core.CanonicalBindings::Binding {role := (ctor Uorc.Core.CanonicalBindings::Role.instruction), value := value}) deadBindings), usage := usage, deadBindings := deadBindings}))),
  axioms: [],
  executable: false
}

definition{
  name: beginIndexAttempt,
  parameters: [
    sequence : Uorc.Core.ReferenceSemantics::Sequence,
    index : Nat,
    usage : Uorc.Core.CanonicalCost::Usage,
    deadBindings : (List Uorc.Core.CanonicalBindings::Binding)
  ],
  result: Step,
  body: (ctor Step.pending (record Cursor {stage := (ctor Stage.indexAttempt sequence index), pending := (Uorc.Core.CanonicalSchedule::indexObservation), usage := usage, deadBindings := deadBindings})),
  axioms: [],
  executable: false
}

definition{
  name: prepareIndex,
  parameters: [
    sequence : Uorc.Core.ReferenceSemantics::Sequence,
    index : Nat,
    usage : Uorc.Core.CanonicalCost::Usage,
    deadBindings : (List Uorc.Core.CanonicalBindings::Binding)
  ],
  result: Step,
  body: (if (index <? (project sequence length)) then (beginIndexAttempt sequence index usage deadBindings) else (finishObserved (indexValue sequence index) usage deadBindings)),
  axioms: [],
  executable: false
}

definition{
  name: prepareIndexOperands,
  parameters: [
    slots : (List Uorc.Core.ReferenceSemantics::Value),
    sequenceReference : Nat,
    indexReference : Nat,
    usage : Uorc.Core.CanonicalCost::Usage,
    deadBindings : (List Uorc.Core.CanonicalBindings::Binding)
  ],
  result: Step,
  body: (match (Uorc.Core.ReferenceSemantics::readSequence slots sequenceReference)
    | Result.error failure => (ctor Step.completed (ctor Result.error [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] failure) usage)
    | Result.ok sequence => (match (Uorc.Core.ReferenceSemantics::readNatural slots indexReference)
      | Result.error failure => (ctor Step.completed (ctor Result.error [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] failure) usage)
      | Result.ok index => (prepareIndex sequence index usage deadBindings))),
  axioms: [],
  executable: false
}

definition{
  name: prepareBasisOperand,
  parameters: [
    basis : (List Uorc.Core.ReferenceSemantics::Sequence),
    slots : (List Uorc.Core.ReferenceSemantics::Value),
    indexReference : Nat,
    usage : Uorc.Core.CanonicalCost::Usage,
    deadBindings : (List Uorc.Core.CanonicalBindings::Binding)
  ],
  result: Step,
  body: (match (Uorc.Core.ReferenceSemantics::readNatural slots indexReference)
    | Result.error failure => (ctor Step.completed (ctor Result.error [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] failure) usage)
    | Result.ok index => (ctor Step.pending (record Cursor {stage := (ctor Stage.basisAttempt basis index), pending := (Uorc.Core.CanonicalSchedule::basisLookupEvents), usage := usage, deadBindings := deadBindings}))),
  axioms: [],
  executable: false
}

definition{
  name: dispatchReady,
  parameters: [cursor : Cursor],
  result: Step,
  body: (match (project cursor stage)
    | Stage.indexOperands slots sr ir => (prepareIndexOperands slots sr ir (project cursor usage) (project cursor deadBindings))
    | Stage.basisOperand basis slots ir => (prepareBasisOperand basis slots ir (project cursor usage) (project cursor deadBindings))
    | Stage.indexAttempt sequence index => (finishObserved (indexValue sequence index) (project cursor usage) (project cursor deadBindings))
    | Stage.basisAttempt basis index => (finishObserved (basisValue basis index) (project cursor usage) (project cursor deadBindings))
    | Stage.returning value => (ctor Step.completed (ctor Result.ok [Uorc.Core.ReferenceSemantics::Value, Uorc.Core.ReferenceSemantics::Failure] value) (project cursor usage))),
  axioms: [],
  executable: false
}

definition{
  name: withCommittedEvent,

    remaining : (List Uorc.Core.CanonicalCost::SemanticEvent),
    achieved : Uorc.Core.CanonicalCost::Usage
  ],
  result: Cursor,

  axioms: [],

definition{
  name: completeAttempt,
  parameters: [
    cursor : Cursor,

  ],
  result: Step,
  body: (match decision
    | Uorc.Core.CanonicalSchedule::EventDecision.refused failure => (ctor Step.refused failure cursor)

  axioms: [],
  executable: false
}

definition{
  name: step,
  parameters: [budget : Uorc.Core.CanonicalCost::Budget, cursor : Cursor],
  result: Step,
  body: (match (project cursor pending)
    | List.nil => (dispatchReady cursor)
    | List.cons event remaining => (completeAttempt cursor remaining (Uorc.Core.CanonicalSchedule::decideEvent budget (project cursor usage) event))),
  axioms: [],
  executable: false
}

  axioms: [],
  executable: false

definition{
  name: stepUsage,
  parameters: [result : Step],
  result: Uorc.Core.CanonicalCost::Usage,
  body: (match result
    | Step.pending c => (project c usage)
    | Step.completed meaning usage => usage
    | Step.refused failure c => (project c usage)),
  axioms: [],
  executable: false
}

definition{
  name: run,
  parameters: [fuel : Nat, budget : Uorc.Core.CanonicalCost::Budget, cursor : Cursor],
  result: Step,
  body: (match fuel
    | Nat.zero => (ctor Step.pending cursor)
    | Nat.succ remaining => (match (step budget cursor)
      | Step.pending next => (run remaining budget next)
      | Step.completed meaning usage => (ctor Step.completed meaning usage)
      | Step.refused failure retained => (ctor Step.refused failure retained))),
  axioms: [],
  executable: false,
  recursive_argument: fuel
}
