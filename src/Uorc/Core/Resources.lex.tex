definition{
  name: eventUsage,
  parameters: [u : Usage, e : Event],
  result: Usage,
  body: (record Usage {steps := ((project u steps) + (project e work)), liveCells := (prim subtract ((project u liveCells) + (project e allocatedCells)) (project e releasedCells) : Nat), peakCells := (if ((project u peakCells) <=? ((project u liveCells) + (project e allocatedCells))) then ((project u liveCells) + (project e allocatedCells)) else (project u peakCells)), outputBytes := ((project u outputBytes) + (project e emittedBytes)), frames := (prim subtract ((project u frames) + (project e enteredFrames)) (project e exitedFrames) : Nat)})
}

definition{
  name: validateUsage,
  parameters: [budget : Budget, u : Usage],
  result: (Result Usage OperationalError),
  body: (if ((project budget maxSteps) <? (project u steps)) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.stepLimit)) else (if ((project budget maxLiveCells) <? (project u peakCells)) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.liveCellLimit)) else (if ((project budget maxOutputBytes) <? (project u outputBytes)) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.outputLimit)) else (if ((project budget maxFrames) <? (project u frames)) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.frameLimit)) else (ctor Result.ok [Usage, OperationalError] u)))))
}

definition{
  name: fitsAddition,
  parameters: [limit : Nat, used : Nat, increase : Nat],
  result: Bool,
  body: (if (used <=? limit) then (increase <=? (prim subtract limit used : Nat)) else false)
}

definition{
  name: releaseUnderflows,
  parameters: [owned : Nat, allocated : Nat, released : Nat],
  result: Bool,
  body: (if (released <=? owned) then false else (allocated <? (prim subtract released owned : Nat)))
}

definition{
  name: charge,
  parameters: [budget : Budget, u : Usage, e : Event],
  result: (Result Usage OperationalError),
  body: (if (not (fitsAddition (project budget maxSteps) (project u steps) (project e work))) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.stepLimit)) else (if ((releaseUnderflows (project u liveCells) (project e allocatedCells) (project e releasedCells)) || (releaseUnderflows (project u frames) (project e enteredFrames) (project e exitedFrames))) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.internalInvariant)) else (if (((project budget maxLiveCells) <? (project u peakCells)) || (not (fitsAddition (project budget maxLiveCells) (project u liveCells) (project e allocatedCells)))) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.liveCellLimit)) else (if (not (fitsAddition (project budget maxOutputBytes) (project u outputBytes) (project e emittedBytes))) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.outputLimit)) else (if (not (fitsAddition (project budget maxFrames) (project u frames) (project e enteredFrames))) then (ctor Result.error [Usage, OperationalError] (ctor OperationalError.frameLimit)) else (ctor Result.ok [Usage, OperationalError] (eventUsage u e)))))))
}

definition{
  name: runEvents,
  parameters: [events : (List Event), budget : Budget, u : Usage],
  result: (Result Usage OperationalError),
  body: (match events
    | List.nil => (validateUsage budget u)
    | List.cons event rest => (match (charge budget u event)
      | Result.error error => (ctor Result.error [Usage, OperationalError] error)
      | Result.ok value => (runEvents rest budget value))),
  recursive_argument: events
}

theorem{
  name: success_guard_observation,
  parameters: [
    isRejected : Bool,
    failureValue : OperationalError,
    continuation : (Result Usage OperationalError),
    failureObservation : Prop,
    observation : (Fn (Usage) Prop)
  ],
  statement: ((match (if isRejected then (ctor Result.error [Usage, OperationalError] failureValue) else continuation)
    | Result.error rejectedError => failureObservation
    | Result.ok committedUsage => (apply observation committedUsage)) <-> (if isRejected then failureObservation else (match continuation
    | Result.error rejectedError => failureObservation
    | Result.ok committedUsage => (apply observation committedUsage)))),
  proof: cases{
    branches: [
      {proof: reflexivity{}, binders: [], constructor: "false"},
      {proof: reflexivity{}, binders: [], constructor: "true"}
    ],
    scrutinee: isRejected
  },
  axioms: []
}
