definition{
  name: lowBit,
  parameters: [key : UInt64],
  result: Bool,
  body: (prim equal (prim bit_and key (1 : UInt64) : UInt64) (1 : UInt64) : Bool),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: nextKey,
  parameters: [key : UInt64],
  result: UInt64,
  body: (match (prim shift_right key (1 : UInt32) : (Option UInt64))
    | Option.none => (0 : UInt64)
    | Option.some shifted => shifted),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: trieEmpty,
  parameters: [tree : Trie],
  result: Bool,
  body: (match tree
    | Trie.empty => true
    | Trie.leaf leafByte => false
    | Trie.branch left right => false)
}

definition{
  name: compactBranch,
  parameters: [left : Trie, right : Trie],
  result: Trie,
  body: (if ((trieEmpty left) && (trieEmpty right)) then (ctor Trie.empty) else (ctor Trie.branch left right))
}

definition{
  name: leftChild,
  parameters: [tree : Trie],
  result: Trie,
  body: (match tree
    | Trie.empty => (ctor Trie.empty)
    | Trie.leaf leafByte => (ctor Trie.empty)
    | Trie.branch left right => left)
}

definition{
  name: rightChild,
  parameters: [tree : Trie],
  result: Trie,
  body: (match tree
    | Trie.empty => (ctor Trie.empty)
    | Trie.leaf leafByte => (ctor Trie.empty)
    | Trie.branch left right => right)
}

definition{
  name: readTrie,
  parameters: [depth : Nat, tree : Trie, key : UInt64, defaultByte : UInt8],
  result: UInt8,
  body: (match depth
    | Nat.zero => (match tree
      | Trie.empty => defaultByte
      | Trie.leaf leafByte => leafByte
      | Trie.branch left right => defaultByte)
    | Nat.succ remaining => (match tree
      | Trie.empty => defaultByte
      | Trie.leaf leafByte => defaultByte
      | Trie.branch left right => (readTrie remaining (if (lowBit key) then right else left) (nextKey key) defaultByte))),
  axioms: [Quot.sound, propext],
  recursive_argument: depth
}

definition{
  name: writeTrie,
  parameters: [depth : Nat, tree : Trie, key : UInt64, value : UInt8, defaultByte : UInt8],
  result: Trie,
  body: (match depth
    | Nat.zero => (if (prim equal value defaultByte : Bool) then (ctor Trie.empty) else (ctor Trie.leaf value))
    | Nat.succ remaining => (if (lowBit key) then (compactBranch (leftChild tree) (writeTrie remaining (rightChild tree) (nextKey key) value defaultByte)) else (compactBranch (writeTrie remaining (leftChild tree) (nextKey key) value defaultByte) (rightChild tree)))),
  axioms: [Quot.sound, propext],
  recursive_argument: depth
}

definition{
  name: validTrie,
  parameters: [depth : Nat, tree : Trie, defaultByte : UInt8],
  result: Bool,
  body: (match depth
    | Nat.zero => (match tree
      | Trie.empty => true
      | Trie.leaf leafByte => (not (prim equal leafByte defaultByte : Bool))
      | Trie.branch left right => false)
    | Nat.succ remaining => (match tree
      | Trie.empty => true
      | Trie.leaf leafByte => false
      | Trie.branch left right => ((not ((trieEmpty left) && (trieEmpty right))) && ((validTrie remaining left defaultByte) && (validTrie remaining right defaultByte))))),
  recursive_argument: depth
}

definition{
  name: newStore,
  parameters: [defaultByte : UInt8],
  result: Store,
  body: (record Store {defaultByte := defaultByte, root := (ctor Trie.empty)})
}

definition{
  name: readStore,
  parameters: [store : Store, key : UInt64],
  result: UInt8,
  body: (readTrie 64 (project store root) key (project store defaultByte)),
  axioms: [Quot.sound, propext]
}

definition{
  name: writeStore,
  parameters: [store : Store, key : UInt64, value : UInt8],
  result: Store,
  body: (if (prim equal (readStore store key) value : Bool) then store else (record Store {defaultByte := (project store defaultByte), root := (writeTrie 64 (project store root) key value (project store defaultByte))})),
  axioms: [Quot.sound, propext]
}

definition{
  name: readExceptions,
  parameters: [exceptions : (List ReferenceException), key : UInt64, defaultByte : UInt8],
  result: UInt8,
  body: (match exceptions
    | List.nil => defaultByte
    | List.cons entry rest => (if (prim equal (project entry key) key : Bool) then (project entry value) else (readExceptions rest key defaultByte))),
  recursive_argument: exceptions
}

definition{
  name: eraseException,
  parameters: [exceptions : (List ReferenceException), key : UInt64],
  result: (List ReferenceException),
  body: (match exceptions
    | List.nil => (nil ReferenceException)
    | List.cons entry rest => (if (prim equal (project entry key) key : Bool) then (eraseException rest key) else (cons entry (eraseException rest key)))),
  recursive_argument: exceptions
}

definition{
  name: newReferenceStore,
  parameters: [defaultByte : UInt8],
  result: ReferenceStore,
  body: (record ReferenceStore {defaultByte := defaultByte, exceptions := (nil ReferenceException)})
}

definition{
  name: readReferenceStore,
  parameters: [store : ReferenceStore, key : UInt64],
  result: UInt8,
  body: (readExceptions (project store exceptions) key (project store defaultByte))
}

definition{
  name: writeReferenceStore,
  parameters: [store : ReferenceStore, key : UInt64, value : UInt8],
  result: ReferenceStore,
  body: (record ReferenceStore {defaultByte := (project store defaultByte), exceptions := (if (prim equal value (project store defaultByte) : Bool) then (eraseException (project store exceptions) key) else (cons (record ReferenceException {key := key, value := value}) (eraseException (project store exceptions) key)))})
}
