definition{
  name: fitsAddition,
  parameters: [limit : UInt64, used : UInt64, increase : UInt64],
  result: Bool,
  body: (match (prim checked_subtract limit used : (Option UInt64))
    | Option.none => false
    | Option.some remaining => (not (Uorc.Core.Heap::less remaining increase))),
  axioms: [Quot.sound, propext],
  executable: true
}

  executable: true

definition{
  name: validateUsage,
  parameters: [budget : Budget, current : Usage],
  result: (Result Usage Error),
  body: (if (fitsAddition (project budget maxWork) (project current work) (0 : UInt64)) then (if (Uorc.Core.Heap::less (project current peakOwnedBytes) (project current ownedBytes)) then (ctor Result.error [Usage, Error] (ctor Error.invalidUsage)) else (if (fitsAddition (project budget maxOwnedBytes) (project current peakOwnedBytes) (0 : UInt64)) then (if (fitsAddition (project budget maxOutputBytes) (project current outputBytes) (0 : UInt64)) then (if (fitsAddition (project budget maxFrames) (project current frames) (0 : UInt64)) then (ctor Result.ok [Usage, Error] current) else (ctor Result.error [Usage, Error] (ctor Error.frameLimit))) else (ctor Result.error [Usage, Error] (ctor Error.outputLimit))) else (ctor Result.error [Usage, Error] (ctor Error.ownedByteLimit)))) else (ctor Result.error [Usage, Error] (ctor Error.workLimit))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: charge,
  parameters: [budget : Budget, current : Usage, event : Event],
  result: (Result Usage Error),
  body: (match (validateUsage budget current)
    | Result.error usageError => (ctor Result.error [Usage, Error] usageError)
    | Result.ok validCurrent => (match (prim checked_add (project current work) (project event work) : (Option UInt64))
      | Option.none => (ctor Result.error [Usage, Error] (ctor Error.workLimit))
      | Option.some newWork => (if (Uorc.Core.Heap::less (project budget maxWork) newWork) then (ctor Result.error [Usage, Error] (ctor Error.workLimit)) else (match (prim checked_add (project current outputBytes) (project event emittedBytes) : (Option UInt64))
        | Option.none => (ctor Result.error [Usage, Error] (ctor Error.outputLimit))
        | Option.some newOutput => (if (Uorc.Core.Heap::less (project budget maxOutputBytes) newOutput) then (ctor Result.error [Usage, Error] (ctor Error.outputLimit)) else (match (prim checked_add (project current frames) (project event enteredFrames) : (Option UInt64))
          | Option.none => (ctor Result.error [Usage, Error] (ctor Error.frameLimit))
          | Option.some transientFrames => (if (Uorc.Core.Heap::less (project budget maxFrames) transientFrames) then (ctor Result.error [Usage, Error] (ctor Error.frameLimit)) else (match (prim checked_subtract transientFrames (project event exitedFrames) : (Option UInt64))
            | Option.none => (ctor Result.error [Usage, Error] (ctor Error.invalidUsage))
            | Option.some newFrames => (ctor Result.ok [Usage, Error] (record Usage {work := newWork, ownedBytes := (project current ownedBytes), peakOwnedBytes := (project current peakOwnedBytes), outputBytes := newOutput, frames := newFrames})))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: observeArena,
  parameters: [budget : Budget, current : Usage, arena : Uorc.Core.Heap::Arena],
  result: (Result Usage Error),
  body: (match (validateUsage budget current)
    | Result.error usageError => (ctor Result.error [Usage, Error] usageError)
    | Result.ok validCurrent => (if (prim equal (project arena maxOwnedBytes) (project budget maxOwnedBytes) : Bool) then (match (Uorc.Core.Heap::totalOwnedBytes arena)
      | Result.error heapError => (ctor Result.error [Usage, Error] (ctor Error.heap heapError))
      | Result.ok owned => (if (Uorc.Core.Heap::less (project budget maxOwnedBytes) owned) then (ctor Result.error [Usage, Error] (ctor Error.ownedByteLimit)) else (ctor Result.ok [Usage, Error] (record Usage {work := (project current work), ownedBytes := owned, peakOwnedBytes := (if (Uorc.Core.Heap::less (project current peakOwnedBytes) owned) then owned else (project current peakOwnedBytes)), outputBytes := (project current outputBytes), frames := (project current frames)})))) else (ctor Result.error [Usage, Error] (ctor Error.arenaBudgetMismatch)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: fromDiagnosticCounters,
  parameters: [
    diagnostic : Uorc.Core.Resources::Usage,
    budget : Budget,
    arena : Uorc.Core.Heap::Arena,
    previousPeakBytes : UInt64
  ],
  result: (Result Usage Error),
  body: (match (Uorc.Core.Heap::wordOfNat (project diagnostic steps))
    | Option.none => (ctor Result.error [Usage, Error] (ctor Error.addressSpaceExhausted))
    | Option.some oldWork => (match (Uorc.Core.Heap::wordOfNat (project diagnostic outputBytes))
      | Option.none => (ctor Result.error [Usage, Error] (ctor Error.addressSpaceExhausted))
      | Option.some oldOutput => (match (Uorc.Core.Heap::wordOfNat (project diagnostic frames))
        | Option.none => (ctor Result.error [Usage, Error] (ctor Error.addressSpaceExhausted))
        | Option.some oldFrames => (observeArena budget (record Usage {work := oldWork, ownedBytes := (0 : UInt64), peakOwnedBytes := previousPeakBytes, outputBytes := oldOutput, frames := oldFrames}) arena)))),
  axioms: [Quot.sound, propext]
}
