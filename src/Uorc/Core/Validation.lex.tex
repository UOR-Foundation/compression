definition{
  name: eventWork,
  parameters: [cursor : Cursor],
  result: Nat,
  body: (match (project cursor mode)
    | Mode.blocks => 1
    | Mode.parameters => (match (project cursor remainingParameters)
      | List.nil => 1
      | List.cons parameter remaining => 129)
    | Mode.instructions => 1
    | Mode.checking => (match (project cursor tasks)
      | List.nil => 1
      | List.cons nextTask otherTasks => (match nextTask
        | Task.expectLocal taskOperand0 taskOperand1 => 129
        | Task.bindLocal taskOperand0 => 129
        | Task.expectBoundLocal taskOperand0 => 129
        | Task.lookupBlock taskOperand0 => 129
        | Task.expectResult taskOperand0 => 1
        | Task.expectBoundResult => 1
        | Task.bindResult => 1
        | Task.prefix taskOperand0 => 129
        | Task.boundPrefix => 129
        | Task.arguments taskOperand0 => (match taskOperand0
          | List.nil => 1
          | List.cons reference remainingArguments => 129)
        | Task.endSignature => 1
        | Task.output taskOperand0 => 129
        | Task.outputBound => 129
        | Task.validRope taskOperand0 => 1))
    | Mode.returnValue => 129
    | Mode.lifetimeInit => 1
    | Mode.lifetime => (match (project cursor lifetimeCursor)
      | Option.none => 1
      | Option.some lifetimeValue => (Uorc.Core.Lifetime::eventWork lifetimeValue))
    | Mode.lifetimeCommit => 129
    | Mode.bodyCommit => 129
    | Mode.commitBlock => 129
    | Mode.entry => 129)
}
