definition{
  name: stepPending,
  parameters: [context : Context, machine : Machine, pending : Pending],
  result: StepResult,
  body: (match pending
    | Pending.resolving target references reversed => (match references
      | List.nil => (ctor StepResult.next (record Machine {frame := (project machine frame), continuations := (project machine continuations), usage := (project machine usage), comparison := (ctor Option.none [Uorc.Core.Bytes::Comparison]), pending := (ctor Option.some [Pending] (ctor Pending.reversing target reversed (nil Uorc.Core.Types::Value)))}))
      | List.cons reference remainingReferences => (match (Uorc.Core.Resources::charge (project context budget) (project machine usage) (record Uorc.Core.Resources::Event {kind := (ctor Uorc.Core.Resources::EventKind.control), work := 128, allocatedCells := 0, releasedCells := 0, emittedBytes := 0, enteredFrames := 0, exitedFrames := 0}))
        | Result.error budgetError => (ctor StepResult.failed (ctor Failure.operational budgetError) (project machine usage))
        | Result.ok chargedUsage => (match (localValue (project (project machine frame) slots) reference)
          | Result.error referenceError => (ctor StepResult.failed referenceError chargedUsage)
          | Result.ok selectedValue => (ctor StepResult.next (record Machine {frame := (project machine frame), continuations := (project machine continuations), usage := chargedUsage, comparison := (ctor Option.none [Uorc.Core.Bytes::Comparison]), pending := (ctor Option.some [Pending] (ctor Pending.resolving target remainingReferences (cons selectedValue reversed)))})))))
    | Pending.reversing target reversed captured => (match reversed
      | List.nil => (finishCaptures context (project machine frame) (project machine continuations) (project machine usage) target captured)
      | List.cons selectedValue remainingReversed => (ctor StepResult.next (record Machine {frame := (project machine frame), continuations := (project machine continuations), usage := (project machine usage), comparison := (ctor Option.none [Uorc.Core.Bytes::Comparison]), pending := (ctor Option.some [Pending] (ctor Pending.reversing target remainingReversed (cons selectedValue captured)))})))
    | Pending.initializing arguments => (match arguments
      | List.nil => (ctor StepResult.next (record Machine {frame := (project machine frame), continuations := (project machine continuations), usage := (project machine usage), comparison := (ctor Option.none [Uorc.Core.Bytes::Comparison]), pending := (ctor Option.none [Pending])}))
      | List.cons argumentValue remainingArguments => (match (Uorc.Core.Resources::charge (project context budget) (project machine usage) (record Uorc.Core.Resources::Event {kind := (ctor Uorc.Core.Resources::EventKind.control), work := 128, allocatedCells := 0, releasedCells := 0, emittedBytes := 0, enteredFrames := 0, exitedFrames := 0}))
        | Result.error budgetError => (ctor StepResult.failed (ctor Failure.operational budgetError) (project machine usage))
        | Result.ok chargedUsage => (match (Uorc.Core.Slots::appendSlot (project (project machine frame) slots) argumentValue)
          | Result.error slotError => (ctor StepResult.failed (match slotError
            | Uorc.Core.Slots::SlotError.invalidReference => (ctor Failure.static (ctor Uorc.Core.Types::StaticError.invalidLocalReference))
            | Uorc.Core.Slots::SlotError.releasedCell => (ctor Failure.operational (ctor Uorc.Core.Resources::OperationalError.internalInvariant))
            | Uorc.Core.Slots::SlotError.addressSpaceExhausted => (ctor Failure.operational (ctor Uorc.Core.Resources::OperationalError.addressSpaceExhausted))) chargedUsage)
          | Result.ok nextSlots => (ctor StepResult.next (record Machine {frame := (record Frame {blockIndex := (project (project machine frame) blockIndex), remaining := (project (project machine frame) remaining), slots := nextSlots, returnDistance := (project (project machine frame) returnDistance), nextInstruction := (project (project machine frame) nextInstruction), releaseBuckets := (project (project machine frame) releaseBuckets), pendingReleases := (project (project machine frame) pendingReleases)}), continuations := (project machine continuations), usage := chargedUsage, comparison := (ctor Option.none [Uorc.Core.Bytes::Comparison]), pending := (ctor Option.some [Pending] (ctor Pending.initializing remainingArguments))})))))
    | Pending.ropeWork workValue target => (match (Uorc.Core.Bytes::workStep workValue)
      | Uorc.Core.Bytes::WorkEvent.advanced nextWork => (ctor StepResult.next (record Machine {frame := (project machine frame), continuations := (project machine continuations), usage := (project machine usage), comparison := (ctor Option.none [Uorc.Core.Bytes::Comparison]), pending := (ctor Option.some [Pending] (ctor Pending.ropeWork nextWork target))}))
      | Uorc.Core.Bytes::WorkEvent.sequenceDone resultRope => (finishRopeWork context (project machine frame) (project machine continuations) (project machine usage) target resultRope)
      | Uorc.Core.Bytes::WorkEvent.byteDone resultByte => (match target
        | RopeTarget.byte => (nextAtomic context (project machine frame) (project machine continuations) (project machine usage) (ctor Result.ok [Uorc.Core.Types::Value, Failure] (ctor Uorc.Core.Types::Value.byte resultByte)))
        | RopeTarget.sequence => (ctor StepResult.failed (ctor Failure.operational (ctor Uorc.Core.Resources::OperationalError.internalInvariant)) (project machine usage)))
      | Uorc.Core.Bytes::WorkEvent.failed ropeError => (ctor StepResult.failed (ropeFailure ropeError) (project machine usage)))
    | Pending.builderWork workValue target => (match (Uorc.Core.Resources::charge (project context budget) (project machine usage) (record Uorc.Core.Resources::Event {kind := (ctor Uorc.Core.Resources::EventKind.byteWrite), work := (Uorc.Core.ByteBuilder::eventWork workValue), allocatedCells := 0, releasedCells := 0, emittedBytes := 0, enteredFrames := 0, exitedFrames := 0}))
      | Result.error builderBudgetError => (ctor StepResult.failed (ctor Failure.operational builderBudgetError) (project machine usage))
      | Result.ok builderUsage => (match (Uorc.Core.ByteBuilder::step workValue)
        | Uorc.Core.ByteBuilder::Event.advanced nextWork => (ctor StepResult.next (record Machine {frame := (project machine frame), continuations := (project machine continuations), usage := builderUsage, comparison := (ctor Option.none [Uorc.Core.Bytes::Comparison]), pending := (ctor Option.some [Pending] (ctor Pending.builderWork nextWork target))}))
        | Uorc.Core.ByteBuilder::Event.appended builder => (finishBuilderAppend context (project machine frame) (project machine continuations) builderUsage target builder)
        | Uorc.Core.ByteBuilder::Event.finished resultRope => (match target
          | BuilderTarget.sequence => (nextAtomic context (project machine frame) (project machine continuations) builderUsage (ctor Result.ok [Uorc.Core.Types::Value, Failure] (ctor Uorc.Core.Types::Value.segmentedSequence resultRope)))
          | BuilderTarget.tabulate count nextIndex bodyIndex captures => (ctor StepResult.failed (ctor Failure.operational (ctor Uorc.Core.Resources::OperationalError.internalInvariant)) builderUsage)
          | BuilderTarget.scanUpdate count index observerIndex observerCaptures updaterIndex updaterCaptures stateValue => (ctor StepResult.failed (ctor Failure.operational (ctor Uorc.Core.Resources::OperationalError.internalInvariant)) builderUsage))
        | Uorc.Core.ByteBuilder::Event.failed builderError => (ctor StepResult.failed (builderFailure builderError) builderUsage))))
}
