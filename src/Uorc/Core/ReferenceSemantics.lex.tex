definition{

  ],
  result: (Result Value Failure),
  body: (match remaining
    | Nat.zero => (ctor Result.ok [Value, Failure] (ctor Value.sequence output))
    | Nat.succ rest => (match (apply (project meaning run) (cons (ctor Value.natural index) captures))

        | Value.predicate payload => (ctor Result.error [Value, Failure] (ctor Failure.static (ctor Uorc.Core.Types::StaticError.typeMismatch)))
        | Value.sequence payload => (ctor Result.error [Value, Failure] (ctor Failure.static (ctor Uorc.Core.Types::StaticError.typeMismatch)))

          | Result.ok operationValue => (ctor Result.ok [Value, Failure] (ctor Value.natural operationValue)))))

      | Result.error leftError => (ctor Result.error [Value, Failure] leftError)
      | Result.ok left => (match (readNatural slots rightReference)
        | Result.error rightError => (ctor Result.error [Value, Failure] rightError)
        | Result.ok right => (match (checkedSubtract left right)
          | Result.error operationError => (ctor Result.error [Value, Failure] operationError)
          | Result.ok operationValue => (ctor Result.ok [Value, Failure] (ctor Value.natural operationValue)))))
    | Uorc.Core.Graph::Instruction.naturalMultiply leftReference rightReference => (match (readNatural slots leftReference)
      | Result.error leftError => (ctor Result.error [Value, Failure] leftError)
