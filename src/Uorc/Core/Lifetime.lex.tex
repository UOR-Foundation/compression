definition{
  name: referenceUses,
  parameters: [references : (List Nat), available : Nat, event : Nat],
  result: (Result (List Use) Uorc.Core.Types::StaticError),
  body: (match references
    | List.nil => (ctor Result.ok [(List Use), Uorc.Core.Types::StaticError] (nil Use))
    | List.cons distance rest => (if ((distance =? 0) || (available <? distance)) then (ctor Result.error [(List Use), Uorc.Core.Types::StaticError] (ctor Uorc.Core.Types::StaticError.invalidLocalReference)) else (match (referenceUses rest available event)
      | Result.error error => (ctor Result.error [(List Use), Uorc.Core.Types::StaticError] error)
      | Result.ok tailUses => (ctor Result.ok [(List Use), Uorc.Core.Types::StaticError] (cons (record Use {slot := (prim subtract available distance : Nat), event := event}) tailUses))))),
  recursive_argument: references
}

definition{
  name: instructionUses,
  parameters: [
    instructions : (List Uorc.Core.Graph::Instruction),
    available : Nat,
    event : Nat,
    returnDistance : Nat
  ],
  result: (Result (List Use) Uorc.Core.Types::StaticError),
  body: (match instructions
    | List.nil => (referenceUses (cons returnDistance (nil Nat)) available event)
    | List.cons instruction rest => (match (referenceUses (operandReferences instruction) available event)
      | Result.error headError => (ctor Result.error [(List Use), Uorc.Core.Types::StaticError] headError)
      | Result.ok headUses => (match (instructionUses rest (available + 1) (event + 1) returnDistance)
        | Result.error tailError => (ctor Result.error [(List Use), Uorc.Core.Types::StaticError] tailError)
        | Result.ok tailUses => (ctor Result.ok [(List Use), Uorc.Core.Types::StaticError] (prim append headUses tailUses : (List Use)))))),
  recursive_argument: instructions
}

definition{

  body: (match uses
    | List.nil => 0
    | List.cons use rest => (if ((if ((project use slot) =? slot) then (project use event) else 0) <=? (lastEvent rest slot)) then (lastEvent rest slot) else (if ((project use slot) =? slot) then (project use event) else 0))),
  recursive_argument: uses

definition{

  recursive_argument: remaining
}

definition{

      | NatTrie.empty => 0

      | NatTrie.empty => 0
      | NatTrie.leaf leafValue => 0
