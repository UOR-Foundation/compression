definition{
  name: scanIndex,
  parameters: [depth : Nat, tree : Uorc.Core.Heap::Index, key : UInt64, bit : UInt64],
  result: Scan,
  body: (match depth
    | Nat.zero => (match tree
      | Uorc.Core.Heap::Index.empty => (record Scan {rows := (nil Row), wellFormed := true})
      | Uorc.Core.Heap::Index.leaf entry => (record Scan {rows := (cons (record Row {handle := (record Uorc.Core.RuntimeValue::Handle {id := key}), entry := entry}) (nil Row)), wellFormed := true})
      | Uorc.Core.Heap::Index.branch left right => (record Scan {rows := (nil Row), wellFormed := false}))
    | Nat.succ remaining => (match tree
      | Uorc.Core.Heap::Index.empty => (record Scan {rows := (nil Row), wellFormed := true})
      | Uorc.Core.Heap::Index.leaf entry => (record Scan {rows := (nil Row), wellFormed := false})
      | Uorc.Core.Heap::Index.branch left right => (let leftScan : Scan := (scanIndex remaining left key (Uorc.Core.Heap::wordAdd bit bit)); (let rightScan : Scan := (scanIndex remaining right (prim bit_or key bit : UInt64) (Uorc.Core.Heap::wordAdd bit bit)); (record Scan {rows := (appendRows (project leftScan rows) (project rightScan rows)), wellFormed := (if (project leftScan wellFormed) then (project rightScan wellFormed) else false)}))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: depth
}

definition{
  name: interpretSequence,
  parameters: [fuel : Nat, arena : Uorc.Core.Heap::Arena, handle : Uorc.Core.RuntimeValue::Handle],
  result: (Result Bytes Uorc.Core.Heap::HeapError),
  body: (match fuel
    | Nat.zero => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
    | Nat.succ remaining => (if (Uorc.Core.RuntimeValue::isNull handle) then (ctor Result.ok [Bytes, Uorc.Core.Heap::HeapError] 0x) else (match (Uorc.Core.Heap::lookup arena handle)
      | Result.error e => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
      | Result.ok entry => (match (project entry node)
        | Uorc.Core.Heap::Node.sequenceLeaf image offset length => (match (Uorc.Core.Heap::lookup arena image)
          | Result.error e => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
          | Result.ok imageEntry => (match (project imageEntry node)
            | Uorc.Core.Heap::Node.image k bs => (match (prim slice bs (wordToNat offset) (wordToNat length) : (Option Bytes))
              | Option.none => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
              | Option.some bytes => (ctor Result.ok [Bytes, Uorc.Core.Heap::HeapError] bytes))
            | Uorc.Core.Heap::Node.sequenceLeaf unused0 unused1 unused2 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.sequenceBranch unused0 unused1 unused2 unused3 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.storeLeaf unused0 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.storeBranch unused0 unused1 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.slot unused0 unused1 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.rootBranch unused0 unused1 unused2 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.metadataWord unused0 unused1 unused2 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.metadataLink unused0 unused1 unused2 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.metadataBranch unused0 unused1 unused2 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.ownerRoot unused0 unused1 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Uorc.Core.Heap::Node.imageReservation unused0 unused1 => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))))
        | Uorc.Core.Heap::Node.sequenceBranch n height left right => (match (interpretSequence remaining arena left)
          | Result.error e => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
          | Result.ok leftBytes => (match (interpretSequence remaining arena right)
            | Result.error e => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
            | Result.ok rightBytes => (ctor Result.ok [Bytes, Uorc.Core.Heap::HeapError] (prim append leftBytes rightBytes : Bytes))))
        | Uorc.Core.Heap::Node.image kind bytes => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
        | Uorc.Core.Heap::Node.storeLeaf byte => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
        | Uorc.Core.Heap::Node.storeBranch left right => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
        | Uorc.Core.Heap::Node.slot role value => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))
        | Uorc.Core.Heap::Node.rootBranch role left right => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode))

        | Uorc.Core.Heap::Node.imageReservation kind length => (ctor Result.error [Bytes, Uorc.Core.Heap::HeapError] (ctor Uorc.Core.Heap::HeapError.invalidNode)))))),
