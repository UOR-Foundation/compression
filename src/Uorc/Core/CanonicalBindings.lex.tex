theorem{
  name: binding_allocation_preserves_storage_relation,
  parameters: [
    usage : Uorc.Core.CanonicalCost::Usage,
    inventory : (List Binding),
    binding : Binding
  ],
  statement: (((project usage semanticStorageUnits) = (inventoryUnits inventory)) -> ((project (Uorc.Core.CanonicalCost::advance usage (ctor Uorc.Core.CanonicalCost::SemanticEvent.allocateValueUnits (units binding))) semanticStorageUnits) = (inventoryUnits (push inventory binding)))),
  proof: linear_arithmetic{
    definitions: [
      Uorc.Core.CanonicalCost::advance,
      Uorc.Core.CanonicalCost::allocatedUnits,
      Uorc.Core.CanonicalCost::releasedUnits,
      inventoryUnits,
      push,
      units
    ]
  },
  axioms: [Quot.sound, propext]
}

theorem{
  name: parameters_preserve_order_and_values,
  parameters: [values : (List Uorc.Core.ReferenceSemantics::Value)],
  statement: ((inventoryValues (parameters values)) = values),
  proof: induction{
    branches: [
      {proof: reflexivity{}, binders: [], constructor: nil},
      {
        proof: simplify{definitions: [ih, inventoryValues, parameter, parameters]},
        binders: [value, rest, ih],
        constructor: cons
      }
    ],
    scrutinee: values
  },
  axioms: [propext]
}

theorem{
  name: nested_binding_inventory_is_compositional,
  parameters: [first : (List Binding), second : (List Binding)],
  statement: ((inventoryUnits (appendBindings var{
    name: first
  } var{name: second})) = ((inventoryUnits var{name: first}) + (inventoryUnits var{name: second}))),
  proof: induction{
    branches: [
      {
        proof: linear_arithmetic{definitions: [appendBindings, inventoryUnits]},
        binders: [],
        constructor: nil
      },
      {
        proof: simplify{
          definitions: [
            appendBindings,
            ih,
            inventoryUnits,
            Uorc.Core.CanonicalCost::work_sum_associativity
          ]
        },
        binders: [binding, rest, ih],
        constructor: cons
      }
    ],
    scrutinee: first
  },
  axioms: [Quot.sound, propext]
}

theorem{
