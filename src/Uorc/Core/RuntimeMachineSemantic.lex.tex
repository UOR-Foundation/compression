definition{
  name: checkedAdvance,
  parameters: [ledger : Ledger, event : Event],
  result: (Result Ledger Error),
  body: (if (profileCompatible (project ledger profile)) then (match (prim checked_add (project ledger semanticWork) (eventWork event) : (Option UInt64))
    | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.wordOverflow))
    | Option.some work => (match (prim checked_add (project ledger semanticStorageUnits) (allocatedUnits event) : (Option UInt64))
      | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.wordOverflow))
      | Option.some allocated => (match (prim checked_subtract allocated (releasedUnits event) : (Option UInt64))
        | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.storageUnderflow))
        | Option.some storage => (match (prim checked_add (project ledger activeFrames) (enteredFrames event) : (Option UInt64))
          | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.wordOverflow))
          | Option.some entered => (match (prim checked_subtract entered (leftFrames event) : (Option UInt64))
            | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.frameUnderflow))
            | Option.some frames => (match (prim checked_add (project ledger outputBytes) (outputCount event) : (Option UInt64))
              | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.wordOverflow))
              | Option.some output => (match (prim checked_add (project ledger acceptedArchives) (eofCount event) : (Option UInt64))
                | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.wordOverflow))
                | Option.some archives => (match (prim checked_add (project ledger basisObjects) (basisCount event) : (Option UInt64))
                  | Option.none => (ctor Result.error [Ledger, Error] (ctor Error.wordOverflow))
                  | Option.some basis => (ctor Result.ok [Ledger, Error] (record Ledger {profile := (project ledger profile), semanticWork := work, semanticStorageUnits := storage, peakSemanticStorageUnits := (maximum (project ledger peakSemanticStorageUnits) storage), outputBytes := output, activeFrames := frames, peakFrames := (maximum (project ledger peakFrames) frames), acceptedArchives := archives, basisObjects := basis})))))))))) else (ctor Result.error [Ledger, Error] (ctor Error.unsupportedProfile))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: charge,
  parameters: [budget : Budget, ledger : Ledger, event : Event],
  result: (Result Ledger Error),
  body: (if (profileCompatible (project budget profile)) then (if (profileCompatible (project ledger profile)) then (if (storageSupported (project budget storage)) then (if (validLedger ledger) then (match (checkedAdvance ledger event)
    | Result.error failure => (ctor Result.error [Ledger, Error] failure)
    | Result.ok next => (if (Uorc.Core.Heap::less (project budget maxSemanticWork) (project next semanticWork)) then (ctor Result.error [Ledger, Error] (ctor Error.workLimit)) else (if (storageBounded (project budget storage)) then (if (Uorc.Core.Heap::less (project budget maxStorageUnits) (project next peakSemanticStorageUnits)) then (ctor Result.error [Ledger, Error] (ctor Error.storageLimit)) else (ctor Result.ok [Ledger, Error] next)) else (ctor Result.ok [Ledger, Error] next)))) else (ctor Result.error [Ledger, Error] (ctor Error.invalidUsage))) else (ctor Result.error [Ledger, Error] (ctor Error.unsupportedStorage))) else (ctor Result.error [Ledger, Error] (ctor Error.unsupportedProfile))) else (ctor Result.error [Ledger, Error] (ctor Error.unsupportedProfile))),
  axioms: [Quot.sound, propext],
  executable: true
