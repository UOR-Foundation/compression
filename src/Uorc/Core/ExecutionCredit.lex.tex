definition{
  name: runWork,
  parameters: [work : Nat, context : Uorc.Core.StepMachine::Context, cursorValue : Cursor],
  result: Tick,
  body: (match work
    | Nat.zero => (ctor Tick.pending cursorValue)
    | Nat.succ remainingWork => (match (tick context cursorValue)
      | Tick.pending nextCursor => (runWork remainingWork context nextCursor)
      | Tick.done execution workLedger => (ctor Tick.done execution workLedger)
      | Tick.failed failure usage workLedger => (ctor Tick.failed failure usage workLedger)
      | Tick.rejected error workLedger => (ctor Tick.rejected error workLedger))),
  recursive_argument: work
