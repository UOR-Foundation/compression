definition{
  name: traceShape,
  parameters: [trace : (List Frame), terminal : Bool],
  result: Bool,
  body: (match trace
    | List.nil => false
    | List.cons head tail => ((frameShape head) && ((prim equal (project head terminal) terminal : Bool) && ((match tail
      | List.nil => (project head first)
      | List.cons prior remaining => ((if (project head first) then false else true) && ((prim equal (project head previous) (project prior digest) : Bool) && ((traceShape tail false) && true)))) && true)))),
  axioms: [Quot.sound, propext],
  recursive_argument: trace
}

definition{
  name: walk,
  parameters: [fuel : Nat, left : (List Frame), right : (List Frame)],
  result: Extraction,
  body: (match fuel
    | Nat.zero => (ctor Extraction.fuelExhausted)
    | Nat.succ rest => (match left
      | List.nil => (ctor Extraction.malformed)
      | List.cons leftHead leftTail => (match right
        | List.nil => (ctor Extraction.malformed)
        | List.cons rightHead rightTail => (if (prim equal (project leftHead digest) (project rightHead digest) : Bool) then (if (prim equal (message leftHead) (message rightHead) : Bool) then (if (project leftHead first) then (if (project rightHead first) then (ctor Extraction.equalStreams) else (ctor Extraction.malformed)) else (if (project rightHead first) then (ctor Extraction.malformed) else (walk rest leftTail rightTail))) else (ctor Extraction.candidate (record Candidate {left := leftHead, right := rightHead}))) else (ctor Extraction.differentDigest))))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: extract,
  parameters: [fuel : Nat, left : (List Frame), right : (List Frame)],
  result: Extraction,
  body: (if ((traceShape left true) && ((traceShape right true) && true)) then (walk fuel left right) else (ctor Extraction.malformed)),
  axioms: [Quot.sound, propext]
}

definition{

  axioms: []
}

definition{
  name: checkCandidate,
  parameters: [candidate : Candidate],
  result: Verification,
  body: (if ((frameShape (project candidate left)) && ((frameShape (project candidate right)) && true)) then (if (prim equal (message (project candidate left)) (message (project candidate right)) : Bool) then (ctor Verification.sameMessage) else (match (Uorc.Integrity.Sha256::hash (message (project candidate left)))
    | Result.error leftFailure => (ctor Verification.shaFailure)
    | Result.ok leftDigest => (match (Uorc.Integrity.Sha256::hash (message (project candidate right)))
      | Result.error rightFailure => (ctor Verification.shaFailure)
      | Result.ok rightDigest => (if (acceptanceGuard ((frameShape (project candidate left)) && ((frameShape (project candidate right)) && true)) (if (prim equal (message (project candidate left)) (message (project candidate right)) : Bool) then false else true) (prim equal leftDigest rightDigest : Bool)) then (ctor Verification.verified (record CheckedWitness {leftMessage := (message (project candidate left)), rightMessage := (message (project candidate right)), digest := leftDigest})) else (ctor Verification.differentDigest))))) else (ctor Verification.malformed)),
  axioms: [Quot.sound, propext]
}
