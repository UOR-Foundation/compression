definition{
  name: runUpdate,
  parameters: [fuel : Nat, job : Update],
  result: (Result State Error),
  body: (match fuel
    | Nat.zero => (ctor Result.error [State, Error] (ctor Error.fuelExhausted))
    | Nat.succ rest => (match (updateStep job)
      | Result.error failure => (ctor Result.error [State, Error] failure)
      | Result.ok event => (match event
        | Event.advanced nextJob => (runUpdate rest nextJob)
        | Event.done state => (ctor Result.ok [State, Error] state)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}

definition{
  name: runFinalize,
  parameters: [fuel : Nat, state : State],
  result: (Result Bytes Error),
  body: (match fuel
    | Nat.zero => (ctor Result.error [Bytes, Error] (ctor Error.fuelExhausted))
    | Nat.succ rest => (match (digest state)
      | Result.ok output => (ctor Result.ok [Bytes, Error] output)
      | Result.error pendingError => (match (step state)
        | Result.error failure => (ctor Result.error [Bytes, Error] failure)
        | Result.ok nextState => (runFinalize rest nextState)))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
