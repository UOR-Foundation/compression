definition{
  name: collectFixture,
  parameters: [
    fuel : Nat,
    event : Event,
    parts : (List Bytes),
    quantum : Nat,
    current : Bytes,
    objects : (List Bytes)
  ],
  result: FixtureResult,
  body: (match fuel
    | Nat.zero => (ctor FixtureResult.pending)
    | Nat.succ fuelLeft => (match event
      | Event.advanced advancedState => (collectFixture fuelLeft (step advancedState quantum) parts quantum current objects)
      | Event.needInput demandState => (match parts
        | List.nil => (collectFixture fuelLeft (finishInput demandState) parts quantum current objects)
        | List.cons nextPart restParts => (collectFixture fuelLeft (feed demandState nextPart) restParts quantum current objects))
      | Event.beginObject ignoredIndex ignoredLength beginState => (collectFixture fuelLeft (step beginState quantum) parts quantum 0x objects)
      | Event.bytes ignoredByteIndex pieceBytes byteState => (collectFixture fuelLeft (step byteState quantum) parts quantum (prim append current pieceBytes : Bytes) objects)
      | Event.endObject ignoredEndIndex endState => (collectFixture fuelLeft (step endState quantum) parts quantum 0x (cons current objects))
      | Event.complete ignoredCount => (ctor FixtureResult.done (reverseObjects objects (nil Bytes)))
      | Event.failed failure => (ctor FixtureResult.failed (project failure kind)))),
  recursive_argument: fuel
}

definition{
  name: fixtureMatches,
  parameters: [actual : FixtureResult, expected : FixtureResult],
  result: Bool,
  body: (match actual
    | FixtureResult.pending => false
    | FixtureResult.done actualObjects => (match expected
      | FixtureResult.pending => false
      | FixtureResult.done expectedObjects => (fixtureObjectsEqual actualObjects expectedObjects)
      | FixtureResult.failed ignoredExpectedError => false)
    | FixtureResult.failed actualError => (match expected
      | FixtureResult.pending => false
      | FixtureResult.done ignoredExpectedObjects => false
      | FixtureResult.failed expectedError => (fixtureErrorsEqual actualError expectedError)))
}
