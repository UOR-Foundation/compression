definition{
  name: step,
  parameters: [s : State, quantum : Nat],
  result: Event,
  body: (if (quantum =? 0) then (ctor Event.failed (ctor EmitError.invalidQuantum) (project s position) ((project s work) + 1)) else (match (project s pending)
    | List.nil => (ctor Event.complete (project s position) (project s work))
    | List.cons head tail => (match head
      | Task.byte value => (match (Uorc.Core.Types::wordOfNatAux 8 value)
        | Option.none => (ctor Event.failed (ctor EmitError.invalidByte) (project s position) ((project s work) + 1))
        | Option.some byteWord => (match (prim checked_convert byteWord : (Option UInt8))
          | Option.none => (ctor Event.failed (ctor EmitError.invalidByte) (project s position) ((project s work) + 1))
          | Option.some byteValue => (ctor Event.produced (Uorc.Core.Types::singletonSequence byteValue) (record State {pending := tail, position := ((project s position) + 1), work := ((project s work) + 1)}))))
      | Task.number value => (if (value <? 128) then (match (Uorc.Core.Types::wordOfNatAux 8 value)
        | Option.none => (ctor Event.failed (ctor EmitError.invalidByte) (project s position) ((project s work) + 1))
        | Option.some byteWord => (match (prim checked_convert byteWord : (Option UInt8))
          | Option.none => (ctor Event.failed (ctor EmitError.invalidByte) (project s position) ((project s work) + 1))
          | Option.some byteValue => (ctor Event.produced (Uorc.Core.Types::singletonSequence byteValue) (record State {pending := tail, position := ((project s position) + 1), work := ((project s work) + 1)})))) else (match (Uorc.Core.Types::wordOfNatAux 8 ((prim remainder value 128 0 : Nat) + 128))
        | Option.none => (ctor Event.failed (ctor EmitError.invalidByte) (project s position) ((project s work) + 1))
        | Option.some byteWord => (match (prim checked_convert byteWord : (Option UInt8))
          | Option.none => (ctor Event.failed (ctor EmitError.invalidByte) (project s position) ((project s work) + 1))
          | Option.some byteValue => (ctor Event.produced (Uorc.Core.Types::singletonSequence byteValue) (record State {pending := (cons (ctor Task.number (prim quotient value 128 0 : Nat)) tail), position := ((project s position) + 1), work := ((project s work) + 1)})))))
      | Task.rope value remaining => (match (Uorc.Core.Bytes::nextChunk value quantum)
        | Result.error ignored => (ctor Event.failed (ctor EmitError.invalidRope) (project s position) ((project s work) + 1))
        | Result.ok ropeEvent => (match ropeEvent
          | Uorc.Core.Bytes::CursorEvent.advanced next => (ctor Event.advanced (record State {pending := (cons (ctor Task.rope next remaining) tail), position := (project s position), work := ((project s work) + 1)}))
          | Uorc.Core.Bytes::CursorEvent.produced chunk next => (ctor Event.produced chunk (record State {pending := (cons (ctor Task.rope next (prim subtract remaining (prim length chunk : Nat) : Nat)) tail), position := ((project s position) + (prim length chunk : Nat)), work := ((project s work) + 1)}))
          | Uorc.Core.Bytes::CursorEvent.done => (ctor Event.advanced (record State {pending := tail, position := (project s position), work := ((project s work) + 1)}))))
      | Task.countTypes remaining original count => (match remaining
        | List.nil => (ctor Event.advanced (record State {pending := (cons (ctor Task.number count) (cons (ctor Task.types original) tail)), position := (project s position), work := ((project s work) + 1)}))
        | List.cons ignoredItemHead rest => (ctor Event.advanced (record State {pending := (cons (ctor Task.countTypes rest original (count + 1)) tail), position := (project s position), work := ((project s work) + 1)})))
      | Task.types values => (match values
        | List.nil => (ctor Event.advanced (record State {pending := tail, position := (project s position), work := ((project s work) + 1)}))
        | List.cons itemHead rest => (ctor Event.advanced (record State {pending := (cons (ctor Task.byte (typeCode itemHead)) (cons (ctor Task.types rest) tail)), position := (project s position), work := ((project s work) + 1)})))
      | Task.countReferences remaining original count => (match remaining
        | List.nil => (ctor Event.advanced (record State {pending := (cons (ctor Task.number count) (cons (ctor Task.references original) tail)), position := (project s position), work := ((project s work) + 1)}))
        | List.cons ignoredItemHead rest => (ctor Event.advanced (record State {pending := (cons (ctor Task.countReferences rest original (count + 1)) tail), position := (project s position), work := ((project s work) + 1)})))
      | Task.references values => (match values
        | List.nil => (ctor Event.advanced (record State {pending := tail, position := (project s position), work := ((project s work) + 1)}))
        | List.cons itemHead rest => (ctor Event.advanced (record State {pending := (cons (ctor Task.number itemHead) (cons (ctor Task.references rest) tail)), position := (project s position), work := ((project s work) + 1)})))
      | Task.countInstructions remaining original count => (match remaining
        | List.nil => (ctor Event.advanced (record State {pending := (cons (ctor Task.number count) (cons (ctor Task.instructions original) tail)), position := (project s position), work := ((project s work) + 1)}))
        | List.cons ignoredItemHead rest => (ctor Event.advanced (record State {pending := (cons (ctor Task.countInstructions rest original (count + 1)) tail), position := (project s position), work := ((project s work) + 1)})))
      | Task.instructions values => (match values
        | List.nil => (ctor Event.advanced (record State {pending := tail, position := (project s position), work := ((project s work) + 1)}))
        | List.cons itemHead rest => (ctor Event.advanced (record State {pending := (cons (ctor Task.instruction itemHead) (cons (ctor Task.instructions rest) tail)), position := (project s position), work := ((project s work) + 1)})))
      | Task.countBlocks remaining original count => (match remaining
        | List.nil => (ctor Event.advanced (record State {pending := (cons (ctor Task.number count) (cons (ctor Task.blocks original) tail)), position := (project s position), work := ((project s work) + 1)}))

      | Task.blocks values => (match values

      | Task.instruction value => (ctor Event.advanced (record State {pending := (prim append (instructionTasks value) tail : (List Task)), position := (project s position), work := ((project s work) + 1)}))
      | Task.block value => (ctor Event.advanced (record State {pending := (cons (ctor Task.countTypes (project value parameters) (project value parameters) 0) (cons (ctor Task.byte (typeCode (project value result))) (cons (ctor Task.countInstructions (project value instructions) (project value instructions) 0) (cons (ctor Task.number (project value returnDistance)) tail)))), position := (project s position), work := ((project s work) + 1)})))))

definition{
  name: collectFixture,
  parameters: [fuel : Nat, s : State, acc : Bytes],
  result: (Option Bytes),
  body: (match fuel
    | Nat.zero => (ctor Option.none [Bytes])
    | Nat.succ remaining => (match (step s 16)
      | Event.complete size work => (ctor Option.some [Bytes] acc)
      | Event.failed reason position work => (ctor Option.none [Bytes])
      | Event.advanced next => (collectFixture remaining next acc)
      | Event.produced chunk next => (collectFixture remaining next (prim append acc chunk : Bytes)))),
  recursive_argument: fuel
}
