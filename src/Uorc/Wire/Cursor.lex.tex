definition{
  name: readStep,
  parameters: [c : Cursor],
  result: ReadEvent,
  body: (match (project c parts)
    | List.nil => (if ((project c offset) =? 0) then (ctor ReadEvent.end c) else (ctor ReadEvent.failed (record LocatedError {kind := (ctor ReadError.invalidCursor), offset := (project c position)})))
    | List.cons head tail => (if ((project c offset) <? (prim length head : Nat)) then (match (prim index head (project c offset) : (Option UInt8))
      | Option.none => (ctor ReadEvent.failed (record LocatedError {kind := (ctor ReadError.invalidCursor), offset := (project c position)}))
      | Option.some b => (ctor ReadEvent.byte b (record Cursor {parts := (project c parts), offset := ((project c offset) + 1), position := ((project c position) + 1)}))) else (if ((project c offset) =? (prim length head : Nat)) then (ctor ReadEvent.advanced (record Cursor {parts := tail, offset := 0, position := (project c position)})) else (ctor ReadEvent.failed (record LocatedError {kind := (ctor ReadError.invalidCursor), offset := (project c position)})))))
}

definition{
  name: readChunkStep,
  parameters: [c : Cursor, quantum : Nat],
  result: ChunkEvent,
  body: (if (quantum =? 0) then (ctor ChunkEvent.failed (record LocatedError {kind := (ctor ReadError.invalidQuantum), offset := (project c position)})) else (match (project c parts)
    | List.nil => (if ((project c offset) =? 0) then (ctor ChunkEvent.end c) else (ctor ChunkEvent.failed (record LocatedError {kind := (ctor ReadError.invalidCursor), offset := (project c position)})))
    | List.cons head tail => (if ((project c offset) <? (prim length head : Nat)) then (match (prim slice head (project c offset) (if ((prim subtract (prim length head : Nat) (project c offset) : Nat) <? quantum) then (prim subtract (prim length head : Nat) (project c offset) : Nat) else quantum) : (Option Bytes))
      | Option.none => (ctor ChunkEvent.failed (record LocatedError {kind := (ctor ReadError.invalidCursor), offset := (project c position)}))
      | Option.some chunk => (ctor ChunkEvent.bytes chunk (record Cursor {parts := (project c parts), offset := ((project c offset) + (if ((prim subtract (prim length head : Nat) (project c offset) : Nat) <? quantum) then (prim subtract (prim length head : Nat) (project c offset) : Nat) else quantum)), position := ((project c position) + (if ((prim subtract (prim length head : Nat) (project c offset) : Nat) <? quantum) then (prim subtract (prim length head : Nat) (project c offset) : Nat) else quantum))}))) else (if ((project c offset) =? (prim length head : Nat)) then (ctor ChunkEvent.advanced (record Cursor {parts := tail, offset := 0, position := (project c position)})) else (ctor ChunkEvent.failed (record LocatedError {kind := (ctor ReadError.invalidCursor), offset := (project c position)}))))))
}

definition{
  name: ulebStep,
  parameters: [u : UlebState],
  result: UlebEvent,
  body: (if (10 <=? (project u consumed)) then (ctor UlebEvent.failed (record LocatedError {kind := (ctor ReadError.overflow), offset := ((project u start) + (project u consumed))})) else (match (readStep (project u cursor))
    | ReadEvent.advanced next => (ctor UlebEvent.more (record UlebState {cursor := next, start := (project u start), value := (project u value), factor := (project u factor), consumed := (project u consumed)}))
    | ReadEvent.byte b next => (if ((prim to_nat b : Nat) <? 128) then (if (((project u consumed) =? 9) && ((1 <? (prim remainder (prim to_nat b : Nat) 128 0 : Nat)) && true)) then (ctor UlebEvent.failed (record LocatedError {kind := (ctor ReadError.overflow), offset := ((project u start) + (project u consumed))})) else (if ((0 <? (project u consumed)) && (((prim remainder (prim to_nat b : Nat) 128 0 : Nat) =? 0) && true)) then (ctor UlebEvent.failed (record LocatedError {kind := (ctor ReadError.nonShortest), offset := ((project u start) + (project u consumed))})) else (ctor UlebEvent.done ((project u value) + (prim multiply (prim remainder (prim to_nat b : Nat) 128 0 : Nat) (project u factor) : Nat)) next))) else (ctor UlebEvent.more (record UlebState {cursor := next, start := (project u start), value := (if ((project u consumed) =? 9) then (project u value) else ((project u value) + (prim multiply (prim remainder (prim to_nat b : Nat) 128 0 : Nat) (project u factor) : Nat))), factor := (if ((project u consumed) =? 9) then (project u factor) else (prim multiply (project u factor) 128 : Nat)), consumed := ((project u consumed) + 1)})))
    | ReadEvent.end next => (ctor UlebEvent.failed (record LocatedError {kind := (ctor ReadError.truncated), offset := ((project u start) + (project u consumed))}))
    | ReadEvent.failed error => (ctor UlebEvent.failed error))),
  axioms: [Quot.sound, propext]
}

definition{
  name: runUleb,
  parameters: [fuel : Nat, u : UlebState],
  result: UlebEvent,
  body: (match fuel
    | Nat.zero => (ctor UlebEvent.more u)
    | Nat.succ remaining => (match (ulebStep u)
      | UlebEvent.more next => (runUleb remaining next)
      | UlebEvent.done value next => (ctor UlebEvent.done value next)
      | UlebEvent.failed error => (ctor UlebEvent.failed error))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}
