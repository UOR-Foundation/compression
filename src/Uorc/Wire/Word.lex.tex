definition{
  name: littleEndian,
  parameters: [fuel : Nat, data : Bytes, at : Nat],
  result: Nat,
  body: (match fuel
    | Nat.zero => 0
    | Nat.succ m => ((match (prim index data at : (Option UInt8))
      | Option.none => (byteRange)
      | Option.some octet => (prim to_nat octet : Nat)) + (prim multiply (byteRange) (littleEndian m data (1 + at)) : Nat))),
  axioms: [Quot.sound, propext],
  recursive_argument: fuel
}
