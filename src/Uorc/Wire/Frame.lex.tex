definition{
  name: startPreparation,
  parameters: [frame : Frame],
  result: PrepareState,
  body: (record PrepareState {frame := frame, remainingLengths := (project frame objectLengths), objectCount := 0, outputLength := 0, measure := (match (project frame body)
    | Body.raw ignoredRaw => (record Uorc.Wire.GraphEmit::State {pending := (nil Uorc.Wire.GraphEmit::Task), position := 0, work := 0})
    | Body.program graph => (Uorc.Wire.GraphEmit::start graph)), phase := (ctor PreparePhase.lengths)})
}

definition{
  name: prepareStep,
  parameters: [s : PrepareState],
  result: PrepareEvent,
  body: (match (project s phase)
    | PreparePhase.lengths => (match (project s remainingLengths)
      | List.nil => (if (match (project (project s frame) scope)
        | Scope.selfContained => true
        | Scope.conditional identity => ((prim length identity : Nat) =? 32)) then (ctor PrepareEvent.advanced (record PrepareState {frame := (project s frame), remainingLengths := (nil Nat), objectCount := (project s objectCount), outputLength := (project s outputLength), measure := (project s measure), phase := (ctor PreparePhase.body)})) else (ctor PrepareEvent.failed (ctor PrepareError.basisIdentityLength)))
      | List.cons headLength tailLengths => (ctor PrepareEvent.advanced (record PrepareState {frame := (project s frame), remainingLengths := tailLengths, objectCount := ((project s objectCount) + 1), outputLength := ((project s outputLength) + headLength), measure := (project s measure), phase := (ctor PreparePhase.lengths)})))
    | PreparePhase.body => (match (project (project s frame) body)
      | Body.raw rawBytes => (if ((Uorc.Core.Bytes::length rawBytes) =? (project s outputLength)) then (ctor PrepareEvent.complete (record Prepared {frame := (project s frame), objectCount := (project s objectCount), outputLength := (project s outputLength), bodyLength := (project s outputLength)})) else (ctor PrepareEvent.failed (ctor PrepareError.rawLengthMismatch)))
      | Body.program ignoredGraph => (match (Uorc.Wire.GraphEmit::measureStep (project s measure))
        | Uorc.Wire.GraphEmit::Event.advanced nextMeasure => (ctor PrepareEvent.advanced (record PrepareState {frame := (project s frame), remainingLengths := (nil Nat), objectCount := (project s objectCount), outputLength := (project s outputLength), measure := nextMeasure, phase := (ctor PreparePhase.body)}))
        | Uorc.Wire.GraphEmit::Event.produced unexpectedBytes nextMeasure => (ctor PrepareEvent.failed (ctor PrepareError.invalidEmitter))
        | Uorc.Wire.GraphEmit::Event.failed reason position events => (ctor PrepareEvent.failed (ctor PrepareError.invalidEmitter))
        | Uorc.Wire.GraphEmit::Event.complete size events => (ctor PrepareEvent.complete (record Prepared {frame := (project s frame), objectCount := (project s objectCount), outputLength := (project s outputLength), bodyLength := size})))))
}

definition{
  name: prepareFixture,
  parameters: [fuel : Nat, s : PrepareState],
  result: PrepareEvent,
  body: (match fuel
    | Nat.zero => (ctor PrepareEvent.advanced s)
    | Nat.succ remaining => (match (prepareStep s)
      | PrepareEvent.advanced next => (prepareFixture remaining next)
      | PrepareEvent.complete preparedValue => (ctor PrepareEvent.complete preparedValue)
      | PrepareEvent.failed reason => (ctor PrepareEvent.failed reason))),
  recursive_argument: fuel
}
