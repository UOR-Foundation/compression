Uorc.Uorc.Integrity.IncrementalLaws.begin_update_success_exact (state : Uorc.Uorc.Integrity.Sha256.State)
  (input : ByteArray) (job : Uorc.Uorc.Integrity.Sha256.Update)
  (success : Uorc.Uorc.Integrity.Sha256.beginUpdate state input = Except.ok job) :
  job.state = state ∧
    job.input = input ∧
      job.offset = 0 ∧
        Uorc.Uorc.Integrity.Sha256.valid state = true ∧ state.phase = Uorc.Uorc.Integrity.Sha256.Phase.absorbing
Uorc.Uorc.Integrity.IncrementalLaws.update_step_done_exact (job : Uorc.Uorc.Integrity.Sha256.Update)
  (out : Uorc.Uorc.Integrity.Sha256.State)
  (success : Uorc.Uorc.Integrity.Sha256.updateStep job = Except.ok (Uorc.Uorc.Integrity.Sha256.UpdateEvent.done out)) :
  out = job.state ∧
    job.offset = LexLeanRuntime.length job.input ∧
      Uorc.Uorc.Integrity.Sha256.valid out = true ∧
        Uorc.Uorc.Integrity.Sha256.isAbsorbing out = true ∧ Uorc.Uorc.Integrity.Sha256.hasWork out = false
Uorc.Uorc.Integrity.IncrementalLaws.update_step_advanced_exact (job next : Uorc.Uorc.Integrity.Sha256.Update)
  (success :
    Uorc.Uorc.Integrity.Sha256.updateStep job = Except.ok (Uorc.Uorc.Integrity.Sha256.UpdateEvent.advanced next)) :
  Uorc.Uorc.Integrity.Sha256.valid job.state = true ∧
    Uorc.Uorc.Integrity.Sha256.isAbsorbing job.state = true ∧
      job.offset ≤ LexLeanRuntime.length job.input ∧
        next.input = job.input ∧
          (Uorc.Uorc.Integrity.Sha256.hasWork job.state = true ∧
              Uorc.Uorc.Integrity.Sha256.advanceWork job.state = Except.ok next.state ∧ next.offset = job.offset ∨
            Uorc.Uorc.Integrity.Sha256.hasWork job.state = false ∧
              job.offset < LexLeanRuntime.length job.input ∧
                Uorc.Uorc.Integrity.Sha256.absorbByte job.state
                      (Uorc.Uorc.Integrity.IncrementalLaws.update_step_advanced_exact.match_1 (fun x => UInt8)
                        (LexLeanRuntime.index job.input job.offset) (fun _ => 0) fun unwrapped => unwrapped) =
                    Except.ok next.state ∧
                  next.offset = job.offset + 1)
