definition{
  name: admission,
  parameters: [argOperation : Uorc.Product.Evidence::Operation, report : Report],
  result: Uorc.Product.Evidence::PublicationDecision,
  body: (match argOperation
    | Uorc.Product.Evidence::Operation.compress => (if ((project (project report facts) parsed) && ((project (project report facts) integrityChecked) && ((project (project report facts) executed) && ((project report boundariesComplete) && ((project report originalsAvailable) && ((project (project report facts) originalCompared) && true)))))) then (Uorc.Product.Evidence::publicationDecision (project report finalCheck) false (certificateCheck (project report certificate))) else (ctor Uorc.Product.Evidence::PublicationDecision.verificationIncomplete))
    | Uorc.Product.Evidence::Operation.decompress => (Uorc.Product.Evidence::decompressionPublicationDecision (project (project report facts) parsed) (project (project report facts) integrityChecked) (project (project report facts) executed) (project report boundariesComplete) false)
    | Uorc.Product.Evidence::Operation.inspect => (if (project (project report facts) parsed) then (ctor Uorc.Product.Evidence::PublicationDecision.publish) else (ctor Uorc.Product.Evidence::PublicationDecision.verificationIncomplete))
    | Uorc.Product.Evidence::Operation.verify => (if (project report originalsAvailable) then (if ((project (project report facts) parsed) && ((project (project report facts) integrityChecked) && ((project (project report facts) executed) && ((project (project report facts) originalCompared) && ((project report boundariesComplete) && true))))) then (Uorc.Product.Evidence::publicationDecision (project report finalCheck) false (certificateCheck (project report certificate))) else (ctor Uorc.Product.Evidence::PublicationDecision.verificationIncomplete)) else (Uorc.Product.Evidence::decompressionPublicationDecision (project (project report facts) parsed) (project (project report facts) integrityChecked) (project (project report facts) executed) (project report boundariesComplete) false))
    | Uorc.Product.Evidence::Operation.resume => (if ((project (project report facts) parsed) && ((project (project report facts) integrityChecked) && ((project (project report facts) executed) && ((project report boundariesComplete) && ((project report originalsAvailable) && ((project (project report facts) originalCompared) && true)))))) then (Uorc.Product.Evidence::publicationDecision (project report finalCheck) false (certificateCheck (project report certificate))) else (ctor Uorc.Product.Evidence::PublicationDecision.verificationIncomplete))
    | Uorc.Product.Evidence::Operation.buildBasis => (if (project report artifactChecked) then (ctor Uorc.Product.Evidence::PublicationDecision.publish) else (ctor Uorc.Product.Evidence::PublicationDecision.verificationIncomplete))
    | Uorc.Product.Evidence::Operation.analyzeFrontier => (ctor Uorc.Product.Evidence::PublicationDecision.publish)
    | Uorc.Product.Evidence::Operation.compare => (ctor Uorc.Product.Evidence::PublicationDecision.publish)
    | Uorc.Product.Evidence::Operation.runAcceptance => (ctor Uorc.Product.Evidence::PublicationDecision.publish)
    | Uorc.Product.Evidence::Operation.runCoreRequest => (ctor Uorc.Product.Evidence::PublicationDecision.publish)),
  executable: true
}
