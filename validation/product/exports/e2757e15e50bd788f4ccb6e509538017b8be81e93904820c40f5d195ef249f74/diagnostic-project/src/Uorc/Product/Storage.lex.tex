definition{
  name: step,
  parameters: [limits : Limits, state : State, event : Event],
  result: State,
  body: (match event
    | Event.reserve category bytes => (match (reserve limits (project state usage) category bytes)
      | Result.error error => (record State {usage := (project state usage), error := (ctor Option.some [Error] error)})
      | Result.ok usage => (record State {usage := usage, error := (ctor Option.none [Error])}))
    | Event.written category reservation achieved => (match (recordWrite (project state usage) category reservation achieved)
      | Result.error error => (record State {usage := (project state usage), error := (ctor Option.some [Error] error)})
      | Result.ok usage => (record State {usage := usage, error := (ctor Option.none [Error])}))
    | Event.failedWrite category reservation achieved => (match (recordWrite (project state usage) category reservation achieved)
      | Result.error error => (record State {usage := (project state usage), error := (ctor Option.some [Error] error)})
      | Result.ok usage => (record State {usage := usage, error := (ctor Option.none [Error])}))
    | Event.release category bytes => (match (release (project state usage) category bytes)
      | Result.error error => (record State {usage := (project state usage), error := (ctor Option.some [Error] error)})
      | Result.ok usage => (record State {usage := usage, error := (ctor Option.none [Error])}))
    | Event.transfer category bytes => (match (release (project state usage) category bytes)
      | Result.error error => (record State {usage := (project state usage), error := (ctor Option.some [Error] error)})
      | Result.ok usage => (record State {usage := usage, error := (ctor Option.none [Error])}))),
  executable: true
}
