definition{
  name: publicationDecision,
  parameters: [finalCheck : FinalCheck, requireCertificate : Bool, certificate : CertificateCheck],
  result: PublicationDecision,
  body: (match finalCheck
    | FinalCheck.equal => (if requireCertificate then (if (match certificate
      | CertificateCheck.absent => false
      | CertificateCheck.valid => true
      | CertificateCheck.invalid => false
      | CertificateCheck.incomplete => false) then (ctor PublicationDecision.publish) else (ctor PublicationDecision.certificateUnmet)) else (ctor PublicationDecision.publish))
    | FinalCheck.different => (ctor PublicationDecision.byteMismatch)
    | FinalCheck.incomplete => (ctor PublicationDecision.verificationIncomplete)
    | FinalCheck.cancelled => (ctor PublicationDecision.cancelled)),
  executable: true
}
