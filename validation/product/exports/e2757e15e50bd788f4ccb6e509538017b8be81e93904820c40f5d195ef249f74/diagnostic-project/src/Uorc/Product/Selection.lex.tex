structure{
  name: Item,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: id, type: Nat},

    {name: observedLength, type: (Option Nat)}
  ]

definition{
  name: count,
  parameters: [items : (List Item)],
  result: Nat,
  body: (match items | List.nil => 0 | List.cons head tail => (1 + (count tail))),
  executable: true,
  recursive_argument: items
}

definition{
  name: contains,
  parameters: [items : (List Item), selectedItemId : Nat],
  result: Bool,
  body: (match items
    | List.nil => false
    | List.cons head tail => (if ((project head id) =? selectedItemId) then true else (contains tail selectedItemId))),
  executable: true,
  recursive_argument: items
}

definition{
  name: valid,
  parameters: [items : (List Item)],
  result: Bool,
  body: (match items
    | List.nil => true
    | List.cons head tail => (if (contains tail (project head id)) then false else (valid tail))),
  executable: true,
  recursive_argument: items
}

definition{
  name: remove,
  parameters: [items : (List Item), selectedItemId : Nat],
  result: (List Item),
  body: (match items
    | List.nil => (nil Item)
    | List.cons head tail => (if ((project head id) =? selectedItemId) then tail else (cons head (remove tail selectedItemId)))),
  executable: true,
  recursive_argument: items
}

definition{
  name: find,
  parameters: [items : (List Item), selectedItemId : Nat],
  result: (Option Item),
  body: (match items
    | List.nil => (ctor Option.none [Item])
    | List.cons head tail => (if ((project head id) =? selectedItemId) then (ctor Option.some [Item] head) else (find tail selectedItemId))),
  executable: true,
  recursive_argument: items
}

definition{
  name: insertAt,
  parameters: [items : (List Item), position : Nat, item : Item],
  result: (List Item),
  body: (match items
    | List.nil => (cons item (nil Item))
    | List.cons head tail => (if (position =? 0) then (cons item (cons head tail)) else (cons head (insertAt tail (prim subtract position 1 : Nat) item)))),
  executable: true,
  recursive_argument: items
}

definition{
  name: moveTo,
  parameters: [items : (List Item), selectedItemId : Nat, position : Nat],
  result: (List Item),
  body: (match (find items selectedItemId)
    | Option.none => items
    | Option.some item => (insertAt (remove items selectedItemId) position item)),
  executable: true
}

definition{
  name: sameOrder,
  parameters: [left : (List Item), right : (List Item)],
  result: Bool,
  body: (match left
    | List.nil => (match right | List.nil => true | List.cons head tail => false)
    | List.cons leftHead leftTail => (match right
      | List.nil => false
      | List.cons rightHead rightTail => (if ((project leftHead id) =? (project rightHead id)) then (sameOrder leftTail rightTail) else false))),
  executable: true,
  recursive_argument: left
}

  executable: true
}

  name: valid_cons_binding,
  parameters: [head : Item, tail : (List Item)],
